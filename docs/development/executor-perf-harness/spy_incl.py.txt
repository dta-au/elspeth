"""Top inclusive frames (function-level) matching a filter. spy_incl.py FILE FILTER [N] [under-pattern]"""
import sys, collections, re
flt = re.compile(sys.argv[2]); N = int(sys.argv[3]) if len(sys.argv) > 3 else 40
under = re.compile(sys.argv[4]) if len(sys.argv) > 4 else None
inc = collections.Counter(); total = 0
for line in open(sys.argv[1]):
    s, _, n = line.rstrip("\n").rpartition(" "); n = int(n); total += n
    fr = s.split(";")
    if under:
        idx = next((i for i, f in enumerate(fr) if under.search(f)), None)
        if idx is None: continue
        fr = fr[idx:]
    seen = set()
    for f in fr:
        k = re.sub(r":\d+\)$", ")", f)
        if flt.search(k) and k not in seen:
            seen.add(k); inc[k] += n
for k, v in inc.most_common(N): print(f"{100*v/total:6.1f}% {v:6d} {k}")
