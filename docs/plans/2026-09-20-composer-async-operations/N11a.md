    factory: Callable[[UserIdentity], PluginAvailabilitySnapshot],

async def _running_job(
    service: SessionServiceImpl,
    engine: Engine,
    session_id: uuid.UUID,
    *,
    kind: ComposerOperationKind = "compose_message",
    content: str = "Build a pipeline.",
    base_state_id: uuid.UUID | None = None,
    expected_user_message_id: uuid.UUID | None = None,
) -> AsyncIterator[_Job]:
    """Admit, claim, start and adopt one job exactly as the N12 worker will."""
    operation_id = str(uuid.uuid4())
    request: SendMessageRequest | RecomposeRequest
    if kind == "compose_message":
        request = SendMessageRequest(operation_id=operation_id, content=content, state_id=base_state_id)
    else:
        assert expected_user_message_id is not None
        request = RecomposeRequest(
            operation_id=operation_id,
            expected_user_message_id=expected_user_message_id,
            state_id=base_state_id,
        )

        engine,
        owner_instance_id=service.session_operation_owner_instance_id,
        claim_lease_seconds=30,
    )
    await run_sync_in_worker(
        authority.admit,
        session_id=session_id,
        operation_id=operation_id,
        kind=kind,
        request_hash=composer_operation_request_hash(session_id=session_id, kind=kind, request=request),
        actor_user_id="alice",
        request_id="req-n11a",

        request_json=request.model_dump_json(),
        deadline_seconds=85.0,
        max_nonterminal=64,
    )

            authority=authority,
        )

def _verb_table(statement: str) -> tuple[str, str]:
    """``(verb, table)`` for an INSERT or UPDATE; ``(verb, "")`` otherwise. No SQL literal lives in this file."""
    words = statement.split()
    verb = words[0].upper()
    if verb == "INSERT":
        return verb, words[2]
    if verb == "UPDATE":
        return verb, words[1]
    return verb, ""


@contextlib.contextmanager
def _captured(engine: Engine) -> Iterator[list[str]]:
    statements: list[str] = []

    def capture(_conn: Any, _cursor: Any, statement: str, _parameters: Any, _context: Any, _executemany: bool) -> None:
        statements.append(" ".join(statement.split()))

    event.listen(engine, "before_cursor_execute", capture)
    try:
        yield statements
    finally:
        event.remove(engine, "before_cursor_execute", capture)

def composer_operation_send_base_on_connection(
    conn: Connection,
    running: ComposerOperationRunning,
) -> UUID | None:
    """Read a running send job's bound base on the caller's transaction. Select-only; no DML.

    ``SessionServiceImpl.add_message_with_transcript`` calls this inside the
    send's user-row transaction (contract E8). The ingress receipt's requested
    state is then proved equal to the base that the request hash and the start
    composite bound (ruling 5). The row must be this claim's running job under
    this exact fence triple; anything else is a lost fence. A running job that
    is not a send is Tier-1: recompose writes no user row and no ingress row.
    """
    claim = running.claim
    fence = running.session_operation_context.fence
    table = composer_async_operations_table
    row = conn.execute(
        select(table.c.kind, table.c.base_state_id).where(
            table.c.session_id == str(claim.session_id),
            table.c.operation_id == claim.operation_id,
            table.c.status == "running",
            table.c.claim_token == claim.claim_token,
            table.c.attempt == claim.attempt,
            table.c.session_operation_id == fence.operation_id,
            table.c.session_operation_lease_token == fence.lease_token,
            table.c.session_operation_epoch == fence.operation_epoch,
        )

        self,
        session_id: UUID,
        role: ChatMessageRole,
        content: str,
        *,
        client_request_id: UUID,
        requested_state_id: UUID | None,
        writer_principal: ChatMessageWriterPrincipal,

        raw_content: str | None = None,
        tool_call_id: str | None = None,
        parent_assistant_id: UUID | None = None,
        session_operation_context: SessionOperationContext,
        running: ComposerOperationRunning | None = None,
    ) -> MessageIngressFresh | MessageIngressAccepted | MessageIngressConflict:
        """Accept one freeform user message and read its transcript in one transaction.

        Write-then-read split across two pooled connections is how the
        freeform send path produced false Tier-1 ``AuditIntegrityError``
        500s: ``add_message`` committed on one connection while the
        follow-up ``get_messages`` snapshot ran on another, so a stale
        reader (read/write-splitting proxy, pinned snapshot) could return
        a transcript that did not yet contain the committed insert. This
        method performs the insert and the transcript SELECT on the SAME
        connection inside a single ``_session_process_locked_begin``
        transaction holding ``_session_write_lock``, so the returned
        transcript contains the inserted row as its maximum
        ``sequence_no`` BY CONSTRUCTION — no retry or tolerance semantics
        are involved, and none may be added.

        Behaviour is otherwise identical to :meth:`add_message` (state
        cross-session guard, sequence allocation under the write lock,
        and ``sessions.updated_at`` bump).

        A same-session receipt binds the client key to the original content
        and nullable requested state before sequence allocation. An exact
        duplicate is an acceptance receipt, never a composition replay.

        ``running`` is the composer async worker's path (contract E8). The
        send's job is bound in this same transaction, in the order user row,
        then ``composer_async_operations.user_message_id``
        (``bind_composer_operation_user_message_on_connection``), then the
        ingress receipt keyed by the operation id. The receipt's requested
        state must be the job's bound base. On that path any non-fresh outcome
        is a Tier-1 ``AuditIntegrityError``: a retry of one operation never
        creates a second user row. ``running=None`` is the still-synchronous
        route's path, unchanged until the cutover (N14) makes ``running``
        required.

            raise ValueError("message ingress requires a route-owned user message")

            raise ValueError("message ingress user row cannot carry provider or tool metadata")

                    f"{running.claim.session_id}, not {session_id}"

                raise AuditIntegrityError(
                    f"Tier 1: message ingress key {client_request_id} is not composer operation {running.claim.operation_id}"
                )

                    f"Tier 1: composer operation {running.claim.operation_id} was handed a session operation "
                    "context that is not its bound fence"

                self._session_process_locked_begin(sid) as conn,
                self._session_write_lock(conn, sid),
                self._session_composer_mutation_transaction(
                    conn,
                    session_id=sid,
                    session_operation_context=session_operation_context,
                    expected_kind=SessionOperationKind.COMPOSE,
                ),
            ):

                    bound_base_state_id = composer_operation_send_base_on_connection(conn, running)
                    if bound_base_state_id != requested_state_id:
                        raise AuditIntegrityError(
                            f"Tier 1: composer operation {running.claim.operation_id} is bound to base "
                            f"{bound_base_state_id}, but its message ingress requested {requested_state_id}"
                        )
                existing = self._existing_message_ingress_result(
                    conn,
                    session_id=sid,
                    client_request_id=client_request_id,
                    content=content,
                    requested_state_id=requested_sid,
                )
                if existing is not None:
                    return existing
                if requested_sid is not None:
                    _assert_state_in_session(
                        conn,
                        state_id=requested_sid,
                        expected_session_id=sid,
                        caller="add_message_with_transcript",
                    )
                if csid is not None:
                    _assert_state_in_session(
                        conn,
                        state_id=csid,
                        expected_session_id=sid,
                        caller="add_message_with_transcript",
                    )
                seq = self._reserve_sequence_range(conn, sid, count=1)
                msg_id_holder["id"] = self._insert_chat_message(
                    conn,
                    session_id=sid,
                    role=role,
                    content=content,
                    raw_content=raw_content,
                    # Same deep_thaw rationale as ``add_message``:
                    # tool_calls may be a frozen mapping/tuple shape the
                    # JSON encoder rejects.
                    tool_calls=deep_thaw(tool_calls) if tool_calls else None,
                    sequence_no=seq,
                    writer_principal=writer_principal,
                    composition_state_id=csid,
                    tool_call_id=tool_call_id,
                    parent_assistant_id=pid,
                    created_at=now,
                    session_operation_context=session_operation_context,
                )
                if running is not None:
                    # E8 order: the job's user_message_id is set before the
                    # receipt, which the ingress->job composite FK (N14) needs.
                    bind_composer_operation_user_message_on_connection(
                        conn,
                        running,
                        user_message_id=UUID(msg_id_holder["id"]),
                    )

                    conn,
                    session_id=sid,
                    client_request_id=request_id,

                # SAME transaction as the insert — this read sees its own
                # write on every dialect, which is the entire point.
                message_rows = conn.execute(
                    select(chat_messages_table).where(chat_messages_table.c.session_id == sid).order_by(chat_messages_table.c.sequence_no)
                ).fetchall()
                return message_rows

        if isinstance(sync_result, (MessageIngressAccepted, MessageIngressConflict)):
            if running is not None:
                raise AuditIntegrityError(
                    "Tier 1: a message ingress receipt already exists for composer operation "
                    f"{running.claim.operation_id}; the worker writes a send's user row once per job"
                )
            return sync_result

        if not transcript or str(transcript[-1].id) != msg_id_holder["id"]:
            # By construction (single transaction, session write lock held
            # for the sequence allocation) the inserted row IS the maximum
            # sequence_no in the snapshot just read. If this fires, the
            # store violated read-your-own-write inside one transaction —
            # genuine engine/store corruption, not a timing artifact.
            raise AuditIntegrityError(
                "Tier 1 audit anomaly: add_message_with_transcript same-"
                f"transaction snapshot for session {sid} does not end at "
                f"inserted message {msg_id_holder['id']}."
            )
        message = replace(transcript[-1], client_request_id=client_request_id)
        transcript[-1] = message
        return MessageIngressFresh(client_request_id=client_request_id, message=message, transcript=tuple(transcript))
