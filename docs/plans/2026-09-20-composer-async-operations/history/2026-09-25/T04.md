def _envelope(
    *,
    http_status: int,
    failure_code: ComposerOperationFailureCode,
    detail: dict[str, JsonValue],
    request_id: str | None,
) -> ComposerOperationError:
    """A D14-shaped envelope: dict detail with the persisted request id injected (D4)."""
    body_detail: dict[str, JsonValue] = dict(detail)
    if request_id is not None:
        body_detail["request_id"] = request_id
    error_type = detail["error_type"]
    assert type(error_type) is str
    return ComposerOperationError(
        http_status=http_status,
        failure_code=failure_code,
        error_type=error_type,
        body={"detail": body_detail},
        diagnostic_id=None,
    )


def _cancelled_failure(*, request_id: str | None) -> ComposerOperationError:
    return _envelope(
        http_status=499,
        failure_code="request_cancelled",
        detail={"error_type": "request_cancelled", "detail": "The composer request was stopped."},
        request_id=request_id,
    )


def _worker_lost(*, request_id: str | None) -> ComposerOperationError:
    return _envelope(
        http_status=503,
        failure_code="worker_lost",
        detail={
            "error_type": "composer_operation_worker_lost",
            "detail": "The server stopped while composing this request. Reload to see what was saved, then resubmit.",
        },
        request_id=request_id,
    )


def _deadline_expired(*, request_id: str | None) -> ComposerOperationError:
    return _envelope(
        http_status=504,
        failure_code="deadline_expired",
        detail={
            "error_type": "composer_operation_deadline_expired",

def _rewind(engine: Engine, *, session_id: UUID, operation_id: str, seconds: int, columns: tuple[str, ...]) -> None:
    """Move stored timestamps into the past TOGETHER, so every ordering CHECK among them still holds."""
    key = (_OPERATIONS.c.session_id == str(session_id), _OPERATIONS.c.operation_id == operation_id)
    with engine.begin() as conn:
        row = conn.execute(select(_OPERATIONS).where(*key)).one()
        conn.execute(
            update(_OPERATIONS).where(*key).values({column: row._mapping[column] - timedelta(seconds=seconds) for column in columns})
        )


def _mark_running(engine: Engine, claim: ComposerOperationClaim, context: SessionOperationContext) -> None:
    """Stand-in for Task 5's composite start: bind the claimed row to one COMPOSE context."""
    with engine.begin() as conn:
        changed = conn.execute(
            update(_OPERATIONS)
            .where(
                _OPERATIONS.c.session_id == str(claim.session_id),
                _OPERATIONS.c.operation_id == claim.operation_id,
                _OPERATIONS.c.status == "queued",
                _OPERATIONS.c.claim_token == claim.claim_token,
            )
            .values(
                status="running",
                session_operation_id=context.fence.operation_id,
                session_operation_lease_token=context.fence.lease_token,
                session_operation_epoch=context.fence.operation_epoch,
                started_at=_OPERATIONS.c.created_at,
            )
        ).rowcount
    assert changed == 1


def _expire_fence(engine: Engine, *, session_id: UUID) -> None:
    with engine.begin() as conn:
        conn.execute(
            update(session_operation_fences_table)
            .where(session_operation_fences_table.c.session_id == str(session_id))
            .values(lease_expires_at=datetime(2000, 1, 1, tzinfo=UTC))
        )


def _operation_count(engine: Engine) -> int:
    with engine.connect() as conn:
        count = conn.execute(select(func.count()).select_from(_OPERATIONS)).scalar_one()
    assert type(count) is int
    return count


def _race[T](*contenders: Callable[[], T]) -> list[T]:
    barrier = threading.Barrier(len(contenders))

    def run(contender: Callable[[], T]) -> T:
        barrier.wait(timeout=10)
        return contender()

    with ThreadPoolExecutor(max_workers=len(contenders)) as pool:
        return list(pool.map(run, contenders))


def _running_job(
    engine: Engine,
    sessions: SessionRepository,
    authority: ComposerAsyncOperationAuthority,
    *,
    dead_owner: str,
) -> tuple[UUID, ComposerOperationRecord, SessionOperationContext]:
    session_id = _new_session(sessions)
    record, _ = _admit(authority, session_id=session_id)
    (claim,) = authority.claim_next(limit=1)
    owner = sessions.acquire(
        session_id=session_id,
        operation_kind=SessionOperationKind.COMPOSE,
        owner_instance_id=dead_owner,
        lease_seconds=30,
    )
    _mark_running(engine, claim, owner)
    return session_id, record, owner


def _archive_session(engine: Engine, *, session_id: UUID) -> None:
    """Leave the state a committed soft archive leaves: ``sessions.archived_at`` set (D7).

    The production path (``decide_and_soft_archive`` under an ARCHIVE fence,
    ``repository.py:718-817``) needs an ARCHIVE acquire and durable history. A
    direct UPDATE is the honest stand-in, as ``_expire_fence`` is for lease
    expiry. The fence is controlled separately, so each guard is proved alone.
    """
    with engine.begin() as conn:
        archived_at = datetime.now(UTC)
        changed = conn.execute(
            update(sessions_table).where(sessions_table.c.id == str(session_id)).values(archived_at=archived_at, updated_at=archived_at)
        ).rowcount
    assert changed == 1


type AdmitOutcome = tuple[ComposerOperationRecord, bool] | ComposerOperationActiveError | ComposerOperationCapacityError


def _attempt(admit: Callable[[], tuple[ComposerOperationRecord, bool]]) -> Callable[[], AdmitOutcome]:
    """Return an expected admission refusal instead of raising it through ``pool.map``."""

    def run() -> AdmitOutcome:
        try:
            return admit()
        except (ComposerOperationActiveError, ComposerOperationCapacityError) as refused:
            return refused

    return run


def _split_outcomes(
    outcomes: list[AdmitOutcome],
) -> tuple[list[ComposerOperationRecord], list[ComposerOperationActiveError], list[ComposerOperationCapacityError]]:
    created: list[ComposerOperationRecord] = []
    active: list[ComposerOperationActiveError] = []
    full: list[ComposerOperationCapacityError] = []
    for outcome in outcomes:
        if type(outcome) is ComposerOperationActiveError:
            active.append(outcome)
        elif type(outcome) is ComposerOperationCapacityError:
            full.append(outcome)
        elif type(outcome) is tuple:
            record, was_created = outcome
            assert was_created is True  # every contender used a fresh operation id
            created.append(record)
        else:
            raise AssertionError(f"unexpected admission outcome {outcome!r}")
    return created, active, full


def _assert_one_admit_takes_the_active_slot(engine: Engine, sessions: SessionRepository) -> None:
    """m4 (Review Focus #4, two tabs): different ids race for one session's D8 slot."""
    session_id = _new_session(sessions)
    contenders = [ComposerAsyncOperationAuthority(engine, owner_instance_id=f"tab-{name}", claim_lease_seconds=30) for name in ("a", "b")]
    kinds: tuple[ComposerOperationKind, ...] = ("compose_message", "compose_recompose")

    outcomes = _race(
        *(_attempt(partial(_admit, contender, session_id=session_id, kind=kind)) for contender, kind in zip(contenders, kinds, strict=True))
    )

    created, active, full = _split_outcomes(outcomes)
    assert (len(created), len(active), full) == (1, 1, [])
    (winner,) = created
    assert (active[0].operation_id, active[0].kind) == (winner.operation_id, winner.kind)
    assert _operation_count(engine) == 1
    assert contenders[0].get(session_id=session_id, operation_id=winner.operation_id) == winner


def _capacity_race(engine: Engine, sessions: SessionRepository, *, max_nonterminal: int, admitters: int) -> int:
    """F-C7: ``admitters`` first admissions, each to its OWN session, race for the cluster cap.

    Returns how many were created. Asserts that every contender either
    created its row or was refused with the capacity error, and that the
    committed count moved by exactly the number created.
    """
    before = ComposerAsyncOperationAuthority(engine, owner_instance_id="counter", claim_lease_seconds=30).count_nonterminal()
    contenders = [
        ComposerAsyncOperationAuthority(engine, owner_instance_id=f"admitter-{index}", claim_lease_seconds=30) for index in range(admitters)
    ]
    session_ids = [_new_session(sessions) for _ in contenders]

    outcomes = _race(
        *(
            _attempt(partial(_admit, contender, session_id=session_id, max_nonterminal=max_nonterminal))
            for contender, session_id in zip(contenders, session_ids, strict=True)
        )
    )

    created, active, full = _split_outcomes(outcomes)
    assert active == []
    assert len(created) + len(full) == admitters
    assert all(refusal.retry_after_seconds == 5 for refusal in full)
    assert contenders[0].count_nonterminal() == before + len(created)
    return len(created)


def _assert_fence_blocked_sessions_do_not_starve_a_startable_job(
    engine: Engine, sessions: SessionRepository, authority: ComposerAsyncOperationAuthority
) -> None:
    """F-M2: N older jobs whose sessions hold a live foreign fence, then one startable job; ``limit=N``.

    Without the discovery filter the N older rows fill every scan (they are
    claimed, refused by the start composite, released and re-claimed) until
    their deadline, and the startable job ages toward 504 behind them.
    """
    blocked: list[tuple[UUID, str, SessionOperationContext]] = []
    for kind in (SessionOperationKind.EXECUTE, SessionOperationKind.COMPOSE):
        session_id = _new_session(sessions)
        record, _ = _admit(authority, session_id=session_id)
        # Strictly older than the startable job, whatever the clock's resolution.
        _rewind(engine, session_id=session_id, operation_id=record.operation_id, seconds=60, columns=("created_at", "updated_at"))
        foreign = sessions.acquire(session_id=session_id, operation_kind=kind, owner_instance_id="foreign-owner", lease_seconds=300)
        blocked.append((session_id, record.operation_id, foreign))
    free_session = _new_session(sessions)
    startable, _ = _admit(authority, session_id=free_session)

    claims = authority.claim_next(limit=len(blocked))

    assert [(claim.session_id, claim.operation_id) for claim in claims] == [(free_session, startable.operation_id)]
    for session_id, operation_id, _foreign in blocked:
        waiting = authority.get(session_id=session_id, operation_id=operation_id)
        assert waiting is not None
        assert (waiting.status, waiting.claim_owner_instance_id) == ("queued", None)

    # A skip, not a drop: once the foreign operation releases, the job is claimable.
    released_session, released_operation, foreign = blocked[0]
    sessions.release(foreign)
    (reclaimed,) = authority.claim_next(limit=len(blocked))
    assert (reclaimed.session_id, reclaimed.operation_id, reclaimed.attempt) == (released_session, released_operation, 1)


def _assert_inactive_session_settle(engine: Engine, sessions: SessionRepository, authority: ComposerAsyncOperationAuthority) -> None:
    """B1: an archived session's ``running`` job settles ``worker_lost``; nothing else may use this path."""
    session_id, record, _owner = _running_job(engine, sessions, authority, dead_owner="worker-a")
    lost = _worker_lost(request_id=record.request_id)

    def settle() -> ComposerOperationRecord:
        return authority.settle_lost_inactive_session(
            session_id=session_id, operation_id=record.operation_id, failure=lost, cancelled_failure=_cancelled_failure
        )

    with pytest.raises(ComposerOperationFenceLost):
        settle()  # not archived: only settle_lost, under a successor COMPOSE context, may settle this row
    _expire_fence(engine, session_id=session_id)
    with pytest.raises(ComposerOperationFenceLost):
        settle()  # still not archived, even with the owner's fence dead
    _archive_session(engine, session_id=session_id)
    with pytest.raises(SessionOperationFenceLost) as inactive:
        sessions.acquire(session_id=session_id, operation_kind=SessionOperationKind.COMPOSE, owner_instance_id="reaper", lease_seconds=30)
    assert inactive.value.reason is FenceLossReason.OWNER_INACTIVE  # why settle_lost can never run for this row
    assert [row.operation_id for row in authority.list_expired_running(limit=10)] == [record.operation_id]
    assert authority.count_nonterminal() == 1
    with pytest.raises(ValueError, match="request_cancelled"):
        authority.settle_lost_inactive_session(
            session_id=session_id,
            operation_id=record.operation_id,
            failure=_cancelled_failure(request_id=record.request_id),
            cancelled_failure=_cancelled_failure,
        )

    settled = settle()

    assert (settled.status, settled.failure_code) == ("failed", "worker_lost")
    assert settled.started_at is not None and settled.settled_at is not None
    assert settled.request_json is None and settled.claim_owner_instance_id is None
    assert settled.result_json is not None
    assert settled.result_sha256 == composer_operation_result_hash(settled.result_json)
    assert ComposerOperationError.model_validate_json(settled.result_json) == lost
    assert authority.count_nonterminal() == 0  # the D3 slot is free again
    assert authority.list_expired_running(limit=10) == ()
    with pytest.raises(ComposerOperationFenceLost):
        settle()  # terminal: refused, never relabelled
    assert authority.get(session_id=session_id, operation_id=record.operation_id) == settled

def _assert_own_lapsed_settle(engine: Engine, sessions: SessionRepository, authority: ComposerAsyncOperationAuthority) -> None:
    """F-M3: the owner settles its own job once its bound fence lapsed, and only then."""
    session_id, record, owner = _running_job(engine, sessions, authority, dead_owner="worker-a")
    lost = _worker_lost(request_id=record.request_id)
    foreign = ComposerAsyncOperationAuthority(engine, owner_instance_id="worker-b", claim_lease_seconds=30)

    def settle(by: ComposerAsyncOperationAuthority, owner_instance_id: str) -> ComposerOperationRecord:
        return by.settle_own_lapsed(
            session_id=session_id,
            operation_id=record.operation_id,
            owner_instance_id=owner_instance_id,
            failure=lost,
            cancelled_failure=_cancelled_failure,
        )

    with pytest.raises(ComposerOperationFenceLost):
        settle(authority, "worker-a")  # the bound fence is live: the owner must use its fenced terminal write
    _expire_fence(engine, session_id=session_id)
    with pytest.raises(ComposerOperationFenceLost):
        settle(foreign, "worker-b")  # not this instance's job
    with pytest.raises(ValueError, match="own instance id"):
        settle(foreign, "worker-a")  # another instance may not settle worker-a's job by naming it
    with pytest.raises(ValueError, match="request_cancelled"):
        authority.settle_own_lapsed(
            session_id=session_id,
            operation_id=record.operation_id,
            owner_instance_id="worker-a",
            failure=_cancelled_failure(request_id=record.request_id),
            cancelled_failure=_cancelled_failure,
        )
    still_running = authority.get(session_id=session_id, operation_id=record.operation_id)
    assert still_running is not None and still_running.status == "running"

    settled = settle(authority, "worker-a")

    assert (settled.status, settled.failure_code) == ("failed", "worker_lost")
    assert settled.result_json is not None
    assert ComposerOperationError.model_validate_json(settled.result_json) == lost
    assert authority.count_nonterminal() == 0
    with engine.connect() as conn:
        fence = conn.execute(
            select(session_operation_fences_table).where(session_operation_fences_table.c.session_id == str(session_id))
        ).one()
    # Read, never advanced, renewed, released or taken over.
    assert (fence.operation_id, fence.lease_token, fence.operation_epoch, fence.released_at) == (
        owner.fence.operation_id,
        owner.fence.lease_token,
        owner.fence.operation_epoch,
        None,
    )
    with pytest.raises(ComposerOperationFenceLost):
        settle(authority, "worker-a")  # terminal: refused, never relabelled

def test_list_expired_running_and_settle_lost_require_a_live_successor_compose_context(
    engine: Engine, sessions: SQLiteLocalSessionOperationAuthority, authority: ComposerAsyncOperationAuthority
) -> None:

    lost = _worker_lost(request_id=record.request_id)

    def settle(context: SessionOperationContext) -> ComposerOperationRecord:
        return authority.settle_lost(
            session_operation_context=context,
            session_id=session_id,

            failure=lost,
            cancelled_failure=_cancelled_failure,
        )

    assert authority.list_expired_running(limit=10) == ()
    with pytest.raises(SessionOperationFenceLost):
        settle(owner)  # the live owner's own context never supersedes its job

    _expire_fence(engine, session_id=session_id)
    (expired,) = authority.list_expired_running(limit=10)

        session_id=session_id, operation_kind=SessionOperationKind.COMPOSE, owner_instance_id="reaper", lease_seconds=30
    )
    try:
        settled = settle(reaper)
        assert settle(reaper) == settled  # a racing reaper reads the terminal row unchanged
    finally:
        sessions.release(reaper)

    assert (settled.status, settled.failure_code) == ("failed", "worker_lost")
    assert settled.started_at == expired.started_at
    assert settled.request_json is None and settled.claim_owner_instance_id is None
    assert settled.result_json is not None

_WORKER_LOST: Final[frozenset[str]] = frozenset({"worker_lost"})
#: The owner of a lapsed job settles the outcome it was publishing: a preserved
#: public error, a generic failure, its own loss, or a deadline. A Stop is never
#: passed in; a committed marker substitutes ``cancelled_failure`` instead.
_OWN_LAPSED_FAILURE_CODES: Final[frozenset[str]] = frozenset({"http_error", "operation_failed", "worker_lost", "deadline_expired"})


class ComposerOperationCancelledFailure(Protocol):
    """Builds the public ``request_cancelled`` envelope for one row's persisted request id.

    Production passes ``composer_operation_errors.request_cancelled_error``.
    The authority calls it inside the settling transaction, so the envelope
    carries the request id stored on the row (D4), never a caller's guess.
    A static callable type only; nothing dispatches on it.
    """

    def __call__(self, *, request_id: str | None) -> ComposerOperationError: ...


def _session_key(session_id: object) -> str:
    if type(session_id) is not UUID:
        raise TypeError("session_id must be a UUID")
    return str(session_id)


def _require_operation_id(operation_id: object) -> str:
    if type(operation_id) is not str or len(operation_id) != 36 or str(UUID(operation_id)) != operation_id:
        raise ValueError("operation_id must be a canonical lowercase UUID string")
    return operation_id


def _require_batch(limit: object) -> int:
    if type(limit) is not int or not 1 <= limit <= _MAX_BATCH:
        raise ValueError(f"limit must be an exact integer from 1 through {_MAX_BATCH}")
    return limit


def _require_claim(claim: object) -> ComposerOperationClaim:
    if type(claim) is not ComposerOperationClaim:
        raise TypeError("claim must be an exact ComposerOperationClaim")
    return claim


def _require_failure(failure: object, allowed: frozenset[str]) -> ComposerOperationError:
    if type(failure) is not ComposerOperationError:
        raise TypeError("failure must be an exact ComposerOperationError")
    if failure.failure_code not in allowed:
        raise ValueError(f"failure_code {failure.failure_code} cannot settle this transition")
    return failure


def _encoded_failure(failure: ComposerOperationError) -> tuple[str, str]:
    """The stored terminal text and its digest: canonical JSON of the strict public envelope."""
    result_json = canonical_json(failure.model_dump(mode="json"))
    return result_json, composer_operation_result_hash(result_json)


def _optional_utc(value: datetime | None) -> datetime | None:
    return None if value is None else _ensure_utc(value)


def _record_from_row(row: Row[Any]) -> ComposerOperationRecord:
    """The Tier-1 read of one row; ``ComposerOperationRecord.__post_init__`` validates it."""
    return ComposerOperationRecord(
        session_id=UUID(row.session_id),
        operation_id=row.operation_id,
        kind=cast("ComposerOperationKind", row.kind),
        status=cast("ComposerOperationStatus", row.status),
        request_hash=row.request_hash,
        actor_user_id=row.actor_user_id,
        request_id=row.request_id,
        deadline_at=_ensure_utc(row.deadline_at),
        created_at=_ensure_utc(row.created_at),
        updated_at=_ensure_utc(row.updated_at),
        started_at=_optional_utc(row.started_at),
        settled_at=_optional_utc(row.settled_at),
        cancel_requested_at=_optional_utc(row.cancel_requested_at),
        claim_owner_instance_id=row.claim_owner_instance_id,
        user_message_id=None if row.user_message_id is None else UUID(row.user_message_id),
        failure_code=cast("ComposerOperationFailureCode | None", row.failure_code),
        result_schema=row.result_schema,
        result_json=row.result_json,
        result_sha256=row.result_sha256,
        request_json=row.request_json,
    )


def _require_same_binding(
    record: ComposerOperationRecord,
    *,
    kind: ComposerOperationKind,
    actor_user_id: str,
    request_hash: str,
) -> None:
    """A replay is the same kind, actor and canonical request; anything else reused the id."""
    if record.kind != kind or record.actor_user_id != actor_user_id or record.request_hash != request_hash:
        raise ComposerOperationConflictError(session_id=record.session_id, operation_id=record.operation_id)


def _row_key(session_id: str, operation_id: str) -> tuple[ColumnElement[bool], ColumnElement[bool]]:
    return (_OPERATIONS.c.session_id == session_id, _OPERATIONS.c.operation_id == operation_id)


def _prior_claim_token(row: Row[Any]) -> ColumnElement[bool]:
    """CAS predicate: the claim token the row held when it was read under the lock."""
    if row.claim_token is None:
        return _OPERATIONS.c.claim_token.is_(None)
    return _OPERATIONS.c.claim_token == row.claim_token


def _claim_is_live(row: Row[Any], now: datetime) -> bool:
    return row.claim_token is not None and _ensure_utc(row.claim_expires_at) > now


def _live_claim(claim: ComposerOperationClaim, owner_instance_id: str, now: datetime) -> tuple[ColumnElement[bool], ...]:
    return (
        *_row_key(str(claim.session_id), claim.operation_id),
        _OPERATIONS.c.status == "queued",
        _OPERATIONS.c.claim_token == claim.claim_token,
        _OPERATIONS.c.attempt == claim.attempt,
        _OPERATIONS.c.claim_owner_instance_id == owner_instance_id,
        _OPERATIONS.c.claim_expires_at > now,
    )


@final
class ComposerAsyncOperationAuthority:
    """Sole writer of admission, claim, cancel-intent, unstarted and lost transitions."""

    __slots__ = ("_claim_lease_seconds", "_clock_sql", "_engine", "_owner_instance_id")

    def __init__(self, engine: Engine, *, owner_instance_id: str, claim_lease_seconds: int) -> None:
        if engine.dialect.name not in _DATABASE_CLOCK_SQL:
            raise NotImplementedError(f"composer async operation authority not implemented for {engine.dialect.name}")
        _require_nonblank(owner_instance_id, "owner_instance_id")
        if type(claim_lease_seconds) is not int or not 5 <= claim_lease_seconds <= 600:
            raise ValueError("claim_lease_seconds must be an exact integer from 5 through 600")
        self._engine = engine
        self._clock_sql = _DATABASE_CLOCK_SQL[engine.dialect.name]
        self._owner_instance_id = owner_instance_id
        self._claim_lease_seconds = claim_lease_seconds

    def admit(
        self,
        *,
        session_id: UUID,
        operation_id: str,
        kind: ComposerOperationKind,
        request_hash: str,
        actor_user_id: str,
        request_id: str | None,
        request_json: str,
        deadline_seconds: float,
        max_nonterminal: int,
    ) -> tuple[ComposerOperationRecord, bool]:
        """Reserve one operation id, or answer its existing row; ``bool`` is ``created``.

        One locked transaction, in D2 order after the caller's charge: the
        existing row by primary key (same kind/actor/hash is a replay,
        anything else is a conflict), then the session's own nonterminal job
        (D8), then the cluster-wide nonterminal count (D3), then the insert.

        D3 is a SOFT cluster cap (review-pass F-C7, accepted). The count and
        the insert run under one session's lock, so on PostgreSQL concurrent
        admissions to different sessions each hold a different advisory lock,
        can all count the same committed rows, and can overshoot
        ``max_nonterminal`` by at most the number of concurrent admitters:
        each admitter inserts at most one row, so after a burst of ``k`` the
        nonterminal count is at most ``max_nonterminal + k - 1``, and once the
        committed count reaches the cap every later admitter is refused. A
        global counter row would make the cap exact only by serializing every
        admission in the cluster, and no new advisory lock is added (spec §3).
        On SQLite the cap is exact: BEGIN IMMEDIATE serializes every writer.
        """
        sid = _session_key(session_id)
        _require_operation_id(operation_id)
        if kind not in COMPOSER_OPERATION_KIND_VALUES:
            raise ValueError("kind is not a composer operation kind")
        if not is_lower_sha256_hex(request_hash):
            raise ValueError("request_hash must be a lowercase SHA-256 hex digest")
        _require_nonblank(actor_user_id, "actor_user_id")
        if request_id is not None:
            _require_nonblank(request_id, "request_id")
        if type(request_json) is not str or not 1 <= len(request_json) <= _MAX_REQUEST_JSON_CHARS:
            raise ValueError(f"request_json must be a string of 1 through {_MAX_REQUEST_JSON_CHARS} characters")
        if type(deadline_seconds) not in (int, float) or not math.isfinite(deadline_seconds) or deadline_seconds <= 0:
            raise ValueError("deadline_seconds must be a finite positive number")
        if type(max_nonterminal) is not int or max_nonterminal < 1:
            raise ValueError("max_nonterminal must be a positive exact integer")
        try:
            with locked_session_transaction(self._engine, sid) as conn:
                now = _database_clock_value(conn.exec_driver_sql(self._clock_sql).scalar_one())
                existing = conn.execute(select(_OPERATIONS).where(*_row_key(sid, operation_id)).with_for_update()).one_or_none()
                if existing is not None:
                    replay = _record_from_row(existing)
                    _require_same_binding(replay, kind=kind, actor_user_id=actor_user_id, request_hash=request_hash)
                    return replay, False
                active = conn.execute(
                    select(_OPERATIONS.c.operation_id, _OPERATIONS.c.kind)
                    .where(_OPERATIONS.c.session_id == sid, _OPERATIONS.c.status.in_(_NONTERMINAL_STATUSES))
                    .order_by(_OPERATIONS.c.created_at, _OPERATIONS.c.operation_id)
                    .limit(1)
                ).one_or_none()
                if active is not None:
                    raise ComposerOperationActiveError(
                        session_id=session_id,
                        operation_id=active.operation_id,
                        kind=cast("ComposerOperationKind", active.kind),
                    )
                nonterminal = conn.execute(
                    select(func.count()).select_from(_OPERATIONS).where(_OPERATIONS.c.status.in_(_NONTERMINAL_STATUSES))
                ).scalar_one()
                if nonterminal >= max_nonterminal:
                    raise ComposerOperationCapacityError(retry_after_seconds=_CAPACITY_RETRY_AFTER_SECONDS)
                inserted = conn.execute(
                    insert(_OPERATIONS)
                    .values(
                        session_id=sid,
                        operation_id=operation_id,
                        kind=kind,
                        status="queued",
                        request_hash=request_hash,
                        actor_user_id=actor_user_id,
                        request_id=request_id,
                        request_json=request_json,
                        deadline_at=now + timedelta(seconds=deadline_seconds),
                        claim_token=None,
                        claim_owner_instance_id=None,
                        claim_expires_at=None,
                        attempt=0,
                        session_operation_id=None,
                        session_operation_lease_token=None,
                        session_operation_epoch=None,
                        user_message_id=None,
                        cancel_requested_at=None,
                        failure_code=None,
                        result_schema=None,
                        result_json=None,
                        result_sha256=None,
                        created_at=now,
                        updated_at=now,
                        started_at=None,
                        settled_at=None,
                    )
                    .returning(_OPERATIONS)
                ).one()
                return _record_from_row(inserted), True
        except IntegrityError:
            # Same-session admits are serialized by the session lock, so a
            # duplicate key here means the lock contract changed; re-read and
            # compare rather than guess. Any other integrity failure (for
            # example a deleted session's foreign key) finds no row and re-raises.
            raced = self.get(session_id=session_id, operation_id=operation_id)
            if raced is None:
                raise
            _require_same_binding(raced, kind=kind, actor_user_id=actor_user_id, request_hash=request_hash)
            return raced, False

    def get(self, *, session_id: UUID, operation_id: str) -> ComposerOperationRecord | None:
        sid = _session_key(session_id)
        _require_operation_id(operation_id)
        with self._engine.connect() as conn:
            row = conn.execute(select(_OPERATIONS).where(*_row_key(sid, operation_id))).one_or_none()
        return None if row is None else _record_from_row(row)

    def _claim_candidates(self, *, limit: int) -> tuple[tuple[str, str], ...]:
        """Optimistic discovery; every candidate is re-decided under its session lock by ``claim_next``."""
        with self._engine.connect() as conn:
            now = _database_clock_value(conn.exec_driver_sql(self._clock_sql).scalar_one())
            rows = conn.execute(
                select(_OPERATIONS.c.session_id, _OPERATIONS.c.operation_id)
                .where(
                    _OPERATIONS.c.status == "queued",
                    or_(_OPERATIONS.c.claim_token.is_(None), _OPERATIONS.c.claim_expires_at <= now),
                    _OPERATIONS.c.deadline_at > now,
                    _OPERATIONS.c.cancel_requested_at.is_(None),
                    # Oldest queued job per session, and none while the session
                    # has a running job. D8 admission keeps one nonterminal job
                    # per session, so this holds trivially today; the predicate
                    # states the contract rather than relying on that.
                    ~select(_BLOCKING_OPERATIONS.c.operation_id)
                    .where(
                        _BLOCKING_OPERATIONS.c.session_id == _OPERATIONS.c.session_id,
                        _BLOCKING_OPERATIONS.c.operation_id != _OPERATIONS.c.operation_id,
                        or_(
                            _BLOCKING_OPERATIONS.c.status == "running",
                            and_(
                                _BLOCKING_OPERATIONS.c.status == "queued",
                                or_(
                                    _BLOCKING_OPERATIONS.c.created_at < _OPERATIONS.c.created_at,
                                    and_(
                                        _BLOCKING_OPERATIONS.c.created_at == _OPERATIONS.c.created_at,
                                        _BLOCKING_OPERATIONS.c.operation_id < _OPERATIONS.c.operation_id,
                                    ),
                                ),
                            ),
                        ),
                    )
                    .exists(),
                    # F-M2: none while the session's fence row is live under ANY
                    # kind (an EXECUTE run, a guided COMPOSE, an archive).
                    # ``acquire`` refuses such a session with
                    # SessionOperationConflictError (repository.py:4801-4803), so
                    # claiming it only claims, conflicts and releases every scan
                    # while startable jobs behind it age toward their deadline.
                    # Advisory, like the rest of discovery: the start composite
                    # still decides.
                    ~select(_FENCES.c.session_id)
                    .where(
                        _FENCES.c.session_id == _OPERATIONS.c.session_id,
                        _FENCES.c.released_at.is_(None),
                        _FENCES.c.lease_expires_at > now,
                    )
                    .exists(),
                )
                .order_by(_OPERATIONS.c.created_at, _OPERATIONS.c.session_id, _OPERATIONS.c.operation_id)
                .limit(limit)
            ).all()
        return tuple((row.session_id, row.operation_id) for row in rows)

    def claim_next(self, *, limit: int) -> tuple[ComposerOperationClaim, ...]:
        """Claim up to ``limit`` startable queued jobs for this instance.

        Discovery is a lock-free read. Each candidate is then claimed in its
        own same-session transaction: re-read with ``FOR UPDATE SKIP LOCKED``
        (PostgreSQL; SQLAlchemy drops the clause on SQLite, where BEGIN
        IMMEDIATE excludes), re-checked against database time, and claimed by
        a compare-and-swap on its prior token and attempt. A job whose
        deadline passed, or that carries a cancel marker, is never claimed:
        it belongs to the reaper (``list_expired_queued``).
        """
        _require_batch(limit)
        claims: list[ComposerOperationClaim] = []
        for session_id, operation_id in self._claim_candidates(limit=limit):
            with locked_session_transaction(self._engine, session_id) as conn:
                now = _database_clock_value(conn.exec_driver_sql(self._clock_sql).scalar_one())
                row = conn.execute(
                    select(_OPERATIONS).where(*_row_key(session_id, operation_id)).with_for_update(skip_locked=True)
                ).one_or_none()
                if (
                    row is None
                    or row.status != "queued"
                    or row.cancel_requested_at is not None
                    or _ensure_utc(row.deadline_at) <= now
                    or _claim_is_live(row, now)
                ):
                    continue
                running = conn.execute(
                    select(_OPERATIONS.c.operation_id).where(_OPERATIONS.c.session_id == session_id, _OPERATIONS.c.status == "running")
                ).first()
                if running is not None:
                    continue
                claim_token = secrets.token_urlsafe(32)
                attempt = row.attempt + 1
                claimed = conn.execute(
                    update(_OPERATIONS)
                    .where(
                        *_row_key(session_id, operation_id),
                        _OPERATIONS.c.status == "queued",
                        _OPERATIONS.c.attempt == row.attempt,
                        _prior_claim_token(row),
                    )
                    .values(
                        claim_token=claim_token,
                        claim_owner_instance_id=self._owner_instance_id,
                        claim_expires_at=now + timedelta(seconds=self._claim_lease_seconds),
                        attempt=attempt,
                        updated_at=now,
                    )
                ).rowcount
                if claimed != 1:
                    raise AuditIntegrityError("Composer operation claim lost its locked compare-and-swap")
            claims.append(
                ComposerOperationClaim(session_id=UUID(session_id), operation_id=operation_id, claim_token=claim_token, attempt=attempt)
            )
        return tuple(claims)

    def renew_claim(self, claim: ComposerOperationClaim) -> ComposerOperationRecord:
        """Extend this instance's exact, still-live queued claim and return the renewed row.

        The row is what a claimed job waiting for the in-process compose lock
        observes (review-pass F-C3): ``cancel_requested_at`` is a committed
        Stop, and ``updated_at`` is the database instant of this renewal, so
        ``record.deadline_at <= record.updated_at`` is a database-time deadline
        test that needs no second clock read. Renewal still succeeds on a
        cancel-marked or deadline-passed claim: the owner needs its live claim
        to settle through ``settle_unstarted(claim, ...)``, which re-verifies
        the marker or the deadline under the lock.
        """
        _require_claim(claim)
        with locked_session_transaction(self._engine, str(claim.session_id)) as conn:
            now = _database_clock_value(conn.exec_driver_sql(self._clock_sql).scalar_one())
            renewed = conn.execute(
                update(_OPERATIONS)
                .where(*_live_claim(claim, self._owner_instance_id, now))
                .values(claim_expires_at=now + timedelta(seconds=self._claim_lease_seconds), updated_at=now)
                .returning(_OPERATIONS)
            ).one_or_none()
            if renewed is None:
                raise ComposerOperationFenceLost(session_id=claim.session_id, operation_id=claim.operation_id)
            return _record_from_row(renewed)

    def release_claim(self, claim: ComposerOperationClaim) -> None:
        """Return this instance's claimed job to the unclaimed queue (the SOL-conflict path)."""
        _require_claim(claim)
        with locked_session_transaction(self._engine, str(claim.session_id)) as conn:
            now = _database_clock_value(conn.exec_driver_sql(self._clock_sql).scalar_one())
            released = conn.execute(
                update(_OPERATIONS)
                .where(
                    *_row_key(str(claim.session_id), claim.operation_id),
                    _OPERATIONS.c.status == "queued",
                    _OPERATIONS.c.claim_token == claim.claim_token,
                    _OPERATIONS.c.attempt == claim.attempt,
                    _OPERATIONS.c.claim_owner_instance_id == self._owner_instance_id,
                )
                .values(claim_token=None, claim_owner_instance_id=None, claim_expires_at=None, updated_at=now)
            ).rowcount
        if released != 1:
            raise ComposerOperationFenceLost(session_id=claim.session_id, operation_id=claim.operation_id)

    def request_cancel(
        self,
        *,
        session_id: UUID,
        operation_id: str,
        cancelled_failure: ComposerOperationCancelledFailure,
    ) -> ComposerOperationRecord | None:
        """Commit the cancel intent; settle at once when no live claim or turn owns the job.

        A queued job no live claim holds settles ``request_cancelled`` in this
        transaction. A claimed or running job only gains ``cancel_requested_at``
        (first marker wins) and stays nonterminal until its owner stops. A
        terminal job is returned unchanged: completion that committed first
        wins. ``None`` when no such job exists.
        """
        sid = _session_key(session_id)
        _require_operation_id(operation_id)
        with locked_session_transaction(self._engine, sid) as conn:
            now = _database_clock_value(conn.exec_driver_sql(self._clock_sql).scalar_one())
            row = conn.execute(select(_OPERATIONS).where(*_row_key(sid, operation_id)).with_for_update()).one_or_none()
            if row is None:
                return None
            if row.status in _TERMINAL_STATUSES:
                return _record_from_row(row)
            if row.status == "queued" and not _claim_is_live(row, now):
                failure = _require_failure(cancelled_failure(request_id=row.request_id), _REQUEST_CANCELLED)
                result_json, result_sha256 = _encoded_failure(failure)
                settled = conn.execute(
                    update(_OPERATIONS)
                    .where(
                        *_row_key(sid, operation_id),
                        _OPERATIONS.c.status == "queued",
                        _OPERATIONS.c.attempt == row.attempt,
                        _prior_claim_token(row),
                    )
                    .values(
                        status="failed",
                        cancel_requested_at=func.coalesce(_OPERATIONS.c.cancel_requested_at, now),
                        failure_code=failure.failure_code,
                        result_schema=COMPOSER_OPERATION_RESULT_SCHEMA_ERROR,
                        result_json=result_json,
                        result_sha256=result_sha256,
                        settled_at=now,
                        updated_at=now,
                        request_json=None,
                        claim_token=None,
                        claim_owner_instance_id=None,
                        claim_expires_at=None,
                    )
                    .returning(_OPERATIONS)
                ).one_or_none()
                if settled is None:
                    raise AuditIntegrityError("Composer operation cancel settlement lost its locked compare-and-swap")
                return _record_from_row(settled)
            if row.cancel_requested_at is not None:
                return _record_from_row(row)
            marked = conn.execute(
                update(_OPERATIONS)
                .where(
                    *_row_key(sid, operation_id),
                    _OPERATIONS.c.status == row.status,
                    _OPERATIONS.c.cancel_requested_at.is_(None),
                )
                .values(cancel_requested_at=now, updated_at=now)
                .returning(_OPERATIONS)
            ).one_or_none()
            if marked is None:
                raise AuditIntegrityError("Composer operation cancel marker lost its locked compare-and-swap")
            return _record_from_row(marked)

    def settle_unstarted(
        self,
        claim_or_none: ComposerOperationClaim | None,
        *,
        session_id: UUID,
        operation_id: str,
        failure: ComposerOperationError,
    ) -> ComposerOperationRecord:
        """Settle a job that never reached ``running``: ``queued -> failed``.

        With a claim, the exact live claim is the fence: a precondition refused
        before ``running``, or a deadline or cancel the claim owner observed.
        Without a claim, this is the reaper's settlement of a job no live claim
        holds, restricted to what the row proves under database time (a passed
        deadline or a committed cancel marker); a job a racer already settled
        is returned unchanged. ``request_json`` and the claim are cleared.
        """
        sid = _session_key(session_id)
        _require_operation_id(operation_id)
        if claim_or_none is None:
            _require_failure(failure, _UNCLAIMED_FAILURE_CODES)
        else:
            claim = _require_claim(claim_or_none)
            _require_failure(failure, _UNSTARTED_FAILURE_CODES)
            if claim.session_id != session_id or claim.operation_id != operation_id:
                raise ValueError("claim does not name this composer operation")
        result_json, result_sha256 = _encoded_failure(failure)
        with locked_session_transaction(self._engine, sid) as conn:
            now = _database_clock_value(conn.exec_driver_sql(self._clock_sql).scalar_one())
            row = conn.execute(select(_OPERATIONS).where(*_row_key(sid, operation_id)).with_for_update()).one_or_none()
            if claim_or_none is None:
                if row is not None and row.status in _TERMINAL_STATUSES:
                    return _record_from_row(row)
                if row is None or row.status != "queued" or _claim_is_live(row, now):
                    raise ComposerOperationFenceLost(session_id=session_id, operation_id=operation_id)
            elif (
                row is None
                or row.status != "queued"
                or row.claim_token != claim_or_none.claim_token
                or row.attempt != claim_or_none.attempt
                or row.claim_owner_instance_id != self._owner_instance_id
                or not _claim_is_live(row, now)
            ):
                raise ComposerOperationFenceLost(session_id=session_id, operation_id=operation_id)
            if failure.failure_code == "deadline_expired" and _ensure_utc(row.deadline_at) > now:
                raise ValueError("composer operation deadline has not passed")
            if failure.failure_code == "request_cancelled" and row.cancel_requested_at is None:
                raise ValueError("composer operation has no committed cancel marker")
            settled = conn.execute(
                update(_OPERATIONS)
                .where(
                    *_row_key(sid, operation_id),
                    _OPERATIONS.c.status == "queued",
                    _OPERATIONS.c.attempt == row.attempt,
                    _prior_claim_token(row),
                )
                .values(
                    status="failed",
                    failure_code=failure.failure_code,
                    result_schema=COMPOSER_OPERATION_RESULT_SCHEMA_ERROR,
                    result_json=result_json,
                    result_sha256=result_sha256,
                    settled_at=now,
                    updated_at=now,
                    request_json=None,
                    claim_token=None,
                    claim_owner_instance_id=None,
                    claim_expires_at=None,
                )
                .returning(_OPERATIONS)
            ).one_or_none()
            if settled is None:
                raise AuditIntegrityError("Composer operation unstarted settlement lost its locked compare-and-swap")
            return _record_from_row(settled)

    def list_expired_queued(self, *, limit: int) -> tuple[ComposerOperationRecord, ...]:

    def settle_lost(
        self,
        *,
        session_operation_context: SessionOperationContext,
        session_id: UUID,
        operation_id: str,
        failure: ComposerOperationError,
        cancelled_failure: ComposerOperationCancelledFailure,
    ) -> ComposerOperationRecord:
        """Settle a ``running`` job whose worker lost its COMPOSE fence: ``running -> failed``.

        Called by the reaper AFTER it acquired COMPOSE on the session itself.
        That context must still be live and must supersede the epoch the job
        is bound to. ``failure`` is the ``worker_lost`` envelope; a cancel
        marker that committed first turns the terminal into
        ``request_cancelled`` built from the row's own request id. No provider
        turn is replayed. A job a racer already settled is returned unchanged.
        """
        sid = _session_key(session_id)
        _require_operation_id(operation_id)
        if type(session_operation_context) is not SessionOperationContext:
            raise TypeError("session_operation_context must be an exact SessionOperationContext")
        fence = session_operation_context.fence
        if session_operation_context.operation_kind is not SessionOperationKind.COMPOSE or fence.session_id != sid:
            raise SessionOperationFenceLost(FenceLossReason.TOKEN_MISMATCH)
        _require_failure(failure, _WORKER_LOST)
        with locked_session_transaction(self._engine, sid) as conn:
            now = _database_clock_value(conn.exec_driver_sql(self._clock_sql).scalar_one())
            live = conn.execute(
                select(_FENCES.c.operation_epoch)
                .where(
                    _FENCES.c.session_id == fence.session_id,
                    _FENCES.c.operation_id == fence.operation_id,
                    _FENCES.c.lease_token == fence.lease_token,
                    _FENCES.c.operation_epoch == fence.operation_epoch,
                    _FENCES.c.operation_kind == SessionOperationKind.COMPOSE.value,
                    _FENCES.c.released_at.is_(None),
                    _FENCES.c.lease_expires_at > now,
                )
                .with_for_update()
            ).one_or_none()
            if live is None:
                raise SessionOperationFenceLost(FenceLossReason.TOKEN_MISMATCH)
            row = conn.execute(select(_OPERATIONS).where(*_row_key(sid, operation_id)).with_for_update()).one_or_none()
            if row is None:
                raise ComposerOperationFenceLost(session_id=session_id, operation_id=operation_id)
            if row.status in _TERMINAL_STATUSES:
                return _record_from_row(row)
            if row.status != "running":
                raise ComposerOperationFenceLost(session_id=session_id, operation_id=operation_id)
            if row.session_operation_epoch >= fence.operation_epoch:
                raise SessionOperationFenceLost(FenceLossReason.STALE_EPOCH)
            settled_failure = (
                failure
                if row.cancel_requested_at is None
                else _require_failure(cancelled_failure(request_id=row.request_id), _REQUEST_CANCELLED)
            )
            result_json, result_sha256 = _encoded_failure(settled_failure)
            settled = conn.execute(
                update(_OPERATIONS)
                .where(
                    *_row_key(sid, operation_id),
                    _OPERATIONS.c.status == "running",
                    _OPERATIONS.c.claim_token == row.claim_token,
                    _OPERATIONS.c.session_operation_id == row.session_operation_id,
                    _OPERATIONS.c.session_operation_lease_token == row.session_operation_lease_token,
                    _OPERATIONS.c.session_operation_epoch == row.session_operation_epoch,
                    _OPERATIONS.c.cancel_requested_at.is_(None)
                    if row.cancel_requested_at is None
                    else _OPERATIONS.c.cancel_requested_at.is_not(None),
                )
                .values(
                    status="failed",
                    failure_code=settled_failure.failure_code,
                    result_schema=COMPOSER_OPERATION_RESULT_SCHEMA_ERROR,
                    result_json=result_json,
                    result_sha256=result_sha256,
                    settled_at=now,
                    updated_at=now,
                    request_json=None,
                    claim_token=None,
                    claim_owner_instance_id=None,
                    claim_expires_at=None,
                )
                .returning(_OPERATIONS)
            ).one_or_none()
            if settled is None:
                raise AuditIntegrityError("Composer operation lost-worker settlement lost its locked compare-and-swap")
            return _record_from_row(settled)

    def settle_lost_inactive_session(
        self,
        *,
        session_id: UUID,
        operation_id: str,
        failure: ComposerOperationError,
        cancelled_failure: ComposerOperationCancelledFailure,
    ) -> ComposerOperationRecord:
        """Settle a ``running`` job of an ARCHIVED session: ``running -> failed`` (review-pass B1).

        ``settle_lost`` can never run for this row. The reaper's COMPOSE
        acquire on an archived session raises ``SessionOperationConflictError``
        while the fence is live and ``SessionOperationFenceLost(OWNER_INACTIVE)``
        once it is not (repository.py:4797-4800), so it never holds the
        successor context ``settle_lost`` requires. Without this settle the row
        would stay ``running``, hold a D3 capacity slot forever, and head
        ``list_expired_running``'s oldest-first batch.

        The settle is unfenced. Its guard is re-decided under the session
        lock: the session row exists with ``archived_at`` set and the job is
        ``running``. The archive is the proof that the owner is finished: an
        archive acquires its own fence, which a live COMPOSE fence refuses,
        and renewal and every fenced mutation of an archived session raise
        OWNER_INACTIVE (repository.py:5023-5036), so the owner can write
        nothing more. ``failure`` is the ``worker_lost`` envelope; a cancel
        marker that committed first turns the terminal into
        ``request_cancelled`` built from the row's own request id. Any failed
        guard, including an already-terminal job, raises
        ``ComposerOperationFenceLost``; the reaper maps that to "already
        settled" and re-reads.
        """
        sid = _session_key(session_id)
        _require_operation_id(operation_id)
        _require_failure(failure, _WORKER_LOST)
        with locked_session_transaction(self._engine, sid) as conn:
            now = _database_clock_value(conn.exec_driver_sql(self._clock_sql).scalar_one())
            session_row = conn.execute(
                select(sessions_table.c.archived_at).where(sessions_table.c.id == sid).with_for_update()
            ).one_or_none()
            if session_row is None or session_row.archived_at is None:
                raise ComposerOperationFenceLost(session_id=session_id, operation_id=operation_id)
            row = conn.execute(select(_OPERATIONS).where(*_row_key(sid, operation_id)).with_for_update()).one_or_none()
            if row is None or row.status != "running":
                raise ComposerOperationFenceLost(session_id=session_id, operation_id=operation_id)
            settled_failure = (
                failure
                if row.cancel_requested_at is None
                else _require_failure(cancelled_failure(request_id=row.request_id), _REQUEST_CANCELLED)
            )
            result_json, result_sha256 = _encoded_failure(settled_failure)
            settled = conn.execute(
                update(_OPERATIONS)
                .where(
                    *_row_key(sid, operation_id),
                    _OPERATIONS.c.status == "running",
                    _OPERATIONS.c.claim_token == row.claim_token,
                    _OPERATIONS.c.session_operation_id == row.session_operation_id,
                    _OPERATIONS.c.session_operation_lease_token == row.session_operation_lease_token,
                    _OPERATIONS.c.session_operation_epoch == row.session_operation_epoch,
                    _OPERATIONS.c.cancel_requested_at.is_(None)
                    if row.cancel_requested_at is None
                    else _OPERATIONS.c.cancel_requested_at.is_not(None),
                )
                .values(
                    status="failed",
                    failure_code=settled_failure.failure_code,
                    result_schema=COMPOSER_OPERATION_RESULT_SCHEMA_ERROR,
                    result_json=result_json,
                    result_sha256=result_sha256,
                    settled_at=now,
                    updated_at=now,
                    request_json=None,
                    claim_token=None,
                    claim_owner_instance_id=None,
                    claim_expires_at=None,
                )
                .returning(_OPERATIONS)
            ).one_or_none()
            if settled is None:
                raise AuditIntegrityError("Composer operation inactive-session settlement lost its locked compare-and-swap")
            return _record_from_row(settled)

    def settle_own_lapsed(
        self,
        *,
        session_id: UUID,
        operation_id: str,
        owner_instance_id: str,
        failure: ComposerOperationError,
        cancelled_failure: ComposerOperationCancelledFailure,
    ) -> ComposerOperationRecord:
        """The owner settles its own ``running`` job whose bound fence lapsed (review-pass F-M3).

        On PostgreSQL a fence whose renewal failed stays unreleased
        (``SessionOperationLease.close()`` skips the release after a renewal
        error, lifecycle.py:740-746), and ``acquire`` refuses to take over an
        expired fence while its owner's ``web_instances`` lease is live
        (repository.py:5780-5790), with no same-instance exemption. So no
        reaper can reach ``settle_lost`` while this instance lives, the owner's
        own fenced terminal write is refused, and without this settle the row
        stays ``running`` and D8 answers every new send with 409.

        ``owner_instance_id`` must be this authority's own instance id
        (``ValueError`` otherwise: a caller defect, not a fence loss). Guard,
        under the session lock: the job is ``running``; its claim owner is
        ``owner_instance_id``; and the session's fence row carries exactly
        the job's bound ``(operation_id, lease_token, operation_epoch)``,
        unreleased, with ``lease_expires_at <= now``. A live fence (the owner
        must use its fenced write), a released or superseded one (a successor
        owns the row through ``settle_lost``), a foreign owner, or a missing or
        terminal job raise ``ComposerOperationFenceLost``. The fence is read
        and never advanced, renewed, released or taken over, so this authorizes
        no session write. ``failure`` is the outcome the owner was publishing
        (any code but ``request_cancelled``); a cancel marker that committed
        first substitutes ``request_cancelled`` built from the row's request id.
        """
        sid = _session_key(session_id)
        _require_operation_id(operation_id)
        _require_nonblank(owner_instance_id, "owner_instance_id")
        if owner_instance_id != self._owner_instance_id:
            # "Own" is this instance: settling another live instance's lapsed
            # job would bypass the PostgreSQL rule that an expired fence is
            # taken over only after its owner's membership lapsed.
            raise ValueError("owner_instance_id must be this authority's own instance id")
        _require_failure(failure, _OWN_LAPSED_FAILURE_CODES)
        with locked_session_transaction(self._engine, sid) as conn:
            now = _database_clock_value(conn.exec_driver_sql(self._clock_sql).scalar_one())
            fence = conn.execute(select(_FENCES).where(_FENCES.c.session_id == sid).with_for_update()).one_or_none()
            row = conn.execute(select(_OPERATIONS).where(*_row_key(sid, operation_id)).with_for_update()).one_or_none()
            if (
                row is None
                or row.status != "running"
                or row.claim_owner_instance_id != owner_instance_id
                or fence is None
                or fence.operation_id != row.session_operation_id
                or fence.lease_token != row.session_operation_lease_token
                or fence.operation_epoch != row.session_operation_epoch
                or fence.released_at is not None
                or _ensure_utc(fence.lease_expires_at) > now
            ):
                raise ComposerOperationFenceLost(session_id=session_id, operation_id=operation_id)
            settled_failure = (
                failure
                if row.cancel_requested_at is None
                else _require_failure(cancelled_failure(request_id=row.request_id), _REQUEST_CANCELLED)
            )
            result_json, result_sha256 = _encoded_failure(settled_failure)
            settled = conn.execute(
                update(_OPERATIONS)
                .where(
                    *_row_key(sid, operation_id),
                    _OPERATIONS.c.status == "running",
                    _OPERATIONS.c.claim_token == row.claim_token,
                    _OPERATIONS.c.claim_owner_instance_id == owner_instance_id,
                    _OPERATIONS.c.session_operation_id == row.session_operation_id,
                    _OPERATIONS.c.session_operation_lease_token == row.session_operation_lease_token,
                    _OPERATIONS.c.session_operation_epoch == row.session_operation_epoch,
                    _OPERATIONS.c.cancel_requested_at.is_(None)
                    if row.cancel_requested_at is None
                    else _OPERATIONS.c.cancel_requested_at.is_not(None),
                )
                .values(
                    status="failed",
                    failure_code=settled_failure.failure_code,
                    result_schema=COMPOSER_OPERATION_RESULT_SCHEMA_ERROR,
                    result_json=result_json,
                    result_sha256=result_sha256,
                    settled_at=now,
                    updated_at=now,
                    request_json=None,
                    claim_token=None,
                    claim_owner_instance_id=None,
                    claim_expires_at=None,
                )
                .returning(_OPERATIONS)
            ).one_or_none()
            if settled is None:
                raise AuditIntegrityError("Composer operation own-lapsed settlement lost its locked compare-and-swap")
            return _record_from_row(settled)

**Review notes:**
- The `IntegrityError` re-read arm in `admit` cannot be reached today. Same-session admits are serialized by the process
  RLock and flock on SQLite, and by `pg_advisory_xact_lock` on PostgreSQL, so the second admit always sees the committed
  row through the normal lookup. The arm is defensive and no test pins it. The concurrent-admit tests prove one row
  through the lookup path.
- Capacity (F-C7, accepted in the contract's review-pass table). It is exact on SQLite, where BEGIN IMMEDIATE
  serializes every writer. It is soft on PostgreSQL: admissions to different sessions hold different advisory locks,
  so the nonterminal count after a burst of `k` concurrent admitters is at most `max_nonterminal + k - 1`, and a
  committed count at the cap refuses every later admitter. A global counter row would make it exact only by
  serializing every admission in the cluster, and spec §3 adds no advisory lock. The `admit` docstring states the
  bound; `test_capacity_is_exact_on_sqlite_when_sessions_race_for_the_last_slot` and
