def _claim_exact(jobs: ComposerAsyncOperationAuthority, *, session_id: UUID, operation_id: str) -> ComposerOperationClaim:
    """Claim exactly this job; every other claim a scan took goes back (the PostgreSQL DB is shared).

    Scans until this job is claimed or a scan returns nothing, so rows left
    queued by other modules cannot starve the lookup. Foreign claims are
    released only after the loop, so no scan can re-take one.
    """
    foreign: list[ComposerOperationClaim] = []
    mine: list[ComposerOperationClaim] = []
    try:
        while not mine:
            claims = jobs.claim_next(limit=64)
            if not claims:
                break
            for claim in claims:
                if claim.session_id == session_id and claim.operation_id == operation_id:
                    mine.append(claim)
                else:
                    foreign.append(claim)
    finally:
        for claim in foreign:
            jobs.release_claim(claim)
    assert len(mine) == 1
    return mine[0]


def _age_job(engine: Engine, session_id: UUID, operation_id: str, *, by: timedelta) -> None:
    """Move every time column of one queued job back together.

    Shifting created/updated/deadline/claim-expiry by the same amount keeps
    every time-ordering CHECK on the row satisfied while putting the claim
    expiry behind the database clock. Admit with a deadline longer than
    ``by`` so only the claim lapses.
    """
    table = composer_async_operations_table
    with engine.begin() as conn:
        row = conn.execute(
            select(table.c.created_at, table.c.updated_at, table.c.deadline_at, table.c.claim_expires_at).where(
                table.c.session_id == str(session_id), table.c.operation_id == operation_id
            )
        ).one()
        conn.execute(
            update(table)
            .where(table.c.session_id == str(session_id), table.c.operation_id == operation_id)
            .values(
                created_at=row.created_at - by,
                updated_at=row.updated_at - by,
                deadline_at=row.deadline_at - by,
                claim_expires_at=row.claim_expires_at - by,
            )
        )


def _job(engine: Engine, session_id: UUID, operation_id: str) -> RowMapping:
    table = composer_async_operations_table
    with engine.connect() as conn:
        return (
            conn.execute(select(table).where(table.c.session_id == str(session_id), table.c.operation_id == operation_id))
            .mappings()
            .one()
        )


def _fence(engine: Engine, session_id: UUID) -> RowMapping:

            .one()

class ComposerOperationCancelledBeforeStart(RuntimeError):
    """A cancel marker committed before the claimed job reached ``running``.

    Raised by ``SessionOperationAuthority.start_composer_async_operation``
    inside its one transaction, so nothing was written: the job is still
    ``queued`` with the worker's claim and the COMPOSE fence did not advance.
    The worker settles it ``request_cancelled`` through
    ``ComposerAsyncOperationAuthority.settle_unstarted``. Distinct from
    :class:`ComposerOperationFenceLost` (the claim is still live). Carries the
    job identity for diagnostics and never the claim token.
    """

    def __init__(self, *, session_id: UUID, operation_id: str) -> None:
        super().__init__(f"composer operation {operation_id} was cancelled before it started")
        self.session_id = session_id
        self.operation_id = operation_id

            raise SessionOperationFenceLost(FenceLossReason.MISSING)

        released_at = row.released_at
        lease_expires_at = _ensure_utc(row.lease_expires_at)
        if session_row.archived_at is not None:
            if released_at is None and lease_expires_at > database_now:
                raise SessionOperationConflictError
            raise SessionOperationFenceLost(FenceLossReason.OWNER_INACTIVE)
        if released_at is None:
            if lease_expires_at > database_now:
                raise SessionOperationConflictError
            if not self._expired_owner_allows_takeover(
                conn,
                owner_instance_id=row.owner_instance_id,
                database_now=database_now,

                raise SessionOperationConflictError

        lease_token = _new_lease_token(owner_instance_id=owner_instance_id)

        One same-session locked transaction (contract §Composite start, R3):

        1. re-prove the queued claim (row present, ``queued``, exact token and
           attempt, claim expiry after the database clock), else
           :class:`ComposerOperationFenceLost`;

           :meth:`_advance_exclusive_fence_on_connection` — its
           :class:`SessionOperationConflictError` and

def _clone_claimed_job(engine: Engine, claim: ComposerOperationClaim) -> ComposerOperationClaim:
    """Insert a second queued, claimed job in the same session by copying the first row.

    Admission refuses a second nonterminal job per session, so no production
    path creates this pair. The copy is valid by construction (the real
    authority wrote the row it copies) and isolates the claim: it proves the
    COMPOSE fence, not admission, is what stops two claimed jobs of one
    session from both reaching ``running``.
    """
    table = composer_async_operations_table
    clone_operation_id = str(uuid4())
    clone_token = f"clone-{uuid4().hex}"
    with engine.begin() as conn:
        row = (
            conn.execute(select(table).where(table.c.session_id == str(claim.session_id), table.c.operation_id == claim.operation_id))
            .mappings()
            .one()
        )
        conn.execute(insert(table).values({**row, "operation_id": clone_operation_id, "claim_token": clone_token}))
    return ComposerOperationClaim(
        session_id=claim.session_id,
        operation_id=clone_operation_id,
        claim_token=clone_token,
        attempt=claim.attempt,
    )


def _race(engine: Engine, claims: tuple[ComposerOperationClaim, ComposerOperationClaim]) -> list[object]:
    barrier = Barrier(2)

    def start(claim: ComposerOperationClaim) -> object:
        contender = PostgresSessionOperationRepository(engine)
        barrier.wait(timeout=10)
        try:
            return contender.start_composer_async_operation(claim, owner_instance_id=f"starter-{uuid4()}", lease_seconds=30)
        except (ComposerOperationFenceLost, SessionOperationConflictError) as exc:
            return exc

    with ThreadPoolExecutor(max_workers=2) as pool:
        return list(pool.map(start, claims))

**Review notes**
- **Lock-order dependency on Task 4.** The composite takes the session advisory lock
  (`transaction_session_lock`), then the job row `FOR UPDATE`, then `sessions` and `session_operation_fences`. On
  PostgreSQL this deadlocks if Task 4's `claim_next` holds a job-row lock (for example through `FOR UPDATE SKIP LOCKED`)
  and then waits for the same session's advisory lock. `claim_next` must either take no session advisory lock while
  holding a row lock, or take the advisory lock first. Step 7's lock-order test pins this task's side only.
- **Queued-deadline expiry is not checked here.** The contract lists no deadline arm for the composite. The
  deadline belongs to the Task 11 worker, **before** start:
  - after `claim_next`, the worker reads the record (`authority.get`);
  - when `deadline_at <= now`, it calls `settle_unstarted(claim, …, failure=deadline_expired_error(...))`, which the
    contract lists for exactly this case. No COMPOSE fence is ever minted for a dead job.

  A deadline that lapses in the small window between that read and the composite still reaches `running`, and the
  turn then gets a budget ≤ 0 (D10), which the worker must settle as `deadline_expired`. If Task 3 has a CHECK such
  as `started_at <= deadline_at`, the running UPDATE raises `IntegrityError` in that window instead. No test here
  ages the deadline, so that case is unmeasured.
- **Consumed-shape assumptions, not verified against Tasks 2-4 (they are not written yet).** Each has a stop check
  in Step 0 or a fail-loud point:
  - the table's Python name is `composer_async_operations_table`;
  - `ComposerOperationFenceLost` takes a single message argument;
  - `RecomposeRequest` exists by Task 2, although the contract's DTO block says "NEW (Task 10)" while its hash
    signature needs it in Task 2;
  - on a claimed queued row, `request_cancel` only sets the marker;
  - `claim_next` reclaims an expired claim and increments `attempt`;
  - a queued row copied column-for-column into a new `operation_id` satisfies every Task 3 CHECK and unique index.
    This last one is used only by the PostgreSQL two-jobs race.
- **`_age_job` shifts `created_at`, `updated_at`, `deadline_at` and `claim_expires_at` together.** It does this so
  that no time-ordering CHECK is violated. If Task 3 adds a CHECK tying these columns to a column not in that set,
  the helper must shift that column too.
- **PostgreSQL shares one database across modules.** `_claim_exact` scans `claim_next(limit=64)` until it finds its
  job or a scan returns nothing, then releases every foreign claim it took. It can therefore briefly re-claim rows
  left queued by other modules, such as Task 4's PostgreSQL tests. This is harmless because those claims are
  released, but it is not isolated. It also depends on `release_claim` accepting a claim taken by this same
  authority instance.
