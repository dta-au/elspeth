    """The provider seam. Each ``compose`` call appends one line to a ledger that every process shares.

    ``hold`` parks the call on ``release`` after the ledger line is flushed,
    so a test can kill, partition, cancel or complete a turn whose provider
    call provably started.

    Like the real compose loop, every call yields one ``ComposerLLMCall``
    (model ``ledger/<instance>``). A returned result carries it in
    ``llm_calls`` with status ``success``. A cancellation during the hold
    carries it on the ``CancelledError`` with status ``cancelled``, through
    ``attach_llm_calls`` exactly as ``ComposerServiceImpl`` does
    (``composer/service.py:8505``). So the turn's LLM-call audit cohort exists
    on every path, and a test can ask whether it survived.
    """

    def __init__(self, *, ledger_path: Path, instance_name: str, hold: bool) -> None:
        self._ledger_path = ledger_path
        self._instance_name = instance_name
        self._hold = hold
        self.entered = asyncio.Event()
        self.release = asyncio.Event()

    async def compose(
        self,
        message: str,
        chat_messages: list[dict[str, object]],
        state: CompositionState,
        *,
        session_id: str | None = None,
        current_state_id: str | None = None,
        user_id: str | None = None,
        progress: object = None,
        guided_terminal: object = None,
        user_message_id: str | None = None,
        session_operation_context: SessionOperationContext | None = None,
        completion_gates: object = None,
        budget_seconds: float | None = None,
    ) -> ComposerResult:
        del chat_messages, current_state_id, user_id, progress, guided_terminal, session_operation_context, completion_gates
        started_at = datetime.now(UTC)
        started_ns = time.monotonic_ns()
        entry = {
            "instance": self._instance_name,
            "pid": os.getpid(),
            "session_id": session_id,
            "message": message,
            "user_message_id": user_message_id,
            "budget_seconds": budget_seconds,
        }
        with self._ledger_path.open("a", encoding="utf-8") as handle:
            handle.write(json.dumps(entry, sort_keys=True) + "\n")
            handle.flush()
            os.fsync(handle.fileno())
        self.entered.set()
        if self._hold:
            try:
                await self.release.wait()
            except asyncio.CancelledError as cancelled:
                recorder = BufferingRecorder()
                recorder.record_llm_call(self._llm_call(message, started_at, started_ns, status=ComposerLLMCallStatus.CANCELLED))
                attach_llm_calls(cancelled, recorder)
                raise
        # The state is returned unchanged: no state commit, so the assistant
        # row is written by the composite terminal transaction itself.
        return ComposerResult(
            message=f"reply-from-{self._instance_name}",
            state=state,
            llm_calls=(self._llm_call(message, started_at, started_ns, status=ComposerLLMCallStatus.SUCCESS),),
        )

    def _llm_call(self, message: str, started_at: datetime, started_ns: int, *, status: ComposerLLMCallStatus) -> ComposerLLMCall:
        cancelled = status is ComposerLLMCallStatus.CANCELLED
        return build_llm_call_record(
            model_requested=f"ledger/{self._instance_name}",
            messages=[{"role": "user", "content": message}],
            tools=None,
            status=status,
            started_at=started_at,
            started_ns=started_ns,
            temperature=None,
            seed=None,
            error_class="CancelledError" if cancelled else None,
            error_message="the held provider call was cancelled" if cancelled else None,
        )


class _Partition:
    """Holds one of this process's liveness writes (SOL renewal, or membership heartbeat) while frozen."""

    def __init__(self) -> None:
        self._open = threading.Event()
        self._open.set()

    def freeze(self) -> None:
        self._open.clear()

    def heal(self) -> None:
        self._open.set()

    def pass_gate(self) -> None:
        if not self._open.wait(timeout=_REPLY_SECONDS * 2):
            raise TimeoutError("the partition was never healed")


_NEVER_RELEASED = threading.Event()


def _install_crash_point(monkeypatch: pytest.MonkeyPatch, crash_point: _CrashPoint, engine: Engine) -> None:
    if crash_point == "after_admission_commit":
        admit = ComposerAsyncOperationAuthority.admit

        def admit_then_lose_the_acknowledgement(
            self: ComposerAsyncOperationAuthority, **arguments: Any
        ) -> tuple[ComposerOperationRecord, bool]:
            admitted = admit(self, **arguments)
            # The queue row is committed; this worker thread never returns,
            # so no 202 is produced. The test kills the process here.
            _NEVER_RELEASED.wait()
            return admitted

        monkeypatch.setattr(ComposerAsyncOperationAuthority, "admit", admit_then_lose_the_acknowledgement)
    elif crash_point == "before_running":

        def park_before_running(
            self: PostgresSessionOperationRepository,
            claim: ComposerOperationClaim,
            *,
            owner_instance_id: str,
            lease_seconds: int,
        ) -> SessionOperationContext:
            # The queued claim is held; the composite start never runs, so no
            # SOL advance, no user row and no provider call can happen.
            _NEVER_RELEASED.wait()
            raise AssertionError(f"unreachable: {claim.operation_id} parks until the test kills this process")

        monkeypatch.setattr(PostgresSessionOperationRepository, "start_composer_async_operation", park_before_running)
    elif crash_point == "cancel_before_terminal":
        complete = SessionServiceImpl.complete_composer_async_operation
        jobs = ComposerAsyncOperationAuthority(
            engine, owner_instance_id=f"cao-stop-{uuid4().hex}", claim_lease_seconds=_CLAIM_LEASE_SECONDS
        )

        async def stop_then_complete(
            self: SessionServiceImpl, running: ComposerOperationRunning, **arguments: Any
        ) -> ComposerOperationRecord:
            # compose() has returned: its result, and the LLM call it carries, exist only in this
            # frame. The user's Stop commits now, before the composite terminal transaction opens,
            # so that transaction cannot complete: its CAS (cancel_requested_at IS NULL) loses, or the
            # owner's watcher reads the marker first and cancels this tail. Either way the cohort must
            # be re-persisted before the request_cancelled terminal.
            claim = running.claim
            record = await asyncio.to_thread(jobs.get, session_id=claim.session_id, operation_id=claim.operation_id)
            assert record is not None and record.status == "running", record
            marked = await asyncio.to_thread(
                jobs.request_cancel,
                session_id=claim.session_id,
                operation_id=claim.operation_id,
                cancelled_failure=request_cancelled_error(request_id=record.request_id),
            )
            assert marked is not None and marked.status == "running" and marked.cancel_requested_at is not None, marked
            return await complete(self, running, **arguments)

        monkeypatch.setattr(SessionServiceImpl, "complete_composer_async_operation", stop_then_complete)


def _install_partition(monkeypatch: pytest.MonkeyPatch, *, sol: _Partition, membership: _Partition) -> None:
    renew = PostgresSessionOperationRepository.renew
    heartbeat = RepositoryWebInstanceMembershipAuthority.heartbeat

    def partitioned_renew(
        self: PostgresSessionOperationRepository, context: SessionOperationContext, *, lease_seconds: int
    ) -> SessionOperationContext:
        sol.pass_gate()
        return renew(self, context, lease_seconds=lease_seconds)

    def partitioned_heartbeat(self: RepositoryWebInstanceMembershipAuthority, instance_id: str, *, lease_seconds: int) -> Any:
        membership.pass_gate()
        return heartbeat(self, instance_id, lease_seconds=lease_seconds)

    monkeypatch.setattr(PostgresSessionOperationRepository, "renew", partitioned_renew)
    monkeypatch.setattr(RepositoryWebInstanceMembershipAuthority, "heartbeat", partitioned_heartbeat)

async def _serve_instance(spec: _InstanceSpec, pipe: Connection) -> None:
    engine = create_session_engine(spec.url)
    composer = _LedgerComposer(ledger_path=Path(spec.ledger_path), instance_name=spec.name, hold=spec.hold_provider)
    app = _instance_app(engine, spec, composer)
    worker = install_composer_async_worker(app)
    # A test host has no supervisor: a background failure must never SIGTERM
    # this child and blur the difference between a kill and a crash.
    recovery = ProcessRecovery()
    recovery.begin_shutdown()
    membership = (
        RegisteredWebInstanceMembership(
            RepositoryWebInstanceMembershipAuthority(engine),
            _instance_identity(spec.instance_id),
            lease_seconds=_MEMBERSHIP_LEASE_SECONDS,
            interval_seconds=1,
            process_recovery=recovery,
        )
        if spec.register_membership
        else None
    )
    sol_partition = _Partition()
    membership_partition = _Partition()
    drives: list[asyncio.Task[None]] = []
    posts: list[asyncio.Task[Response]] = []
    worker_running = False
    base = f"/api/sessions/{spec.session_id}"
    try:
        with pytest.MonkeyPatch.context() as monkeypatch:
            _install_crash_point(monkeypatch, spec.crash_point, engine)
            _install_partition(monkeypatch, sol=sol_partition, membership=membership_partition)
            if membership is not None:
                await membership.start()
            async with AsyncClient(transport=ASGITransport(app=app), base_url="http://composer.test") as client:
                pipe.send("ready")
                while True:
                    command, argument = await asyncio.to_thread(pipe.recv)
                    if command == "stop":
                        break
                    if command == "post":
                        route, body, request_id = argument
                        response = await client.post(f"{base}/{route}", json=body, headers={"X-Request-ID": request_id})
                        pipe.send((response.status_code, response.json()))
                    elif command == "post_unacknowledged":
                        route, body, request_id = argument
                        posts.append(asyncio.create_task(client.post(f"{base}/{route}", json=body, headers={"X-Request-ID": request_id})))
                        pipe.send("posting")
                    elif command == "poll":
                        # A fresh client per poll models a reload: nothing is inherited.
                        async with AsyncClient(transport=ASGITransport(app=app), base_url="http://composer.test") as reader:
                            response = await reader.get(f"{base}/operations/{argument}")
                        pipe.send((response.status_code, response.json()))
                    elif command == "cancel":
                        response = await client.post(f"{base}/operations/{argument}/cancel")
                        pipe.send((response.status_code, response.json()))
                    elif command == "messages":
                        response = await client.get(f"{base}/messages")
                        assert response.status_code == 200, response.text
                        pipe.send([(message["role"], message["content"], message["id"]) for message in response.json()])
                    elif command == "drive":
                        await asyncio.wait_for(worker.run_until_idle(), timeout=_REPLY_SECONDS)
                        pipe.send("idle")
                    elif command == "drive_in_background":
                        drives.append(asyncio.create_task(worker.run_until_idle()))
                        pipe.send("driving")
                    elif command == "join_drive":
                        await asyncio.wait_for(drives.pop(), timeout=_REPLY_SECONDS)
                        pipe.send("joined")
                    elif command == "await_provider":
                        await asyncio.wait_for(composer.entered.wait(), timeout=_REPLY_SECONDS)
                        pipe.send("provider-entered")
                    elif command == "release_provider":
                        composer.release.set()
                        pipe.send("released")
                    elif command == "reap":
                        pipe.send(await worker.reap_once())
                    elif command == "start_worker":
                        worker.start()
                        worker_running = True
                        pipe.send("started")
                    elif command == "stop_worker":
                        await worker.stop()
                        worker_running = False
                        pipe.send("stopped")
                    elif command == "partition":
                        sol_partition.freeze()
                        membership_partition.freeze()
                        pipe.send("partitioned")
                    elif command == "partition_sol":
                        # Only SOL renewal stops; the membership heartbeat keeps this instance provably alive.
                        sol_partition.freeze()
                        pipe.send("sol-partitioned")
                    elif command == "heal":
                        sol_partition.heal()
                        membership_partition.heal()
                        pipe.send("healed")
                    else:
                        raise AssertionError(f"Unknown instance command: {command}")
                sol_partition.heal()
                membership_partition.heal()
                composer.release.set()
                for task in posts:
                    task.cancel()
                await asyncio.gather(*posts, return_exceptions=True)
                # A drive that raised fails the child's exit code, which the parent asserts is 0.
                await asyncio.wait_for(asyncio.gather(*drives), timeout=_REPLY_SECONDS)
                if worker_running:
                    await worker.stop()
            if membership is not None:
                await membership.stop()
    finally:
        engine.dispose()
        pipe.close()

def _receive(pipe: Connection) -> object:
    assert pipe.poll(_REPLY_SECONDS), "web instance did not respond"
    return pipe.recv()


class _Instance:
    """A handle on one spawned web instance."""

    def __init__(self, spec: _InstanceSpec, process: BaseProcess, pipe: Connection) -> None:
        self.spec = spec
        self.process = process
        self.pipe = pipe
        self.killed = False

    def send(self, command: str, argument: object = None) -> None:
        self.pipe.send((command, argument))

    def receive(self) -> object:
        return _receive(self.pipe)

    def command(self, command: str, argument: object = None) -> object:
        self.send(command, argument)
        return self.receive()

    def http(self, command: str, argument: object) -> tuple[int, Any]:
        reply = self.command(command, argument)
        assert type(reply) is tuple and len(reply) == 2, reply
        status, body = reply
        assert type(status) is int, reply
        return status, body

    def kill(self) -> None:
        self.process.kill()
        self.process.join(30)
        assert self.process.exitcode == -signal.SIGKILL, self.process.exitcode
        self.killed = True

@contextmanager
def _instances(*specs: _InstanceSpec) -> Iterator[tuple[_Instance, ...]]:
    context = multiprocessing.get_context("spawn")
    handles: list[_Instance] = []
    children: list[Connection] = []
    try:
        for spec in specs:
            parent, child = context.Pipe()
            process = context.Process(target=_instance_process, args=(spec, child))
            process.start()
            handles.append(_Instance(spec, process, parent))
            children.append(child)
        for handle in handles:
            assert handle.receive() == "ready"
        yield tuple(handles)
        live = [handle for handle in handles if not handle.killed]
        for handle in live:
            handle.send("stop")
        for handle in live:
            handle.process.join(_REPLY_SECONDS)
            assert handle.process.exitcode == 0, f"instance {handle.spec.name} exited {handle.process.exitcode}"
    finally:
        for handle in handles:
            if handle.process.is_alive():
                handle.process.kill()
                handle.process.join(30)
            handle.pipe.close()
        for child in children:
            child.close()

    assert row is not None, f"no composer_async_operations row for {operation_id}"
    return row


def _wait_for_job(deployment: _Deployment, operation_id: str, *, until: Callable[[dict[str, Any]], bool], what: str) -> dict[str, Any]:
    for _ in range(300):
        row = _job_or_none(deployment, operation_id)
        if row is not None and until(row):
            return row
        _pg_sleep(deployment, 0.1)
    raise AssertionError(f"timed out waiting for {what}; last row: {_job_or_none(deployment, operation_id)}")

def _llm_call_audit_rows(deployment: _Deployment) -> list[tuple[str, str]]:
    """``(model_requested, status)`` of every LLM-call audit row in the session, in transcript order.

    These are the ``role='audit'`` ``chat_messages`` rows whose ``tool_calls``
    envelope is ``{"_kind": "llm_call_audit", "call": {...}}``
    (``composer/audit.py:385-394``): what ``_persist_llm_calls``, the composite
    terminal's audit cohort and the cancelled-turn audit path all write.
    """
    with deployment.engine.connect() as conn:
        envelopes = (
            conn.execute(
                select(chat_messages_table.c.tool_calls)
                .where(chat_messages_table.c.session_id == deployment.session_id, chat_messages_table.c.role == "audit")
                .order_by(chat_messages_table.c.sequence_no)
            )
            .scalars()
            .all()
        )
    return [
        (entry["call"]["model_requested"], entry["call"]["status"])
        for envelope in envelopes
        if envelope is not None
        for entry in envelope
        if entry["_kind"] == "llm_call_audit"
    ]


def _stable(body: dict[str, Any]) -> dict[str, Any]:
    """A status body without ``deadline_remaining_ms``, which each read computes from the database clock (F-B3).

    Two reads of one operation seconds apart differ in that field only, so every
    cross-read equality compares through this. The field is checked, not dropped
    blind: an absent or out-of-range value fails here.
    """
    remaining = body["deadline_remaining_ms"]
    assert type(remaining) is int and 0 <= remaining <= int(_COMPOSER_TIMEOUT_SECONDS * 1000), body
    return {key: value for key, value in body.items() if key != "deadline_remaining_ms"}

        "failure_code": "worker_lost",
        "error_type": "composer_operation_worker_lost",
        "body": {"detail": {"error_type": "composer_operation_worker_lost", "detail": _WORKER_LOST_DETAIL, "request_id": request_id}},
        "diagnostic_id": None,
    }


def _request_cancelled(request_id: str) -> dict[str, object]:
    return {
        "http_status": 499,
        "failure_code": "request_cancelled",
        "error_type": "request_cancelled",
        "body": {"detail": {"error_type": "request_cancelled", "detail": _CANCELLED_DETAIL, "request_id": request_id}},
        "diagnostic_id": None,
    }


def _await_terminal(instance: _Instance, deployment: _Deployment, operation_id: str) -> dict[str, Any]:
    body: Any = None
    for _ in range(150):
        status, body = instance.http("poll", operation_id)
        assert status == 200, body
        if body["status"] in _TERMINAL:
            return body
        _pg_sleep(deployment, 0.2)
    raise AssertionError(f"operation {operation_id} never became terminal; last poll: {body}")


def _drive_until_terminal(instance: _Instance, deployment: _Deployment, operation_id: str) -> dict[str, Any]:
    body: Any = None
    for _ in range(20):
        assert instance.command("drive") == "idle"
        status, body = instance.http("poll", operation_id)
        assert status == 200, body
        if body["status"] in _TERMINAL:
            return body
        _pg_sleep(deployment, 0.5)
    raise AssertionError(f"operation {operation_id} never became terminal under {instance.spec.name}; last poll: {body}")


def _reap_until_settled(instance: _Instance, deployment: _Deployment) -> None:
    """Wait out the dead or partitioned owner's leases, then reap with a bounded retry."""
    _pg_sleep(deployment, _PAST_OWNER_LEASES_SECONDS)
    for _ in range(20):
        reaped = instance.command("reap")
        if reaped == 1:
            return
        assert reaped == 0, reaped
        _pg_sleep(deployment, 0.5)
    raise AssertionError("the peer reaper never settled the lapsed owner's running job")


def _start_held_turn(deployment: _Deployment, owner: _Instance, operation_id: str, request_id: str) -> dict[str, Any]:
    """Admit on ``owner``, drive it in the background, and return the ``running`` row once the provider is entered."""
    assert owner.http("post", ("messages", _send_body(operation_id), request_id)) == (202, _accepted(operation_id, status="queued"))
    assert owner.command("drive_in_background") == "driving"
    assert owner.command("await_provider") == "provider-entered"
    running = _job(deployment, operation_id)
    assert running["status"] == "running", running
    assert running["claim_owner_instance_id"] == owner.spec.instance_id
    assert running["attempt"] == 1
    assert running["started_at"] is not None and running["user_message_id"] is not None
    assert running["request_id"] == request_id
    assert [call["instance"] for call in _provider_calls(deployment)] == [owner.spec.name]
    return running

def test_lost_acknowledgement_retry_replays_the_one_row_and_one_provider_turn(deployment: _Deployment) -> None:

    body = _send_body(operation_id)
    with _instances(deployment.spec("a", crash_point="after_admission_commit"), deployment.spec("b")) as (a, b):

@pytest.mark.parametrize("order", ["cancel_then_complete", "complete_then_cancel", "concurrent"])
def test_cancel_and_completion_settle_exactly_one_terminal(deployment: _Deployment, order: str) -> None:
    """Window 5: one token-checked terminal transition wins. The loser reads it, and the live owner is never reaped."""
    operation_id = str(uuid4())
    with _instances(deployment.spec("a", hold_provider=True), deployment.spec("b")) as (a, b):
        _start_held_turn(deployment, a, operation_id, "cao-race-post")
        if order == "cancel_then_complete":
            cancel_reply = b.http("cancel", operation_id)
            # The owner is alive and renewing: its running job is not reapable.
            assert b.command("reap") == 0
            assert a.command("release_provider") == "released"
            assert a.command("join_drive") == "joined"
        elif order == "complete_then_cancel":
            assert a.command("release_provider") == "released"
            assert a.command("join_drive") == "joined"
            completed_row = _job(deployment, operation_id)
            cancel_reply = b.http("cancel", operation_id)
            assert _job(deployment, operation_id) == completed_row
        else:
            a.send("release_provider")
            b.send("cancel", operation_id)
            assert a.receive() == "released"
            reply = b.receive()
            assert type(reply) is tuple and len(reply) == 2, reply
            cancel_reply = (reply[0], reply[1])
            assert a.command("join_drive") == "joined"
        assert b.command("reap") == 0
        final = _await_terminal(b, deployment, operation_id)
        status, from_a = a.http("poll", operation_id)
        assert (status, _stable(from_a)) == (200, _stable(final))
    cancel_status, cancel_body = cancel_reply
    if order == "cancel_then_complete":
        assert cancel_status == 202
    if order == "complete_then_cancel":
        assert cancel_status == 200
    if cancel_status == 202:
        # Cancel committed first. Either the owner's watcher stopped the turn, or the terminal CAS
        # (cancel_requested_at IS NULL) rejected completion and rolled back its transaction. Either way no assistant row survives.
        assert (cancel_body["status"], cancel_body["cancel_requested"]) == ("running", True)
        assert final["status"] == "failed" and final["error"] == _request_cancelled("cao-race-post")
        assert _conversation(deployment) == [("user", _CONTENT)]
        # The provider call's audit row always survives the cancel (spec §3; contract F-B2/C2 and D5): ``success``
        # when compose() returned before the cancel decided the turn (the marker landed in the tail, or the
        # terminal CAS lost), ``cancelled`` when the watcher stopped compose() itself. Which one a run takes is
        # timing; the deterministic post-compose cancel (CAS loss on its common path) is
        # test_cancel_committed_after_the_provider_returned_keeps_the_llm_call_audit_cohort.
        audit = _llm_call_audit_rows(deployment)
        assert len(audit) == 1 and audit[0][0] == "ledger/a" and audit[0][1] in ("success", "cancelled"), audit
    else:
        # Completion committed first: cancel returned the completed row unchanged.
        assert cancel_status == 200
        assert _stable(cancel_body) == _stable(final)
        assert (final["status"], final["cancel_requested"], final["error"]) == ("completed", False, None)
        assert final["result"]["message"]["content"] == "reply-from-a"
        assert _conversation(deployment) == [("user", _CONTENT), ("assistant", "reply-from-a")]
        # Positive control for the audit reader: the completed turn's cohort committed with its composite.
        assert _llm_call_audit_rows(deployment) == [("ledger/a", "success")]
    assert _operation_count(deployment) == 1
    assert [call["instance"] for call in _provider_calls(deployment)] == ["a"]


def test_cancel_committed_after_the_provider_returned_keeps_the_llm_call_audit_cohort(deployment: _Deployment) -> None:
    """Window 5, a deterministic post-compose cancel: a Stop committed after compose() returned.

    The ``cancel_before_terminal`` crash point commits the cancel marker inside
    ``complete_composer_async_operation`` before delegating. The provider call
    has happened and been charged. On the common path the composite's CAS
    (``cancel_requested_at IS NULL``) then loses; on the rarer path the owner's
    watcher reads the marker first (about 0.25 s backoff) and cancels the tail.
    On both, the assistant row is never committed; the provider call's LLM-call
    audit row must be (spec §3 l.192-194; contract F-B2/C2: persisted through
    the audit-only path, joined, before the ``request_cancelled`` terminal).
    """
    operation_id = str(uuid4())
    with _instances(deployment.spec("a", crash_point="cancel_before_terminal")) as (a,):
        assert a.http("post", ("messages", _send_body(operation_id), "cao-b2-post")) == (202, _accepted(operation_id, status="queued"))
        final = _drive_until_terminal(a, deployment, operation_id)
    assert (final["status"], final["cancel_requested"], final["result"]) == ("failed", True, None)
    assert final["error"] == _request_cancelled("cao-b2-post")
    settled = _job(deployment, operation_id)
    assert (settled["status"], settled["failure_code"], settled["attempt"]) == ("failed", "request_cancelled", 1)
    assert settled["cancel_requested_at"] is not None
    # compose() ran exactly once, and nothing replayed it.
    assert [call["instance"] for call in _provider_calls(deployment)] == ["a"]
    # The reply rolled back with the composite ...
    assert _conversation(deployment) == [("user", _CONTENT)]
    # ... and the provider call's audit row survived it, as the completed call it was.
    assert _llm_call_audit_rows(deployment) == [("ledger/a", "success")]


def test_live_owner_whose_bound_lease_lapsed_settles_its_own_job_and_frees_the_session(deployment: _Deployment) -> None:
    """A live owner's bound SOL lapses; the owner settles its own job (contract F-M3, ``settle_own_lapsed``).

    Only SOL renewal is frozen. A's membership heartbeat keeps it provably alive,
    so PostgreSQL refuses every peer takeover (``_expired_owner_allows_takeover``,
    ``repository.py:5780-5790``) and no peer reaper can settle the row. When the
    frozen renewal resumes it finds the lease expired and raises, A records the
    lease as lost, its fenced terminal write is refused, and A settles its own job
    through the guarded owner-side settle. Without that settle the row would hold
    the session behind D8's 409 until A restarted.
    """
    operation_id = str(uuid4())
    with _instances(deployment.spec("a", hold_provider=True), deployment.spec("b")) as (a, b):
        running = _start_held_turn(deployment, a, operation_id, "cao-m3-post")
        fence_running = _fence(deployment)
        assert fence_running[0] == running["session_operation_epoch"] and fence_running[3] is None
        assert a.command("partition_sol") == "sol-partitioned"
        _pg_sleep(deployment, _PAST_OWNER_LEASES_SECONDS)
        # The wedge: A's SOL has lapsed in the database, but A is alive, so no peer may take over.
        for _ in range(3):
            assert b.command("reap") == 0
            _pg_sleep(deployment, 0.5)
        assert _job(deployment, operation_id)["status"] == "running"
        # Control for the admission assertion below: while the row is running, a new id is refused (D8).
        status, refused = b.http("post", ("messages", _send_body(str(uuid4()), "A second action"), "cao-m3-refused"))
        assert status == 409, refused
        assert (refused["detail"]["error_type"], refused["detail"]["operation_id"]) == ("composer_operation_active", operation_id)
        # The frozen renewal resumes, finds its lease expired and raises: A loses its lease while A lives.
        assert a.command("heal") == "healed"
        final = _await_terminal(b, deployment, operation_id)
        assert a.command("join_drive") == "joined"
        status, from_a = a.http("poll", operation_id)
        assert (status, _stable(from_a)) == (200, _stable(final))
        # The session is free again: the next send is admitted, not refused.
        next_id = str(uuid4())
        assert b.http("post", ("messages", _send_body(next_id, "Try again"), "cao-m3-next")) == (202, _accepted(next_id, status="queued"))
    assert (final["status"], final["cancel_requested"], final["result"]) == ("failed", False, None)
    error = final["error"]
    # A server-side loss, never a user Stop. The exact code and body are Task 11's for a lost lease (Review notes 10).
    assert error["http_status"] == 503 and error["failure_code"] in ("worker_lost", "http_error"), error
    assert error["body"]["detail"]["request_id"] == "cao-m3-post"
    # settle_own_lapsed reads the bound fence and never advances, takes over or releases it. A peer reaper
    # settles only after acquiring SOL COMPOSE itself, which advances the epoch: the unchanged fence is the
    # evidence that A, not a peer, settled the row.
    assert _fence(deployment) == fence_running
    assert _job(deployment, next_id)["status"] == "queued"
    assert _operation_count(deployment) == 2
    assert [call["instance"] for call in _provider_calls(deployment)] == ["a"]
    assert _conversation(deployment) == [("user", _CONTENT)]

7. **Timing.** The lease waits are 3.5 s (SOL and membership) and 5.5 s (claim, fixed by the `ge=5` settings
   floor), each followed by a bounded retry loop. A reaper that settles too early is caught by the immediate

   every 1.0 s with a 3 s lease, so this has at least 2 s of margin. Under heavy host load a renewal can be late
   enough that the "live owner" race arms see a lapsed lease. The membership heartbeat is the second guard (both
   must lapse), which is why the membership is heartbeated rather than registered once.

   T13 `:945`) and drops it. Comparisons against a literal (`_accepted`, `_worker_lost`, `_request_cancelled`)
   never involved the status body's clock and are unchanged.
