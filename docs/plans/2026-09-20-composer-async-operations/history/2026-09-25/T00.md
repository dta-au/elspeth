Usage: python m6_result_size.py <sqlite-copy-path>

async def main(path: str) -> int:
    engine = create_session_engine(f"sqlite:///{path}")
    service = SessionServiceImpl(engine, telemetry=build_sessions_telemetry(), log=structlog.get_logger("m6"))
    with engine.connect() as conn:
        session_ids = [UUID(row[0]) for row in conn.execute(text("SELECT id FROM sessions WHERE archived_at IS NULL"))]
        totals = conn.execute(text("SELECT (SELECT count(*) FROM sessions), (SELECT count(*) FROM composition_states), "
                                   "(SELECT count(*) FROM composition_proposals WHERE status = 'pending')")).one()
    print(f"STORE sessions={totals[0]} states={totals[1]} pending_proposals={totals[2]}")

def url_suffix(node: ast.AST) -> str | None:
    text: str | None = None
    if isinstance(node, ast.Constant) and isinstance(node.value, str):
        text = node.value
    elif isinstance(node, ast.JoinedStr) and node.values:

    tree = ast.parse(source)
