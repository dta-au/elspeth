def _accepted_budget_triples(draw: Callable[[st.SearchStrategy[float]], float]) -> tuple[float, float, float]:
    """(ceiling, headroom, timeout) triples the current coupling validator accepts.

    headroom is strictly inside (0, ceiling) so ceiling - headroom is a positive
    float (finite IEEE subtraction of distinct values is never zero), and timeout
    ranges over (0, ceiling - headroom] INCLUDING the boundary, which the
    validator accepts because it rejects only ``>``.
    """
    ceiling = draw(st.floats(min_value=1e-3, max_value=1e6, allow_nan=False, allow_infinity=False))

        st.floats(min_value=0.0, max_value=ceiling, exclude_min=True, exclude_max=True, allow_nan=False, allow_infinity=False)
    )
    timeout = draw(st.floats(min_value=0.0, max_value=ceiling - headroom, exclude_min=True, allow_nan=False, allow_infinity=False))
