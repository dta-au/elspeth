async def test_post_is_accepted_before_the_held_provider_and_poll_returns_the_exact_former_result(tmp_path: Path, kind: str) -> None:
    app, service = _make_app(tmp_path)
    composer = _BlockingRecordingComposer()
    app.state.composer_service = composer
    async with _client(app) as setup:
        created = await setup.post("/api/sessions", json={"title": "Gate 2"})
    assert created.status_code == 201, created.text
    session_id = uuid.UUID(created.json()["id"])
    operation_id = str(uuid.uuid4())
    if kind == "compose_message":
        path = f"/api/sessions/{session_id}/messages"
        body = {"operation_id": operation_id, "content": "Build a pipeline"}
    else:
        await service.add_message(session_id, "user", "Build a pipeline", writer_principal="route_user_message")
        path = f"/api/sessions/{session_id}/recompose"
        body = {"operation_id": operation_id}
    # One slot: the first POST spends it, so a replay that were charged would answer 429 (D2).
    app.state.rate_limiter = ComposerRateLimiter(limit=1)
    worker = install_composer_async_worker(app)
    poll_path = f"/api/sessions/{session_id}/operations/{operation_id}"

    post_client = _client(app)
    accepted = await asyncio.wait_for(post_client.post(path, json=body), timeout=5.0)
    assert accepted.status_code == 202, accepted.text
    assert ComposerOperationAcceptedResponse.model_validate_json(accepted.content) == ComposerOperationAcceptedResponse(
        operation_id=operation_id,
        kind=kind,
        status="queued",
        poll_after_ms=app.state.settings.composer_async_poll_after_ms,
    )
    assert composer.calls == [], "the POST answered before any provider call"

    worker_task = asyncio.create_task(worker.run_until_idle())
    await asyncio.wait_for(composer.first_call_started.wait(), timeout=5.0)
    # The provider is mid-call. The POST connection goes away now.
    await post_client.aclose()
    assert post_client.is_closed
    await asyncio.sleep(0)
    assert not worker_task.done(), "closing the POST connection must not end the turn"
    assert not composer.release_first_call.is_set()

    async with _client(app) as poller:
        running = await poller.get(poll_path)
        assert running.status_code == 200, running.text
        assert running.headers["cache-control"] == "no-store"
        running_status = ComposerOperationStatusResponse.model_validate_json(running.content)
        assert running_status.status == "running"
        assert running_status.result is None
        assert running_status.error is None

        replay = await poller.post(path, json=body)
        assert replay.status_code == 202, replay.text
        assert replay.json()["operation_id"] == operation_id
        assert replay.json()["status"] == "running"
        assert len(composer.calls) == 1, "a same-id replay mid-turn must not start a second provider turn"

        composer.release_first_call.set()
        await asyncio.wait_for(worker_task, timeout=10.0)
        final = await poller.get(poll_path)

    assert final.status_code == 200, final.text
    settled = ComposerOperationStatusResponse.model_validate_json(final.content)
    assert settled.status == "completed"
    assert settled.error is None
    persisted = await service.get_messages(session_id, limit=None)
    users = [message for message in persisted if message.role == "user"]
    assistants = [message for message in persisted if message.role == "assistant"]
    assert len(users) == 1, "exactly one originating user row"

"""Whole-tree gate: no test drives the freeform composer routes the synchronous way.

``POST /api/sessions/{id}/messages`` and ``/recompose`` answer 202 and settle
by poll (plan Task 13). A test that still POSTs either route and reads a 200
body, a terminal 4xx/5xx, or an escaping exception is testing a contract that
no longer exists. Such a test goes through ``tests.helpers.composer_operations``
(``settle_sync`` / ``submit_and_settle``), which submit, drive the worker and
poll to the terminal row.

A direct POST stays legal only when it is an admission test: it sends a body
carrying ``operation_id`` and every status it asserts is one the short POST
itself answers (202, or a 401/403/404/409/422/429 refusal). Raw ASGI scopes
aimed at the two paths are forbidden outright: the 499 disconnect contract they
encoded is retired for these routes (the Stop test uses the cancel endpoint).

The instrument is AST-based. Its blind spots, stated so a clean result is read
honestly: a URL built by a helper call or passed in as a parameter is not
resolved (only a string literal, an f-string, or a same-function ``name = <str>``
assignment is). Statuses are read within 25 lines of the POST, and only as
``<name>.status_code == N`` on the name the POST's response is bound to
(through ``await`` / ``asyncio.wait_for``), so a later poll's ``== 200`` is not
charged to the POST; an unbound POST falls back to any ``status_code ==`` /
``status ==`` in the window. The synthetic controls below prove the classifier
on each shape it claims.
"""

@dataclass(frozen=True, slots=True)
class RoutePost:
    path: str
    line: int
    function: str
    statuses: frozenset[int]
    has_operation_id: bool
    inside_pytest_raises: bool
    kind: str


def _url_tail(node: ast.expr) -> str | None:
    if isinstance(node, ast.Constant) and isinstance(node.value, str):
        return node.value
    if isinstance(node, ast.JoinedStr) and node.values:
        last = node.values[-1]
        if isinstance(last, ast.Constant) and isinstance(last.value, str):
            return last.value
    return None


def _is_operation_body(node: ast.expr) -> bool:
    if isinstance(node, ast.Dict):
        return any(isinstance(key, ast.Constant) and key.value == "operation_id" for key in node.keys)
    return isinstance(node, ast.Call) and isinstance(node.func, ast.Name) and node.func.id in _BODY_BUILDERS


def _function_assignments(function: ast.AST | None) -> tuple[dict[str, str], frozenset[str]]:
    """Same-function ``name = <route url string>`` and ``name = <operation body>`` assignments."""
    urls: dict[str, str] = {}
    bodies: set[str] = set()
    if function is None:
        return urls, frozenset()
    for node in ast.walk(function):
        if not (isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name)):
            continue
        name = node.targets[0].id
        tail = _url_tail(node.value)
        if tail is not None and _ROUTE_TAIL.search(tail):
            urls[name] = tail
        if _is_operation_body(node.value):
            bodies.add(name)
    return urls, frozenset(bodies)


def _call_has_operation_body(call: ast.Call, body_names: frozenset[str]) -> bool:
    for keyword in call.keywords:
        if keyword.arg != "json":
            continue
        if isinstance(keyword.value, ast.Name):
            return keyword.value.id in body_names
        return _is_operation_body(keyword.value)
    return False


def _candidate(node: ast.AST, decorator_ids: frozenset[int]) -> tuple[str, ast.expr] | None:
    if isinstance(node, ast.Call) and id(node) not in decorator_ids and isinstance(node.func, ast.Attribute):
        if node.func.attr == "post" and node.args:
            return "post", node.args[0]
        if node.func.attr == "request" and len(node.args) >= 2 and isinstance(node.args[0], ast.Constant) and node.args[0].value == "POST":
            return "request", node.args[1]
        return None
    if isinstance(node, ast.Dict):
        keys = [key.value if isinstance(key, ast.Constant) else None for key in node.keys]
        if "type" in keys and "path" in keys:
            return "asgi_scope", node.values[keys.index("path")]
    return None


def _assigned_name(node: ast.AST, parents: dict[ast.AST, ast.AST]) -> str | None:
    """The name a POST's response is bound to (through ``await`` and ``asyncio.wait_for(<post>, …)``), if any."""
    current: ast.AST = node
    parent = parents.get(current)
    while isinstance(parent, ast.Await) or (isinstance(parent, ast.Call) and current in parent.args):
        current = parent
        parent = parents.get(current)
    if isinstance(parent, ast.Assign) and len(parent.targets) == 1 and isinstance(parent.targets[0], ast.Name):
        return parent.targets[0].id
    return None


def scan_route_posts(relative_path: str, source: str, tree: ast.Module) -> list[RoutePost]:
    """Every POST-shaped reference to the two freeform routes in one module."""
    lines = source.splitlines()
    parents = {child: parent for parent in ast.walk(tree) for child in ast.iter_child_nodes(parent)}
    decorator_ids = frozenset(
        id(decorator)
        for node in ast.walk(tree)
        if isinstance(node, ast.FunctionDef | ast.AsyncFunctionDef)
        for decorator in node.decorator_list
    )
    posts: list[RoutePost] = []
    for node in ast.walk(tree):
        candidate = _candidate(node, decorator_ids)
        if candidate is None:
            continue
        kind, url = candidate
        enclosing: ast.AST | None = parents.get(node)
        inside_raises = False
        while enclosing is not None and not isinstance(enclosing, ast.FunctionDef | ast.AsyncFunctionDef):
            if isinstance(enclosing, ast.With | ast.AsyncWith) and any(
                ast.unparse(item.context_expr).startswith("pytest.raises") for item in enclosing.items
            ):
                inside_raises = True
            enclosing = parents.get(enclosing)
        url_names, body_names = _function_assignments(enclosing)
        tail = url_names.get(url.id) if isinstance(url, ast.Name) else _url_tail(url)
        if tail is None or not _ROUTE_TAIL.search(tail):
            continue
        line = node.lineno if isinstance(node, ast.Call | ast.Dict) else 0
        is_def = isinstance(enclosing, ast.FunctionDef | ast.AsyncFunctionDef)
        end = enclosing.end_lineno if is_def and enclosing.end_lineno is not None else len(lines)
        window = "\n".join(lines[line - 1 : min(line - 1 + _WINDOW_LINES, end)])
        response_name = _assigned_name(node, parents)
        status_assert = (
            _STATUS_ASSERT if response_name is None else re.compile(rf"\b{re.escape(response_name)}\.status_code\s*==\s*(\d{{3}})\b")
        )
        posts.append(
            RoutePost(
                path=relative_path,
                line=line,
                function=enclosing.name if is_def else "<module>",
                statuses=frozenset(int(code) for code in status_assert.findall(window)),
                has_operation_id=isinstance(node, ast.Call) and _call_has_operation_body(node, body_names),
                inside_pytest_raises=inside_raises,
                kind=kind,
            )
        )
    return posts


def is_offender(post: RoutePost) -> bool:
    """A POST that still expects the synchronous contract."""
    if post.kind == "asgi_scope" or post.inside_pytest_raises or not post.has_operation_id:
        return True
    return not post.statuses or not post.statuses <= ADMISSION_STATUSES


def _scan_text(source: str) -> list[RoutePost]:
    return scan_route_posts("synthetic.py", source, ast.parse(source))

```python
"""Short admission for the two asynchronous freeform composer routes (plan Task 13, D2).

``POST /{session_id}/messages`` and ``POST /{session_id}/recompose`` no longer
run a composer turn. They validate, reserve one durable
``composer_async_operations`` row and answer 202; the app-lifespan
``ComposerAsyncWorker`` runs the turn, and the client polls
``GET /api/sessions/{session_id}/operations/{operation_id}`` for its outcome.

Admission order (D2). Authentication and session ownership run as
dependencies before the strict body is parsed
(``_verify_session_ownership_before_body``). Then, here:

1. an existing row for ``(session_id, operation_id)`` is looked up first, so a
   replay of an accepted operation answers 202 with its current status and is
   never charged against the per-user rate limit;
2. only a new operation is charged (``WebRateLimiter.check``: 429, or 503 when
   the shared limiter is unavailable). The limiter is a coroutine and ``admit``
   runs on the worker pool, so Task 4's ``admit_composer_operation`` charges it
   between the lookup and ``admit``;
3. ``ComposerAsyncOperationAuthority.admit`` then, in one locked transaction,
   re-compares a racing duplicate (409 ``composer_operation_conflict``),
   refuses a second nonterminal operation for the session (409
   ``composer_operation_active`` naming the live one, D8), bounds the cluster
   queue (429 ``composer_queue_full`` with ``Retry-After``, D3) and inserts the
   ``queued`` row.

Nothing here waits for a compose lock, a session-operation lease, a provider or
a worker claim. Checks whose answer can change while a job waits for the
session lease (transcript shape, ``state_id``, quota, stale state) are the
worker's, and arrive in the terminal envelope.
"""

from __future__ import annotations

from functools import partial
from typing import Final

from fastapi import HTTPException, Request

from elspeth.web.auth.models import UserIdentity
from elspeth.web.config import WebSettings
from elspeth.web.coordination.composer_operation_authority import (
    ComposerAsyncOperationAuthority,
    admit_composer_operation,
)
from elspeth.web.middleware.rate_limit import WebRateLimiter
from elspeth.web.sessions.composer_operations import (
    ComposerOperationActiveError,
    ComposerOperationCapacityError,
    ComposerOperationConflictError,
    ComposerOperationKind,
    composer_operation_request_hash,
)
from elspeth.web.sessions.protocol import SessionRecord
from elspeth.web.sessions.routes._helpers import _failure_log_request_id
from elspeth.web.sessions.schemas import ComposerOperationAcceptedResponse, RecomposeRequest, SendMessageRequest

COMPOSER_OPERATION_CONFLICT_DETAIL: Final = "This operation id was already used for a different request."
COMPOSER_OPERATION_ACTIVE_DETAIL: Final = (
    "Another composer request is still running for this session. Wait for it to finish, then send again."
)


async def accept_composer_operation(
    request: Request,
    *,
    session: SessionRecord,
    user: UserIdentity,
    rate_limiter: WebRateLimiter,
    kind: ComposerOperationKind,
    body: SendMessageRequest | RecomposeRequest,
) -> ComposerOperationAcceptedResponse:
    """Reserve (or replay) one freeform composer operation and return its 202 handle."""
    authority: ComposerAsyncOperationAuthority = request.app.state.composer_async_operation_authority
    settings: WebSettings = request.app.state.settings
    try:
        # Lookup first; the per-user limiter is charged only for an id with no
        # row (D2), then the locked admission transaction (Task 4 D-4a).
        record, _created = await admit_composer_operation(
            authority,
            rate_limit=partial(rate_limiter.check, user.user_id),
            session_id=session.id,
            operation_id=body.operation_id,
            kind=kind,
            request_hash=composer_operation_request_hash(session_id=session.id, kind=kind, request=body),
            actor_user_id=user.user_id,
            request_id=_failure_log_request_id(request),
            request_json=body.model_dump_json(),
            deadline_seconds=settings.composer_timeout_seconds,
            max_nonterminal=settings.composer_async_max_queued_operations,
        )
    except ComposerOperationConflictError:
        raise HTTPException(
            status_code=409,
            detail={"error_type": "composer_operation_conflict", "detail": COMPOSER_OPERATION_CONFLICT_DETAIL},
        ) from None
    except ComposerOperationActiveError as exc:
        raise HTTPException(
            status_code=409,
            detail={
                "error_type": "composer_operation_active",
                "detail": COMPOSER_OPERATION_ACTIVE_DETAIL,
                "operation_id": exc.operation_id,
                "kind": exc.kind,
            },
        ) from None
    except ComposerOperationCapacityError as exc:
        retry_after = exc.retry_after_seconds
        raise HTTPException(
            status_code=429,
            detail={
                "error_type": "composer_queue_full",
                "detail": f"The composer queue is full. Try again in {retry_after} seconds.",
                "retry_after": retry_after,
            },
            headers={"Retry-After": str(retry_after)},
        ) from None
    return ComposerOperationAcceptedResponse(
        operation_id=record.operation_id,
        kind=record.kind,
        status=record.status,
        poll_after_ms=settings.composer_async_poll_after_ms,
    )
```

def strip_request_id(detail: object) -> object:
    """Drop the D4 correlation id from a terminal dict ``detail`` so it compares with the former body.

    Route-test apps mount no ``RequestIdMiddleware``, so the worker persisted
    ``request_id=None``. The key is asserted present with a ``str | None``

    TERMINAL envelopes only: an admission refusal from these test apps has no
    app-level handler and carries no ``request_id``.
    """
    if type(detail) is not dict:
        return detail
    assert "request_id" in detail, f"terminal dict detail lacks the D4 request_id: {detail}"
    value = detail["request_id"]
    assert value is None or type(value) is str, detail
    return {key: item for key, item in detail.items() if key != "request_id"}

        Replaces the raw-ASGI disconnect test (elspeth-e08063c3a5): the 499

            """compose() parks forever; records whether it was cancelled."""

            def __init__(self, started: asyncio.Event) -> None:

def _operation_terminal(polled: HttpResponse | None) -> HttpResponse | None:
    """The former synchronous response for a terminal operation poll; None while running or when the poll failed.

    ``completed`` yields ``HttpResponse(200, <MessageWithStateResponse>)``; ``failed`` yields
    ``HttpResponse(error.http_status, error.body)``, the exact body the synchronous route used to send, so the
    terminal classification in ``run_prompt`` (422_detail, planner_5xx, auth) is unchanged.
    """
    if polled is None or polled.status_code != 200 or not isinstance(polled.body, dict):
        return None
    status = polled.body.get("status")
    if status == "completed":
        result = polled.body.get("result")
        return HttpResponse(200, result, json.dumps(result))
    if status == "failed":
        error = polled.body.get("error")
        if isinstance(error, dict) and isinstance(error.get("http_status"), int):
            body = error.get("body")
            return HttpResponse(error["http_status"], body, json.dumps(body))
        return HttpResponse(500, None, json.dumps(polled.body))
    return None
