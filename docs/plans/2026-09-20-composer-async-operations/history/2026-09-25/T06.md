```python
"""Line-only re-pin instrument for one source file's identities in the sessions mutation-authority gate.

For every reviewed identity on the file it finds the live scanned site that has the same
(symbol, table, operation, fingerprint, ordinal, authority, connection_escape). It demands that site
is unique on both sides and prints REPIN when only ``line`` moved. Live sites with no reviewed twin
print as UNREVIEWED, which is where the new writers' fp/line come from.

Usage: ``repin_service_lines.py [REL]``. REL defaults to service.py; Task 6 also runs it on
``src/elspeth/web/coordination/repository.py`` (F-B2/C2 edits there).
"""

from __future__ import annotations

import sys
from collections import Counter
from pathlib import Path

from tests.unit.architecture import test_session_db_mutation_authority as gate

REL = sys.argv[1] if len(sys.argv) > 1 else "src/elspeth/web/sessions/service.py"


def _key(site: gate.WriterIdentity) -> tuple[object, ...]:
    return (site.symbol, site.table, site.operation, site.fingerprint, site.ordinal, site.authority, site.connection_escape)


def main() -> int:
    root = Path.cwd()
    live = gate.scan_production_writers([root / REL], anchor=root)
    reviewed = [site for site in (*gate._REVIEWED_WRITERS, *gate._REVIEWED_READ_CONNECTIONS) if site.path == REL]
    live_lines: dict[tuple[object, ...], list[int]] = {}
    for site in live:
        live_lines.setdefault(_key(site), []).append(site.line)
    reviewed_counts = Counter(_key(site) for site in reviewed)
    failed = False
    for site in reviewed:
        lines = live_lines.get(_key(site), [])
        if reviewed_counts[_key(site)] != 1 or len(lines) != 1:
            failed = True
            print(
                f"NOT-UNIQUE-OR-MISSING {site.symbol} {site.operation} {site.table} "
                f"fp={site.fingerprint}#{site.ordinal} reviewed_line={site.line} live_lines={lines}"
            )
        elif lines[0] != site.line:
            print(f"REPIN {site.symbol} {site.operation} {site.table} fp={site.fingerprint}#{site.ordinal} line={site.line} -> line={lines[0]}")
    for site in live:
        if _key(site) not in reviewed_counts:
            print(
                f"UNREVIEWED {site.symbol} {site.operation} {site.table} fp={site.fingerprint}#{site.ordinal} "
                f"authority={site.authority} escape={site.connection_escape} line={site.line}"
            )
    print(f"reviewed={len(reviewed)} live={len(live)}")
    return 1 if failed else 0


if __name__ == "__main__":
    sys.exit(main())
```

        settles the job with one compare-and-swap on its running fence and
        ``cancel_requested_at IS NULL``. Every failure, including a lost
        CAS, raises inside the transaction, so nothing above commits without
        the terminal.

        refused at the entry proof (F-B2/C2) with
        ``ComposerOperationCancelledDuringTurn``, before any write; the CAS
        clause restates that rule (spec §4). A lost CAS raises
        ``ComposerOperationCancelledDuringTurn`` when a committed cancel marker
        won it on this owner's running row, and ``ComposerOperationFenceLost``

            raise TypeError("assistant_record must be an exact ChatMessageRecord or None")
        if type(audit_cohort) is not tuple or any(type(draft) is not AuditMessageDraft for draft in audit_cohort):

        if audit_composition_state_id is not None and type(audit_composition_state_id) is not UUID:
            raise TypeError("audit_composition_state_id must be an exact UUID or None")
        if (assistant is None) == (assistant_record is None):
            # Refused before any lock is taken; _terminal_assistant_on_connection
            # repeats the check only as its exhaustive-return guard.
            raise ValueError("complete_composer_async_operation needs exactly one of assistant or assistant_record")
        context = running.session_operation_context
        sid = str(running.claim.session_id)
        audit_csid = str(audit_composition_state_id) if audit_composition_state_id is not None else None
        now = self._now()

                self._session_process_locked_begin(sid) as conn,
                self._session_write_lock(conn, sid),
                self._session_composer_mutation_transaction(

                    session_operation_context=context,
                    expected_kind=SessionOperationKind.COMPOSE,

                    audit_only=False,
                )

- **Cancellation of the awaiting task after commit.** Both terminal methods run through
