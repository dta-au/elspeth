def _imported_modules(module_file: str) -> set[str]:
    tree = ast.parse(Path(module_file).read_text(encoding="utf-8"))

def _require_uuid(value: object, field_name: str) -> None:
    if type(value) is not UUID:
        raise AuditIntegrityError(f"{field_name} must be an exact UUID")


def _require_canonical_uuid_text(value: object, field_name: str) -> None:
    if type(value) is not str or len(value) != _CANONICAL_UUID_LENGTH:
        raise AuditIntegrityError(f"{field_name} must be a canonical UUID string")
    try:
        parsed = UUID(value)
    except ValueError as exc:
        raise AuditIntegrityError(f"{field_name} must be a canonical UUID string") from exc
    if str(parsed) != value:
        raise AuditIntegrityError(f"{field_name} must be a canonical UUID string")


def _require_bounded_text(value: object, field_name: str, *, max_length: int) -> None:
    if type(value) is not str or not value.strip() or len(value) > max_length:
        raise AuditIntegrityError(f"{field_name} must be a nonblank exact string of at most {max_length} characters")


def _require_optional_bounded_text(value: object, field_name: str, *, max_length: int) -> None:
    if value is not None:
        _require_bounded_text(value, field_name, max_length=max_length)


def _require_aware_datetime(value: object, field_name: str) -> None:
    if type(value) is not datetime or value.tzinfo is None or value.utcoffset() is None:
        raise AuditIntegrityError(f"{field_name} must be a timezone-aware datetime")


def _require_optional_aware_datetime(value: object, field_name: str) -> None:
    if value is not None:
        _require_aware_datetime(value, field_name)


def composer_operation_request_hash(
    *,
    session_id: UUID,
    kind: ComposerOperationKind,
    request: SendMessageRequest | RecomposeRequest,
) -> str:
    """Bind one strict request DTO to a session and operation kind.

    The client ``operation_id`` is transport retry state, not request
    semantics, so it is excluded. Defaults and explicit ``None`` values are
    materialized so an omitted field and its explicit default share one
    canonical replay identity.
    """

    _require_uuid(session_id, "composer_operation_request_hash.session_id")
    if kind not in COMPOSER_OPERATION_KIND_VALUES:
        raise AuditIntegrityError("Composer operation hashing requires a closed composer operation kind")
    config = type(request).model_config
    if "strict" not in config or config["strict"] is not True or "extra" not in config or config["extra"] != "forbid":
        raise AuditIntegrityError("Composer operation hashing requires a strict, extra-forbid request DTO")
    if "operation_id" not in type(request).model_fields:
        raise AuditIntegrityError("Composer operation request DTO is missing operation_id")
    normalized = request.model_dump(
        mode="json",
        exclude={"operation_id"},
        exclude_unset=False,
        exclude_defaults=False,
        exclude_none=False,
    )
    return stable_hash(
        {
            "schema": COMPOSER_OPERATION_REQUEST_SCHEMA,
            "session_id": str(session_id),
            "kind": kind,
            "request": normalized,
        }
    )


def composer_operation_result_hash(result_json: str) -> str:
    """Hash stored terminal result JSON, refusing text that is not canonical.

    The stored text must be exactly ``canonical_json`` of its own parse (RFC
    8785), so the digest equals ``stable_hash(json.loads(result_json))`` and a
    stored row has one byte representation. Writers produce it with
    ``canonical_json(dto.model_dump(mode="json"))``.
    """

    if type(result_json) is not str:
        raise AuditIntegrityError("Composer operation result JSON must be an exact string")
    try:
        parsed = json.loads(result_json)
    except ValueError as exc:
        raise AuditIntegrityError("Composer operation result JSON is not valid JSON") from exc
    try:
        canonical = canonical_json(parsed)
    except (TypeError, ValueError) as exc:
        raise AuditIntegrityError("Composer operation result JSON cannot be canonicalized") from exc
    if canonical != result_json:
        raise AuditIntegrityError("Composer operation result JSON is not canonical")
    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


class ComposerOperationError(BaseModel):
    """Public terminal error envelope of a composer operation.

    It is both the ``error`` member of a poll response and the stored JSON of a
    ``failed`` row (result schema ``composer_operation_error.v1``). ``body`` is
    the exact JSON body the synchronous route would have sent: ``{"detail":
    ...}`` plus any keys an app exception handler adds, with ``request_id``
    injected the way ``handle_http_exception`` does. The four non-``http_error``
    codes carry fixed statuses, and ``diagnostic_id`` (the server-side
    reference for an internal defect) exists exactly for ``operation_failed``.

    Tier-1 response rules: the config is ``schemas._StrictResponse``'s
    (``strict=True, extra="forbid"``), restated here because ``schemas``
    imports this module and cannot be imported back.
    """

    model_config = ConfigDict(strict=True, extra="forbid")

    http_status: int = Field(ge=400, le=599)
    failure_code: ComposerOperationFailureCode
    error_type: str | None
    body: dict[str, JsonValue]
    diagnostic_id: str | None

    @model_validator(mode="after")
    def _validate_envelope(self) -> ComposerOperationError:
        if "detail" not in self.body:
            raise ValueError("body must carry the route's detail key")
        if self.failure_code == "operation_failed":
            fixed_status: int | None = 500
        elif self.failure_code == "worker_lost":
            fixed_status = 503
        elif self.failure_code == "deadline_expired":
            fixed_status = 504
        elif self.failure_code == "request_cancelled":
            fixed_status = 499
        else:
            fixed_status = None
        if fixed_status is not None and self.http_status != fixed_status:
            raise ValueError(f"failure_code {self.failure_code!r} requires http_status {fixed_status}")
        if (self.failure_code == "operation_failed") != (self.diagnostic_id is not None):
            raise ValueError("diagnostic_id is required for operation_failed and forbidden otherwise")
        if self.diagnostic_id is not None and not self.diagnostic_id.strip():
            raise ValueError("diagnostic_id must be nonblank")
        if self.error_type is not None and not self.error_type.strip():
            raise ValueError("error_type must be nonblank when present")
        return self

@dataclass(frozen=True, slots=True)
class ComposerOperationRecord:
    """Tier-1 read of one ``composer_async_operations`` row.

    Tokens and session-operation lease columns are deliberately absent: a
    record is safe to log, return from a poll, or attach to an exception.
    Readers must hand every datetime over timezone-aware (SQLite returns
    naive values for ``DateTime(timezone=True)``; normalize to UTC first).
    """

    session_id: UUID
    operation_id: str
    kind: ComposerOperationKind
    status: ComposerOperationStatus
    request_hash: str
    actor_user_id: str
    request_id: str | None
    deadline_at: datetime
    created_at: datetime
    updated_at: datetime
    started_at: datetime | None
    settled_at: datetime | None
    cancel_requested_at: datetime | None
    claim_owner_instance_id: str | None
    user_message_id: UUID | None
    failure_code: ComposerOperationFailureCode | None
    result_schema: str | None
    result_json: str | None
    result_sha256: str | None
    request_json: str | None

    def __post_init__(self) -> None:
        _require_uuid(self.session_id, "ComposerOperationRecord.session_id")
        _require_canonical_uuid_text(self.operation_id, "ComposerOperationRecord.operation_id")
        if self.kind not in COMPOSER_OPERATION_KIND_VALUES:
            raise AuditIntegrityError("ComposerOperationRecord.kind is outside the closed kind set")
        if self.status not in COMPOSER_OPERATION_STATUS_VALUES:
            raise AuditIntegrityError("ComposerOperationRecord.status is outside the closed status set")
        if not is_lower_sha256_hex(self.request_hash):
            raise AuditIntegrityError("ComposerOperationRecord.request_hash must be a lowercase SHA-256 hex digest")
        _require_bounded_text(self.actor_user_id, "ComposerOperationRecord.actor_user_id", max_length=_IDENTIFIER_MAX_LENGTH)
        _require_optional_bounded_text(self.request_id, "ComposerOperationRecord.request_id", max_length=_IDENTIFIER_MAX_LENGTH)
        _require_aware_datetime(self.deadline_at, "ComposerOperationRecord.deadline_at")
        _require_aware_datetime(self.created_at, "ComposerOperationRecord.created_at")
        _require_aware_datetime(self.updated_at, "ComposerOperationRecord.updated_at")
        _require_optional_aware_datetime(self.started_at, "ComposerOperationRecord.started_at")
        _require_optional_aware_datetime(self.settled_at, "ComposerOperationRecord.settled_at")
        _require_optional_aware_datetime(self.cancel_requested_at, "ComposerOperationRecord.cancel_requested_at")
        _require_optional_bounded_text(
            self.claim_owner_instance_id,
            "ComposerOperationRecord.claim_owner_instance_id",
            max_length=_IDENTIFIER_MAX_LENGTH,
        )
        if self.user_message_id is not None:
            _require_uuid(self.user_message_id, "ComposerOperationRecord.user_message_id")
        if self.failure_code is not None and self.failure_code not in COMPOSER_OPERATION_FAILURE_CODE_VALUES:
            raise AuditIntegrityError("ComposerOperationRecord.failure_code is outside the closed failure-code set")
        if self.result_schema is not None and (
            type(self.result_schema) is not str or self.result_schema not in COMPOSER_OPERATION_RESULT_SCHEMA_VALUES
        ):
            raise AuditIntegrityError("ComposerOperationRecord.result_schema is outside the closed result-schema set")
        if self.result_json is not None and type(self.result_json) is not str:
            raise AuditIntegrityError("ComposerOperationRecord.result_json must be an exact string")
        if self.result_sha256 is not None and not is_lower_sha256_hex(self.result_sha256):
            raise AuditIntegrityError("ComposerOperationRecord.result_sha256 must be a lowercase SHA-256 hex digest")
        if self.request_json is not None and (
            type(self.request_json) is not str or len(self.request_json) > COMPOSER_OPERATION_REQUEST_JSON_MAX_LENGTH
        ):
            raise AuditIntegrityError("ComposerOperationRecord.request_json must be an exact bounded string")
        self._require_time_order()
        self._require_status_bundle()
        if self.result_json is not None and composer_operation_result_hash(self.result_json) != self.result_sha256:
            raise AuditIntegrityError("ComposerOperationRecord.result_sha256 does not match its result_json")

    def _require_time_order(self) -> None:
        if self.updated_at < self.created_at:
            raise AuditIntegrityError("ComposerOperationRecord.updated_at precedes created_at")
        if self.deadline_at < self.created_at:
            raise AuditIntegrityError("ComposerOperationRecord.deadline_at precedes created_at")
        if self.started_at is not None and self.started_at < self.created_at:
            raise AuditIntegrityError("ComposerOperationRecord.started_at precedes created_at")
        if self.settled_at is not None and self.settled_at < self.created_at:
            raise AuditIntegrityError("ComposerOperationRecord.settled_at precedes created_at")
        if self.started_at is not None and self.settled_at is not None and self.settled_at < self.started_at:
            raise AuditIntegrityError("ComposerOperationRecord.settled_at precedes started_at")
        if self.cancel_requested_at is not None and self.cancel_requested_at < self.created_at:
            raise AuditIntegrityError("ComposerOperationRecord.cancel_requested_at precedes created_at")

    def _require_status_bundle(self) -> None:
        result_and_failure_absent = (
            self.settled_at is None
            and self.failure_code is None
            and self.result_schema is None
            and self.result_json is None
            and self.result_sha256 is None
        )
        if self.status == "queued":
            # No turn side effect is allowed before ``running``, so a queued
            # row has not started and has recorded no user message.
            legal = self.request_json is not None and self.started_at is None and self.user_message_id is None and result_and_failure_absent
        elif self.status == "running":
            legal = (
                self.request_json is not None
                and self.started_at is not None
                and self.claim_owner_instance_id is not None
                and result_and_failure_absent
            )
        elif self.status == "completed":
            # The completion CAS requires ``cancel_requested_at IS NULL`` and a
            # cancel never relabels a terminal row, so a completed row carries
            # no cancel marker.
            legal = (
                self.request_json is None
                and self.claim_owner_instance_id is None
                and self.started_at is not None
                and self.cancel_requested_at is None
                and self.failure_code is None
                and self.result_schema == COMPOSER_OPERATION_RESULT_SCHEMA_SUCCESS
                and self.result_json is not None
                and self.result_sha256 is not None
                and self.settled_at is not None
            )
        else:
            # ``failed`` may never have started (queued deadline expiry, a
            # cancel of a queued row, a precondition refused before running).
            legal = (
                self.request_json is None
                and self.claim_owner_instance_id is None
                and self.failure_code is not None
                and self.result_schema == COMPOSER_OPERATION_RESULT_SCHEMA_ERROR
                and self.result_json is not None
                and self.result_sha256 is not None
                and self.settled_at is not None
            )
        if not legal:
            raise AuditIntegrityError(f"ComposerOperationRecord status bundle is illegal for status {self.status!r}")
