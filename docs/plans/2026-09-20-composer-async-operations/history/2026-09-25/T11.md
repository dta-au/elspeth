```python
@dataclass(frozen=True)
class _Harness:
    service: SessionServiceImpl
    engine: Engine
    session_id: UUID
    authority: ComposerAsyncOperationAuthority
    registry: ComposerProgressRegistry
    worker: ComposerAsyncWorker
    services: ComposerAppServices


async def _harness(tmp_path: Path, composer: object) -> _Harness:
    app, service = _make_app(tmp_path)
    engine = cast(Engine, app.state.session_engine)
    session = await service.create_session("alice", "Composer worker", "local")
    registry = ComposerProgressRegistry()
    services = ComposerAppServices(
        session_service=service,
        composer_service=cast(Any, composer),
        settings=app.state.settings,
        catalog_service=app.state.catalog_service,
        operator_profile_registry=app.state.operator_profile_registry,
        scoped_secret_resolver=app.state.scoped_secret_resolver,
        session_engine=engine,
        plugin_snapshot_for_user_id=app.state.plugin_snapshot_factory,
        progress_registry=registry,
        compose_locks=_SessionComposeLockRegistry(),
    )
    authority = ComposerAsyncOperationAuthority(
        engine,
        owner_instance_id=service.session_operation_owner_instance_id,
        claim_lease_seconds=30,
    )
    worker = ComposerAsyncWorker(
        services=services,
        authority=authority,
        concurrency=4,
        scan_interval_seconds=0.05,
        claim_lease_seconds=30,
        owner_instance_id=service.session_operation_owner_instance_id,
        process_recovery=ProcessRecovery(),
        instance_draining=threading.Event(),
    )
    return _Harness(
        service=service,
        engine=engine,
        session_id=session.id,
        authority=authority,
        registry=registry,
        worker=worker,
        services=services,
    )


def _peer_worker(h: _Harness, *, owner_instance_id: str) -> ComposerAsyncWorker:
    """A second instance's worker over the same store: it owns none of this instance's rows."""
    return ComposerAsyncWorker(
        services=h.services,
        authority=h.authority,
        concurrency=4,
        scan_interval_seconds=0.05,
        claim_lease_seconds=30,
        owner_instance_id=owner_instance_id,
        process_recovery=ProcessRecovery(),
        instance_draining=threading.Event(),
    )


async def _admit(h: _Harness, *, kind: ComposerOperationKind = "compose_message", deadline_seconds: float = 85.0) -> str:
    operation_id = str(uuid4())
    request: SendMessageRequest | RecomposeRequest
    if kind == "compose_message":
        request = SendMessageRequest(operation_id=operation_id, content=_REQUEST_CONTENT)
    else:
        request = RecomposeRequest(operation_id=operation_id)
    record, created = await run_sync_in_worker(
        h.authority.admit,
        session_id=h.session_id,
        operation_id=operation_id,
        kind=kind,
        request_hash=composer_operation_request_hash(session_id=h.session_id, kind=kind, request=request),
        actor_user_id="alice",
        request_id=_REQUEST_ID,
        request_json=request.model_dump_json(),
        deadline_seconds=deadline_seconds,
        max_nonterminal=64,
    )
    assert created is True
    assert record.status == "queued"
    return operation_id


async def _record(h: _Harness, operation_id: str) -> ComposerOperationRecord:
    record = await run_sync_in_worker(h.authority.get, session_id=h.session_id, operation_id=operation_id)
    assert record is not None
    return record


async def _failure(h: _Harness, operation_id: str) -> ComposerOperationError:
    record = await _record(h, operation_id)
    assert record.status == "failed"
    assert record.request_json is None
    assert record.result_json is not None
    return ComposerOperationError.model_validate_json(record.result_json)


async def _seed_user_message(h: _Harness, content: str) -> None:
    lease = await SessionOperationLease.acquire(
        h.service.session_operation_authority,
        session_id=h.session_id,
        operation_kind=SessionOperationKind.COMPOSE,
        owner_instance_id=h.service.session_operation_owner_instance_id,
        lease_seconds=h.service.session_operation_lease_seconds,
    )
    async with lease:
        await h.service.add_message(
            h.session_id,
            "user",
            content,
            writer_principal="route_user_message",
            session_operation_context=lease.context,
        )


def _expire_session_fence(engine: Engine, session_id: UUID) -> None:
    """Age the session fence into the past: the process that started the row is dead."""
    with engine.begin() as conn:
        conn.execute(
            update(session_operation_fences_table)
            .where(session_operation_fences_table.c.session_id == str(session_id))
            .values(lease_expires_at=datetime(2000, 1, 1, tzinfo=UTC))
        )


async def _start_with_dead_owner(h: _Harness, operation_id: str) -> None:
    """Claim and start the row exactly as a worker would, then never adopt or renew it."""
    claims = await run_sync_in_worker(h.authority.claim_next, limit=1)
    assert [claim.operation_id for claim in claims] == [operation_id]
    await run_sync_in_worker(
        h.service.session_operation_authority.start_composer_async_operation,
        claims[0],
        owner_instance_id=h.service.session_operation_owner_instance_id,
        lease_seconds=h.service.session_operation_lease_seconds,
    )


def _expire_claim(engine: Engine, session_id: UUID) -> None:
    """Age the queued claim past its lease: its owner's renewal outage outlived it.

    ``created_at`` moves back as well, because T03 pins ``claim_expires_at > created_at``.
    """
    with engine.begin() as conn:
        conn.execute(
            update(composer_async_operations_table)
            .where(composer_async_operations_table.c.session_id == str(session_id))
            .values(
                created_at=datetime(2000, 1, 1, tzinfo=UTC),
                claim_expires_at=datetime(2000, 1, 1, 0, 0, 1, tzinfo=UTC),
            )
        )


def _archive_by_sql(engine: Engine, session_id: UUID) -> None:
    """Mark the session archived without the ARCHIVE lease's filesystem phases (D7's end state)."""
    with engine.begin() as conn:
        conn.execute(update(sessions_table).where(sessions_table.c.id == str(session_id)).values(archived_at=datetime.now(UTC)))


def _change_session_owner_by_sql(engine: Engine, session_id: UUID, *, change: str) -> None:
    """Take the session away from the admitting actor without archiving it (T00 P15's two live causes)."""
    session_row = sessions_table.c.id == str(session_id)
    with engine.begin() as conn:
        if change == "reassigned":
            # sessions.user_id is an FK (RESTRICT) to identities: the new owner must exist.
            ensure_test_identity(conn, identity_id="bob")
            conn.execute(update(sessions_table).where(session_row).values(user_id="bob"))
        else:
            # Same user id under another auth provider (_verify_session_ownership's third predicate).
            conn.execute(update(sessions_table).where(session_row).values(auth_provider_type="oidc"))

class _ParkedComposer:
    """``compose`` parks until cancelled and records the task that ran it."""

    def __init__(self, *, attach: ComposerLLMCall | None = None, extra_cancel: bool = False) -> None:
        self.started = asyncio.Event()
        self.task: asyncio.Task[Any] | None = None
        self.cancelling_seen: int | None = None
        self.cancelled = False
        self._attach = attach
        self._extra_cancel = extra_cancel

    async def compose(self, *args: Any, **kwargs: Any) -> None:
        del args, kwargs
        self.task = asyncio.current_task()
        self.started.set()
        try:
            await asyncio.Event().wait()
        except asyncio.CancelledError as exc:
            self.cancelled = True
            assert self.task is not None
            self.cancelling_seen = self.task.cancelling()
            if self._attach is not None:
                # Production attaches buffered calls to the SAME instance
                # (llm_response_parsing.attach_llm_calls), keeping the marker.
                cast(Any, exc).llm_calls = (self._attach,)
            if self._extra_cancel:
                # An external cancel racing the worker's marker.
                self.task.cancel()
            raise
        raise AssertionError("unreachable: compose never completes")


def _heartbeat_fails_once_running(
    monkeypatch: pytest.MonkeyPatch,
    registry: ComposerProgressRegistry,
    running: asyncio.Event,
    failure: BaseException,
) -> None:
    """Every heartbeat renewal after ``running`` raises ``failure``; time moves only with the heartbeat."""
    clock = _FakeHeartbeatClock()
    real_renew = registry.renew_request

    async def renew_request(lease: ComposerRequestLease) -> None:
        if running.is_set():
            raise failure
        await real_renew(lease)

    async def wait(seconds: float) -> None:
        await running.wait()
        await clock.wait(seconds)

    monkeypatch.setattr(registry, "renew_request", renew_request)
    monkeypatch.setattr(
        _helpers,
        "_COMPOSER_HEARTBEAT_TIMER",
        _helpers._ComposerHeartbeatTimer(now=clock.now, wait=wait),
    )


def test_operation_cancel_markers_are_recognised_by_identity_only() -> None:
    for marker in (_COMPOSER_OPERATION_CANCEL_REQUESTED, _COMPOSER_OPERATION_LEASE_LOST, _COMPOSER_OPERATION_SHUTDOWN):
        assert _composer_operation_cancel_of(asyncio.CancelledError(marker)) is marker
    assert _composer_operation_cancel_of(asyncio.CancelledError()) is None
    assert _composer_operation_cancel_of(asyncio.CancelledError("shutdown")) is None
    # An equal but distinct instance is not the worker's marker.
    assert _composer_operation_cancel_of(asyncio.CancelledError(_ComposerOperationCancel(kind="cancel_requested"))) is None

@pytest.mark.asyncio
async def test_local_cancel_reaches_the_worker_turn_task_and_settles_request_cancelled(tmp_path: Path) -> None:
    composer = _ParkedComposer()
    h = await _harness(tmp_path, composer)
    operation_id = await _admit(h)
    idle = asyncio.create_task(h.worker.run_until_idle())
    await asyncio.wait_for(composer.started.wait(), timeout=5)

    marked = await run_sync_in_worker(
        h.authority.request_cancel,
        session_id=h.session_id,
        operation_id=operation_id,
        cancelled_failure=request_cancelled_error,
    )
    assert marked is not None
    assert marked.status == "running"
    assert marked.cancel_requested_at is not None
    assert h.worker.signal_local_cancel(session_id=h.session_id, operation_id=operation_id) is True
    await asyncio.wait_for(idle, timeout=10)

    turn_task = composer.task
    assert turn_task is not None
    # The cancel landed in the worker's own job task, as that task's only request...
    assert turn_task.get_name() == f"composer-async-job-{operation_id}"
    assert composer.cancelling_seen == 1
    # ...and the worker retired exactly that request: the task ended normally.
    assert turn_task.done()
    assert not turn_task.cancelled()
    assert turn_task.cancelling() == 0
    assert await _failure(h, operation_id) == request_cancelled_error(request_id=_REQUEST_ID)
    snapshot = await h.registry.get_latest(str(h.session_id))
    assert snapshot.phase == "cancelled"
    assert snapshot.reason == "client_cancelled"
    assert snapshot.inflight_requests == 0
    assert h.worker.signal_local_cancel(session_id=h.session_id, operation_id=operation_id) is False

@pytest.mark.asyncio
async def test_cancelled_turn_persists_its_llm_call_audit_before_the_terminal_write(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    llm_call = _llm_call(
        status=ComposerLLMCallStatus.CANCELLED,
        model_returned=None,
        prompt_tokens=None,
        completion_tokens=None,
        total_tokens=None,
        provider_request_id=None,
        error_class="CancelledError",
        error_message="CancelledError",
    )
    composer = _ParkedComposer(attach=llm_call)
    h = await _harness(tmp_path, composer)
    operation_id = await _admit(h)
    audit_rows_at_terminal: list[int] = []
    real_fail = h.service.fail_composer_async_operation

    async def recording_fail(running: Any, *, failure: ComposerOperationError) -> ComposerOperationRecord:
        audit_rows_at_terminal.append(len(_llm_call_audit_rows(await h.service.get_messages(h.session_id))))
        return await real_fail(running, failure=failure)

    monkeypatch.setattr(h.service, "fail_composer_async_operation", recording_fail)
    idle = asyncio.create_task(h.worker.run_until_idle())
    await asyncio.wait_for(composer.started.wait(), timeout=5)
    await run_sync_in_worker(
        h.authority.request_cancel,
        session_id=h.session_id,
        operation_id=operation_id,
        cancelled_failure=request_cancelled_error,
    )
    assert h.worker.signal_local_cancel(session_id=h.session_id, operation_id=operation_id) is True
    await asyncio.wait_for(idle, timeout=10)

    # The sidecar was durable when the terminal write began, not merely after it.
    assert audit_rows_at_terminal == [1]
    rows = _llm_call_audit_rows(await h.service.get_messages(h.session_id))
    assert len(rows) == 1
    _row, tool_call = rows[0]
    assert tool_call["call"]["status"] == "cancelled"
    assert tool_call["call"]["messages_hash"] == llm_call.messages_hash
    assert await _failure(h, operation_id) == request_cancelled_error(request_id=_REQUEST_ID)


@pytest.mark.asyncio
async def test_external_cancel_racing_the_cancel_marker_settles_the_committed_cancel_and_keeps_unwinding(tmp_path: Path) -> None:
    composer = _ParkedComposer(extra_cancel=True)
    h = await _harness(tmp_path, composer)
    operation_id = await _admit(h)
    idle = asyncio.create_task(h.worker.run_until_idle())
    await asyncio.wait_for(composer.started.wait(), timeout=5)

    await run_sync_in_worker(
        h.authority.request_cancel,
        session_id=h.session_id,
        operation_id=operation_id,
        cancelled_failure=request_cancelled_error,
    )
    assert h.worker.signal_local_cancel(session_id=h.session_id, operation_id=operation_id) is True
    await asyncio.wait({idle}, timeout=10)

    assert composer.cancelling_seen == 1
    turn_task = composer.task
    assert turn_task is not None
    # Two pending requests: the marker was retired, the external one keeps unwinding.
    assert turn_task.cancelled()
    assert idle.cancelled()
    # The external cancel classifies as worker_lost, but the cancel marker
    # committed first: Task 6's fail CAS refuses a non-cancel failure while the
    # marker is set, and the worker re-settles the committed cancel (spec §4).
    assert await _failure(h, operation_id) == request_cancelled_error(request_id=_REQUEST_ID)


@pytest.mark.asyncio
async def test_stop_settles_a_running_turn_as_worker_lost_and_joins_it(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> None:
    signals: list[int] = []
    monkeypatch.setattr("elspeth.web.process_recovery.os.kill", lambda _pid, signal_number: signals.append(signal_number))

    assert snapshot.reason == "service_setup_failed"
    assert snapshot.inflight_requests == 0
    assert signals == []
```

```python
One ``ComposerAsyncWorker`` runs on every web instance. It claims committed
``queued`` rows of ``composer_async_operations``, starts each one under the
session's COMPOSE ``SessionOperationLease`` (the composite start binds the

loss, the task the per-job watcher cancels with a ``_ComposerOperationCancel``

    exc_class: str | None  # the class that ended the turn (m2 diagnostics); None on success


@final
class _ComposerJob:
    """One claimed operation and the task that owns it."""

    __slots__ = (
        "claim",
        "claim_live_until",
        "delivered",
        "local_cancel",
        "monotonic_at_running",
        "phase",
        "shutdown_requested",
        "task",
        "unstarted_stop",
    )

    def __init__(self, claim: ComposerOperationClaim, *, claim_live_until: float) -> None:
        self.claim = claim
        # Conservative local bound on the queued claim (review-pass M1): read
        # before claim_next ran, advanced from before each successful renewal.
        self.claim_live_until = claim_live_until
        self.delivered: _ComposerOperationCancel | None = None
        self.local_cancel = asyncio.Event()
        # F-C5: the monotonic instant the start composite returned.
        self.monotonic_at_running: float | None = None
        self.phase: _JobPhase = "claimed"
        self.shutdown_requested = False
        self.task: asyncio.Task[_JobOutcome] | None = None
        # F-C3: set by a local Stop or by a renewed row that carries a committed
        # cancel or a passed deadline; read only while waiting for the compose lock.
        self.unstarted_stop = asyncio.Event()

    def observe_claim_row(self, record: ComposerOperationRecord) -> None:
        """A renewed queued row ends the lock wait once a cancel committed or its deadline passed (DB time)."""
        if record.cancel_requested_at is not None or record.deadline_at <= record.updated_at:
            self.unstarted_stop.set()

    def deliver(self, marker: _ComposerOperationCancel) -> None:
        """Cancel the running turn once, with a marker; completion wins after the turn returns."""
        task = self.task
        if self.phase != "running" or self.delivered is not None or task is None or task.done():
            return
        self.delivered = marker
        task.cancel(marker)

    def request_shutdown(self) -> None:
        self.shutdown_requested = True
        task = self.task
        if task is None or task.done():
            return
        if self.phase == "claimed":
            # Nothing has started: a plain cancel releases the claim.
            task.cancel()
        elif self.phase == "running":
            self.deliver(_COMPOSER_OPERATION_SHUTDOWN)
        # "starting": the job reads shutdown_requested after adopt and never
        # begins the turn. "settling": the terminal write finishes.


def _current_task() -> asyncio.Task[Any]:
    task = asyncio.current_task()
    if task is None:
        raise RuntimeError("composer operation job requires an owning task")
    return task


def _log_watcher_retry(exc: Exception, *, operation_id: str, consecutive_failures: int) -> None:
    """Class names only, no ``exc_info``: SQLAlchemy cause chains carry the URL."""
    if consecutive_failures >= _WATCHER_ESCALATE_AFTER_FAILURES:
        slog.error(
            "composer_operation.watcher_poll_degraded",
            operation_id=operation_id,
            exc_class=type(exc).__name__,
            consecutive_failures=consecutive_failures,
        )
    else:
        slog.warning(
            "composer_operation.watcher_poll_retrying",
            operation_id=operation_id,
            exc_class=type(exc).__name__,
            consecutive_failures=consecutive_failures,
        )


def _turn_input(record: ComposerOperationRecord, *, budget_seconds: float) -> ComposerTurnInput:
    """Rebuild the typed turn from the row alone: any instance may run it (spec §3)."""
    if record.request_json is None:
        raise AuditIntegrityError(f"Running composer operation {record.operation_id} has no request body")
    request: SendMessageRequest | RecomposeRequest
    try:
        if record.kind == "compose_message":
            request = SendMessageRequest.model_validate_json(record.request_json)
        else:
            request = RecomposeRequest.model_validate_json(record.request_json)
    except ValidationError as exc:
        raise AuditIntegrityError(f"Composer operation {record.operation_id} stored an invalid request body") from exc
    if request.operation_id != record.operation_id:
        raise AuditIntegrityError(f"Composer operation {record.operation_id} stored another operation's request")
    return ComposerTurnInput(
        session_id=record.session_id,
        operation_id=record.operation_id,
        kind=record.kind,
        actor_user_id=record.actor_user_id,
        request=request,
        request_id=record.request_id,
        budget_seconds=budget_seconds,
    )


@final
class ComposerAsyncWorker:
    """Claims, runs, cancels and reaps freeform composer operations for one instance."""

    __slots__ = (
        "_authority",
        "_claim_lease_seconds",
        "_concurrency",
        "_instance_draining",
        "_jobs",
        "_owner_instance_id",
        "_process_recovery",
        "_reap_task",
        "_scan_interval_seconds",
        "_scan_task",
        "_services",
        "_stopping",
    )

    def __init__(
        self,
        *,
        services: ComposerAppServices,
        authority: ComposerAsyncOperationAuthority,
        concurrency: int,
        scan_interval_seconds: float,
        claim_lease_seconds: int,
        owner_instance_id: str,
        process_recovery: ProcessRecovery,
        instance_draining: threading.Event,
    ) -> None:
        if type(services) is not ComposerAppServices:
            raise TypeError("services must be an exact ComposerAppServices")
        if type(authority) is not ComposerAsyncOperationAuthority:
            raise TypeError("authority must be an exact ComposerAsyncOperationAuthority")
        if type(process_recovery) is not ProcessRecovery:
            raise TypeError("process_recovery must be an exact ProcessRecovery")
        if type(instance_draining) is not threading.Event:
            raise TypeError("instance_draining must be an exact threading.Event")
        if type(concurrency) is not int or concurrency < 1:
            raise ValueError("concurrency must be a positive exact integer")
        if not scan_interval_seconds > 0:
            raise ValueError("scan_interval_seconds must be positive")
        if type(claim_lease_seconds) is not int or not 5 <= claim_lease_seconds <= 600:
            raise ValueError("claim_lease_seconds must be an exact integer from 5 through 600")
        if type(owner_instance_id) is not str or not owner_instance_id.strip():
            raise ValueError("owner_instance_id must be a nonblank exact string")
        self._services = services
        self._authority = authority
        self._concurrency = concurrency
        self._scan_interval_seconds = scan_interval_seconds
        self._claim_lease_seconds = claim_lease_seconds
        self._owner_instance_id = owner_instance_id
        self._process_recovery = process_recovery
        self._instance_draining = instance_draining
        self._jobs: dict[tuple[UUID, str], _ComposerJob] = {}
        self._scan_task: asyncio.Task[None] | None = None
        self._reap_task: asyncio.Task[None] | None = None
        self._stopping = False

    # ── lifespan surface ────────────────────────────────────────────────

    def start(self) -> None:
        """Start the scan and reap loops; a dead loop drains and restarts the process."""
        if self._scan_task is not None or self._reap_task is not None:
            raise RuntimeError("composer async worker already started")
        if self._stopping:
            raise RuntimeError("composer async worker is stopping")
        reap_interval_seconds = float(heartbeat_interval_seconds(self._services.session_service.session_operation_lease_seconds))
        self._scan_task = asyncio.create_task(
            self._loop_forever(
                self._scan_once,
                interval_seconds=self._scan_interval_seconds,
                failed_event="composer_async_worker_scan_failed",
                escalating_event="composer_async_worker_scan_escalating",
            ),
            name="composer-async-worker-scan",
        )
        self._reap_task = asyncio.create_task(
            self._loop_forever(
                self.reap_once,
                interval_seconds=reap_interval_seconds,
                failed_event="composer_async_worker_reap_failed",
                escalating_event="composer_async_worker_reap_escalating",
            ),
            name="composer-async-worker-reap",
        )
        self._scan_task.add_done_callback(self._escalate_loop_failure)
        self._reap_task.add_done_callback(self._escalate_loop_failure)

    async def stop(self) -> None:
        """Stop owned jobs (each settles under its own fence), join them, then stop the loops.

        Runs while this instance's membership lease is still live, so a
        running job's SOL can still prove the terminal write. A loop that
        had already died re-raises its stored failure here, after every job
        joined, exactly as ``RegisteredWebInstanceMembership.stop`` does.
        """
        self._stopping = True
        jobs = tuple(self._jobs.values())
        for job in jobs:
            job.request_shutdown()
        job_tasks = tuple(job.task for job in jobs if job.task is not None)
        if job_tasks:
            await asyncio.wait(job_tasks)
        loop_tasks = tuple(task for task in (self._scan_task, self._reap_task) if task is not None)
        self._scan_task = None
        self._reap_task = None
        for loop_task in loop_tasks:
            loop_task.cancel()
        if loop_tasks:
            await asyncio.wait(loop_tasks)
        # M8 / codex C6: a scan that was inside claim_next when stopping began
        # hands its claims back (_claim_and_spawn re-checks after the await), so
        # nothing new should be here. Drain anything that is, rather than leave
        # a job running past shutdown.
        late = tuple(job for job in self._jobs.values() if job not in jobs)
        for job in late:
            job.request_shutdown()
        late_tasks = tuple(job.task for job in late if job.task is not None)
        if late_tasks:
            await asyncio.wait(late_tasks)
        failures: list[BaseException] = []
        for finished in (*loop_tasks, *job_tasks, *late_tasks):
            if finished.cancelled():
                continue
            error = finished.exception()
            if error is not None:
                failures.append(error)
        if len(failures) == 1:
            raise failures[0]
        if failures:
            raise BaseExceptionGroup("Composer async worker shutdown failed", failures)

    async def run_until_idle(self) -> None:
        """Test seam (D16): reap once, then run every claimable job to its end in this loop."""
        await self.reap_once()
        while True:
            spawned = await self._claim_and_spawn()
            if not spawned:
                return
            outcomes = await asyncio.gather(*spawned)
            if _JobOutcome.SETTLED not in outcomes:
                return

    def signal_local_cancel(self, *, session_id: UUID, operation_id: str) -> bool:
        """Wake the local owner's watcher; the durable marker must already be committed."""
        key = (session_id, operation_id)
        if key not in self._jobs:
            return False
        job = self._jobs[key]
        job.local_cancel.set()
        # F-C3: a claimed job still waiting for the compose lock observes it there.
        job.unstarted_stop.set()
        return True

    async def reap_once(self) -> int:
        """Settle expired or cancelled queued rows and ``running`` rows whose session fence lapsed; never runs a turn."""
        settled = 0
        queued = await run_sync_in_worker(self._authority.list_expired_queued, limit=_REAP_BATCH_SIZE)
        for record in queued:
            if (record.session_id, record.operation_id) in self._jobs:
                continue
            if await self._reap_queued(record) is _ReapOutcome.SETTLED:
                settled += 1
        rows = await run_sync_in_worker(self._authority.list_expired_running, limit=_REAP_BATCH_SIZE)
        for row in rows:
            if (row.session_id, row.operation_id) in self._jobs:
                continue
            if await self._reap_running(row) is _ReapOutcome.SETTLED:
                settled += 1
        return settled

    # ── loops ───────────────────────────────────────────────────────────

    async def _loop_forever(
        self,
        step: Callable[[], Awaitable[int]],
        *,
        interval_seconds: float,
        failed_event: str,
        escalating_event: str,
    ) -> None:
        consecutive_failures = 0
        while True:
            await asyncio.sleep(interval_seconds)
            try:
                await step()
            except _LOOP_TRANSIENT_ERRORS as exc:
                # exc_info omitted: SQLAlchemy cause chains carry the URL.
                consecutive_failures += 1
                if consecutive_failures >= _LOOP_MAX_CONSECUTIVE_FAILURES:
                    slog.error(escalating_event, exc_class=type(exc).__name__, consecutive_failures=consecutive_failures)
                    raise
                slog.error(failed_event, exc_class=type(exc).__name__, consecutive_failures=consecutive_failures)
            else:
                consecutive_failures = 0

    def _escalate_loop_failure(self, completed: asyncio.Task[None]) -> None:
        if not completed.cancelled() and completed.exception() is not None:
            self._instance_draining.set()
            self._process_recovery.request_shutdown()

    async def _scan_once(self) -> int:
        if self._stopping or self._instance_draining.is_set():
            return 0
        return len(await self._claim_and_spawn())

    async def _claim_and_spawn(self) -> tuple[asyncio.Task[_JobOutcome], ...]:
        free = self._concurrency - len(self._jobs)
        if free <= 0:
            return ()
        # Read before the claim CAS commits, so the local bound errs early.
        claim_live_until = _monotonic() + self._claim_lease_seconds
        # Shielded (M8 / codex C6): a claim that committed in the pool thread is
        # always seen here, even when stop() cancels this loop mid-call, so it is
        # handed back rather than left to lapse.
        claims = await _join_shielded_task_after_cancellation(
            asyncio.create_task(
                run_sync_in_worker(self._authority.claim_next, limit=free),
                name="composer-async-claim-next",
            )
        )
        current = _current_task()
        if self._stopping or self._instance_draining.is_set() or current.cancelling() > 0:
            await self._release_claims_on_stop(claims)
            if current.cancelling() > 0:
                # The cancellation the shield absorbed resumes now.
                raise asyncio.CancelledError()
            return ()
        spawned: list[asyncio.Task[_JobOutcome]] = []
        for claim in claims:
            task = await self._spawn_or_supersede(claim, claim_live_until=claim_live_until)
            if task is not None:
                spawned.append(task)
        return tuple(spawned)

    async def _release_claims_on_stop(self, claims: tuple[ComposerOperationClaim, ...]) -> int:
        released = 0
        for claim in claims:
            outcome = await _join_shielded_task_after_cancellation(
                asyncio.create_task(
                    self._release_unstarted_claim(claim),
                    name=f"composer-async-claim-release-{claim.operation_id}",
                )
            )
            if outcome is _JobOutcome.REQUEUED:
                released += 1
        if claims:
            slog.info("composer_operation.claims_released_on_stop", claimed=len(claims), released=released)
        return released

    async def _spawn_or_supersede(self, claim: ComposerOperationClaim, *, claim_live_until: float) -> asyncio.Task[_JobOutcome] | None:
        key = (claim.session_id, claim.operation_id)
        if key not in self._jobs:
            return self._spawn(claim, claim_live_until=claim_live_until)
        # Review-pass M1: this worker's own job lost its claim. A renewal outage
        # outlived the lease while the job waited for the compose lock, and
        # claim_next handed the row back to this instance. The stale job can no
        # longer start (its token is superseded, so the composite refuses it).
        # Stop it if nothing has started, hand the new claim back, and let the
        # next scan take the row cleanly. Never raise: a dead scan loop drains
        # the whole instance.
        stale = self._jobs[key]
        if stale.phase == "claimed" and stale.task is not None and not stale.task.done():
            stale.task.cancel()
        outcome = await self._release_unstarted_claim(claim)
        slog.warning(
            "composer_operation.stale_claim_superseded",
            operation_id=claim.operation_id,
            stale_phase=stale.phase,
            new_claim=outcome.value,
        )
        return None

    def _spawn(self, claim: ComposerOperationClaim, *, claim_live_until: float) -> asyncio.Task[_JobOutcome]:
        key = (claim.session_id, claim.operation_id)
        job = _ComposerJob(claim, claim_live_until=claim_live_until)
        task = asyncio.create_task(self._run_job(job), name=f"composer-async-job-{claim.operation_id}")
        job.task = task
        self._jobs[key] = job
        task.add_done_callback(partial(self._forget_job, key))
        return task

    def _forget_job(self, key: tuple[UUID, str], completed: asyncio.Task[_JobOutcome]) -> None:
        del self._jobs[key]
        if completed.cancelled():
            return
        error = completed.exception()
        if error is not None:
            # A defect in one job never takes the instance down (a poison row
            # would crash-loop it); its row is left to the reaper.
            slog.error("composer_operation.job_failed", operation_id=key[1], exc_class=type(error).__name__)

    # ── one job ─────────────────────────────────────────────────────────

    async def _run_job(self, job: _ComposerJob) -> _JobOutcome:
        claim = job.claim
        claim_renewal = asyncio.create_task(
            self._renew_claim(job),
            name=f"composer-async-claim-renewal-{claim.operation_id}",
        )
        try:
            # Same order as the routes: in-process lock, then the durable fence.
            # The wait observes a Stop, a committed cancel and the deadline
            # (F-C3), so a turn holding this lock elsewhere (a guided turn on the
            # same session) cannot hold a claimed job past any of them.
            lock = await self._services.compose_locks.get_lock(str(claim.session_id))
            if not await self._acquire_compose_lock(job, lock):
                return await self._settle_stopped_wait(job, claim_renewal)
            try:
                started = await self._start(job, claim_renewal)
                if type(started) is _JobOutcome:
                    return started
                return await self._run_started(job, started)
            finally:
                lock.release()
        except asyncio.CancelledError:
            if job.phase == "claimed":
                # Nothing has started: hand the row back rather than make
                # peers wait out the claim lease.
                await _join_shielded_task_after_cancellation(
                    asyncio.create_task(
                        self._release_unstarted_claim(claim),
                        name=f"composer-async-claim-release-{claim.operation_id}",
                    )
                )
            raise
        finally:
            if job.phase == "claimed":
                await self._stop_claim_renewal(claim_renewal)

    async def _acquire_compose_lock(self, job: _ComposerJob, lock: asyncio.Lock) -> bool:
        """Wait for the in-process compose lock; ``False`` (lock not held) once the wait must stop (F-C3)."""
        operation_id = job.claim.operation_id
        acquire = asyncio.create_task(lock.acquire(), name=f"composer-async-lock-wait-{operation_id}")
        stop = asyncio.create_task(job.unstarted_stop.wait(), name=f"composer-async-lock-stop-{operation_id}")
        waited = False
        try:
            await asyncio.wait({acquire, stop}, return_when=asyncio.FIRST_COMPLETED)
            waited = True
        finally:
            stop.cancel()
            if not acquire.done():
                acquire.cancel()
            await asyncio.wait({acquire, stop})
            acquired = acquire.done() and not acquire.cancelled() and acquire.exception() is None
            if acquired and (not waited or job.unstarted_stop.is_set()):
                # Handed the lock while being cancelled or stopped: give it straight back.
                lock.release()
        if acquire.cancelled():
            return False
        acquire.result()  # re-raises a lock defect
        return not job.unstarted_stop.is_set()

    async def _settle_stopped_wait(self, job: _ComposerJob, claim_renewal: asyncio.Task[None]) -> _JobOutcome:
        """Settle a claimed job whose lock wait a cancel or its deadline ended; the lock is not held (F-C3).

        The background renewal stops first, so it cannot race the settle and
        log a spurious ``claim_renewal_failed``. The claim keeps at least
        lease minus interval seconds. One more renewal then re-reads the row
        under database time and proves the claim is still live, and
        ``settle_unstarted`` re-verifies the marker or the deadline under the
        session lock.
        """
        claim = job.claim
        await self._stop_claim_renewal(claim_renewal)
        try:
            record = await run_sync_in_worker(self._authority.renew_claim, claim)
        except ComposerOperationFenceLost:
            return _JobOutcome.ABANDONED
        if record.cancel_requested_at is not None:
            reason = "cancel_requested"
            failure = request_cancelled_error(request_id=record.request_id)
        elif record.deadline_at <= record.updated_at:
            reason = "deadline_expired"
            failure = deadline_expired_error(
                request_id=record.request_id,
                timeout_seconds=self._services.settings.composer_timeout_seconds,
            )
        else:
            # signal_local_cancel's contract: the cancel route commits the marker first.
            raise AuditIntegrityError(
                f"Composer operation {claim.operation_id} stopped its lock wait with no committed cancel and no passed deadline"
            )
        slog.info("composer_operation.lock_wait_settled", operation_id=claim.operation_id, reason=reason)
        return await self._settle_unstarted(claim, failure)

    async def _renew_claim(self, job: _ComposerJob) -> None:
        claim = job.claim
        interval_seconds = heartbeat_interval_seconds(self._claim_lease_seconds)
        retry_seconds = min(interval_seconds, _CLAIM_RENEW_RETRY_SECONDS)
        consecutive_failures = 0
        while True:
            await asyncio.sleep(interval_seconds if consecutive_failures == 0 else retry_seconds)
            attempted_at = _monotonic()
            try:
                record = await run_sync_in_worker(self._authority.renew_claim, claim)
            except _LOOP_TRANSIENT_ERRORS as exc:
                consecutive_failures += 1
                if _monotonic() >= job.claim_live_until:
                    # The claim has lapsed by this process's own clock. Stop; the
                    # composite (or the next scan's same-key path) decides from the row.
                    slog.error(
                        "composer_operation.claim_renewal_exhausted",
                        operation_id=claim.operation_id,
                        exc_class=type(exc).__name__,
                        consecutive_failures=consecutive_failures,
                    )
                    raise
                slog.warning(
                    "composer_operation.claim_renewal_retrying",
                    operation_id=claim.operation_id,
                    exc_class=type(exc).__name__,
                    consecutive_failures=consecutive_failures,
                )
                continue
            consecutive_failures = 0
            job.claim_live_until = attempted_at + self._claim_lease_seconds
            job.observe_claim_row(record)

    async def _stop_claim_renewal(self, renewal: asyncio.Task[None]) -> None:
        if not renewal.done():
            renewal.cancel()
        await asyncio.wait({renewal})
        if renewal.cancelled():
            return
        error = renewal.exception()
        if error is not None:
            # The composite start re-verifies the claim token and decides.
            slog.warning("composer_operation.claim_renewal_failed", exc_class=type(error).__name__)

    async def _release_unstarted_claim(self, claim: ComposerOperationClaim) -> _JobOutcome:
        try:
            await run_sync_in_worker(self._authority.release_claim, claim)
        except ComposerOperationFenceLost:
            return _JobOutcome.ABANDONED
        return _JobOutcome.REQUEUED

    async def _settle_unstarted(self, claim: ComposerOperationClaim, failure: ComposerOperationError) -> _JobOutcome:
        try:
            await run_sync_in_worker(
                self._authority.settle_unstarted,
                claim,
                session_id=claim.session_id,
                operation_id=claim.operation_id,
                failure=failure,
            )
        except ComposerOperationFenceLost:
            return _JobOutcome.ABANDONED
        return _JobOutcome.SETTLED

    async def _start(self, job: _ComposerJob, claim_renewal: asyncio.Task[None]) -> SessionOperationContext | _JobOutcome:
        claim = job.claim
        record = await run_sync_in_worker(self._authority.get, session_id=claim.session_id, operation_id=claim.operation_id)
        if record is None or record.status != "queued":
            # Cascaded away with its session, or the claim was superseded.
            return _JobOutcome.ABANDONED
        if record.deadline_at <= record.updated_at:
            # Expired while queued: no turn side effect has started (spec §3).
            return await self._settle_unstarted(
                claim,
                deadline_expired_error(
                    request_id=record.request_id,
                    timeout_seconds=self._services.settings.composer_timeout_seconds,
                ),
            )
        # The claim keeps at least lease - interval seconds; the composite is one transaction.
        await self._stop_claim_renewal(claim_renewal)
        job.phase = "starting"
        try:
            context = await self._start_composite(claim)
        except SessionOperationConflictError:
            # Another session operation holds the session: stay queued for a later scan.
            return await self._release_unstarted_claim(claim)
        except ComposerOperationCancelledBeforeStart:
            return await self._settle_unstarted(claim, request_cancelled_error(request_id=record.request_id))
        except SessionOperationFenceLost as exc:
            # D7: an archive won the session fence before the turn started.
            slog.info("composer_operation.start_fence_lost", operation_id=claim.operation_id, reason=exc.reason.value)
            return await self._settle_unstarted(
                claim,
                project_composer_operation_error(
                    exc,
                    request_id=record.request_id,
                    session_engine_dialect=self._services.session_engine.dialect.name,
                ),
            )
        except ComposerOperationFenceLost:
            return _JobOutcome.ABANDONED
        except Exception as exc:
            # F-B4: any other start defect settles the public generic envelope
            # (operation_failed 500 with a diagnostic id) while the claim is
            # still live. Otherwise the claim would lapse, be reclaimed, fail
            # again, and loop until the reaper settled a misleading 504.
            failure = project_composer_operation_error(
                exc,
                request_id=record.request_id,
                session_engine_dialect=self._services.session_engine.dialect.name,
            )
            slog.error(
                "composer_operation.start_failed",
                operation_id=claim.operation_id,
                exc_class=type(exc).__name__,
                failure_code=failure.failure_code,
                diagnostic_id=failure.diagnostic_id,
            )
            return await self._settle_unstarted(claim, failure)
        # F-C5: the compose budget is measured from the instant the composite
        # returned (Task 10's ComposerBudgetAnchor contract).
        job.monotonic_at_running = _monotonic()
        return context

    async def _start_composite(self, claim: ComposerOperationClaim) -> SessionOperationContext:
        service = self._services.session_service
        start_task = asyncio.create_task(
            run_sync_in_worker(
                service.session_operation_authority.start_composer_async_operation,
                claim,
                owner_instance_id=self._owner_instance_id,
                lease_seconds=service.session_operation_lease_seconds,
            ),
            name=f"composer-async-start-{claim.operation_id}",
        )
        # A composite that reached the worker may already have committed
        # ``running``: its context is joined even under cancellation so it is
        # always adopted and settled, never orphaned. An absorbed cancellation
        # stays counted on this task and is read after adopt.
        return await _join_shielded_task_after_cancellation(start_task)

    async def _run_started(self, job: _ComposerJob, context: SessionOperationContext) -> _JobOutcome:
        claim = job.claim
        service = self._services.session_service
        try:
            lease = await SessionOperationLease.adopt(
                service.session_operation_authority,
                context,
                lease_seconds=service.session_operation_lease_seconds,
            )
        except SessionOperationFenceLost:
            slog.warning("composer_operation.adopt_fence_lost", operation_id=claim.operation_id)
            return _JobOutcome.ABANDONED
        except Exception as exc:
            # F-B4: the composite already committed ``running``; never leave it to
            # a claim-expiry loop.
            return await self._settle_unadopted(job, context, exc)
        running = ComposerOperationRunning(claim=claim, session_operation_context=lease.context)
        task = _current_task()
        observation = ComposerTurnObservation()
        outcome = _JobOutcome.ABANDONED
        external_cancel = False
        try:
            record = await run_sync_in_worker(self._authority.get, session_id=claim.session_id, operation_id=claim.operation_id)
            if record is None or record.status != "running":
                raise AuditIntegrityError(f"Started composer operation {claim.operation_id} is not a running row")
            end = await self._run_or_refuse_turn(job, record, lease, running, task, observation)
            external_cancel = self._consume_delivered_cancel(job, task)
            failure = self._terminal_failure(job, end, record, external_cancel=external_cancel)
            outcome = await _join_shielded_task_after_cancellation(
                asyncio.create_task(
                    self._finish_running(running, lease, observation, end, failure, request_id=record.request_id),
                    name=f"composer-async-settle-{claim.operation_id}",
                )
            )
        finally:
            # D12: the terminal committed under this lease; closing it comes after.
            await self._close_lease(lease, operation_id=claim.operation_id, settled=outcome is _JobOutcome.SETTLED)
        if external_cancel:
            # A cancellation this worker did not deliver keeps unwinding, as the
            # disconnect watcher does (routes/_helpers.py:2309-2327).
            raise asyncio.CancelledError()
        return outcome

    async def _run_or_refuse_turn(
        self,
        job: _ComposerJob,
        record: ComposerOperationRecord,
        lease: SessionOperationLease,
        running: ComposerOperationRunning,
        task: asyncio.Task[Any],
        observation: ComposerTurnObservation,
    ) -> _TurnEnd:
        if record.started_at is None:
            raise AuditIntegrityError(f"Running composer operation {record.operation_id} has no started_at")
        if job.monotonic_at_running is None:
            raise RuntimeError(f"Composer operation {record.operation_id} reached its turn without a budget anchor")
        if job.shutdown_requested or task.cancelling() > 0:
            # Stopped, or cancelled from outside, while the composite start ran:
            # never begin a provider turn.
            return _TurnEnd(failure=None, cancelled=True, exc_class="CancelledError")
        # F-C5: queue time (deadline_at - started_at, DB clock) and every await
        # since the composite returned (monotonic) both consume the one budget.
        budget_anchor = ComposerBudgetAnchor(
            remaining_at_running_seconds=(record.deadline_at - record.started_at).total_seconds(),
            monotonic_at_running=job.monotonic_at_running,
        )
        if budget_anchor.remaining_seconds(monotonic_now=_monotonic()) <= 0:
            return _TurnEnd(
                failure=deadline_expired_error(
                    request_id=record.request_id,
                    timeout_seconds=self._services.settings.composer_timeout_seconds,
                ),
                cancelled=False,
                exc_class=None,
            )
        job.phase = "running"
        watcher = asyncio.create_task(self._watch(job, lease), name=f"composer-async-watch-{record.operation_id}")
        watcher.add_done_callback(partial(self._on_watcher_done, job))
        try:
            return await self._run_turn(job, record, lease, running, observation, budget_anchor=budget_anchor)
        finally:
            job.phase = "settling"
            watcher.cancel()
            await asyncio.wait({watcher})

    async def _run_turn(
        self,
        job: _ComposerJob,
        record: ComposerOperationRecord,
        lease: SessionOperationLease,
        running: ComposerOperationRunning,
        observation: ComposerTurnObservation,
        *,
        budget_anchor: ComposerBudgetAnchor,
    ) -> _TurnEnd:
        request_id = record.request_id
        try:
            # D10: the turn reports the budget it received at running; Task 10
            # refuses a turn whose budget_seconds differs from the anchor's.
            turn = _turn_input(record, budget_seconds=budget_anchor.remaining_at_running_seconds)
            await self._require_actor_owns_session(turn)
            async with _composer_request_lifecycle(
                self._services.progress_registry,
                session_id=str(turn.session_id),
                user_id=turn.actor_user_id,
                surface="freeform",
                owner_task=_current_task(),
            ) as request_lease:
                try:
                    await run_composer_turn(
                        self._services,
                        turn,
                        lease=lease,
                        running=running,
                        request_lease=request_lease,
                        observation=observation,
                        budget_anchor=budget_anchor,
                    )
                finally:
                    # No marker may land once the turn has returned: completion wins.
                    job.phase = "settling"
            return _TurnEnd(failure=None, cancelled=False, exc_class=None)
        except asyncio.CancelledError:
            return _TurnEnd(failure=None, cancelled=True, exc_class="CancelledError")
        except ComposerOperationCancelledDuringTurn:
            # A committed cancel fenced one of the turn's writes, or beat the
            # success CAS (Task 6, F-B2/C2): the cancel wins. The cohort join in
            # _finish_running runs before this terminal.
            return _TurnEnd(
                failure=request_cancelled_error(request_id=request_id),
                cancelled=False,
                exc_class="ComposerOperationCancelledDuringTurn",
            )
        except ComposerTurnDeadlineExpired:
            # F-C5: the budget measured immediately before compose() was spent
            # (Task 10). No provider call was made.
            return _TurnEnd(
                failure=deadline_expired_error(
                    request_id=request_id,
                    timeout_seconds=self._services.settings.composer_timeout_seconds,
                ),
                cancelled=False,
                exc_class="ComposerTurnDeadlineExpired",
            )
        except (ComposerOperationFenceLost, SessionOperationFenceLost) as exc:
            # The bound fence was lost mid-turn: the server stopped this turn
            # (D14). The fail write is then refused too, and _settle_running
            # falls back to settle_own_lapsed (F-M3).
            return _TurnEnd(failure=worker_lost_error(request_id=request_id), cancelled=False, exc_class=type(exc).__name__)
        except Exception as exc:
            # The app handlers' projection (Task 9): a preserved public HTTP error,
            # or the generic envelope with a server-side diagnostic id.
            return _TurnEnd(
                failure=project_composer_operation_error(
                    exc,
                    request_id=request_id,
                    session_engine_dialect=self._services.session_engine.dialect.name,
                ),
                cancelled=False,
                exc_class=type(exc).__name__,
            )

    async def _require_actor_owns_session(self, turn: ComposerTurnInput) -> None:
        """Ownership recheck under the COMPOSE fence (spec §3), as ``_verify_session_ownership``."""
        try:
            session = await self._services.session_service.get_session(turn.session_id)
        except ValueError:
            raise HTTPException(status_code=404, detail="Session not found") from None
        if (
            session.archived_at is not None
            or session.user_id != turn.actor_user_id
            or session.auth_provider_type != self._services.settings.auth_provider
        ):
            raise HTTPException(status_code=404, detail="Session not found")

    @staticmethod
    def _consume_delivered_cancel(job: _ComposerJob, task: asyncio.Task[Any]) -> bool:
        """Retire this worker's own cancel request; True when another stays pending.

        Read in the job task itself (spec gate 4). ``cancelling()`` counts
        every request on this task, so a marker converts to a terminal only
        when it was the sole pending cancellation: the disconnect watcher's
        rule (routes/_helpers.py:2289-2308).
        """
        pending = task.cancelling()
        if job.delivered is not None:
            pending = task.uncancel()
        return pending > 0

    @staticmethod
    def _terminal_failure(
        job: _ComposerJob,
        end: _TurnEnd,
        record: ComposerOperationRecord,
        *,
        external_cancel: bool,
    ) -> ComposerOperationError | None:
        if not end.cancelled:
            return end.failure
        if not external_cancel and job.delivered is _COMPOSER_OPERATION_CANCEL_REQUESTED:
            return request_cancelled_error(request_id=record.request_id)
        # Lease loss, shutdown, or a cancellation from outside the worker: the
        # server stopped this turn (D14 worker_lost).
        return worker_lost_error(request_id=record.request_id)

    async def _finish_running(
        self,
        running: ComposerOperationRunning,
        lease: SessionOperationLease,
        observation: ComposerTurnObservation,
        end: _TurnEnd,
        failure: ComposerOperationError | None,
        *,
        request_id: str | None,
    ) -> _JobOutcome:
        """Audit first, then the one terminal (spec §3 l.192-194, §4 l.234-237)."""
        if failure is not None:
            # F-B2/C2: every non-success exit joins the turn's LLM-call and tool
            # audit rows BEFORE any failure terminal, request_cancelled included.
            # This covers a Stop, lease loss or shutdown that landed after
            # compose() returned, and a committed cancel that fenced a tail write
            # or beat the success CAS. It is a no-op before compose() returned or
            # once the cohort is durable, and it logs rather than raises on a
            # database or fence failure (Task 10), so the terminal still decides.
            await persist_cancelled_turn_audit(self._services, running, observation)
        return await self._settle_running(running, lease, observation, end, failure, request_id=request_id)

    async def _settle_running(
        self,
        running: ComposerOperationRunning,
        lease: SessionOperationLease | None,
        observation: ComposerTurnObservation,
        end: _TurnEnd,
        failure: ComposerOperationError | None,
        *,
        request_id: str | None,
    ) -> _JobOutcome:
        if failure is None:
            # The turn committed its own completed terminal.
            return _JobOutcome.SETTLED
        claim = running.claim
        attempt = 0
        while True:
            try:
                current = await run_sync_in_worker(self._authority.get, session_id=claim.session_id, operation_id=claim.operation_id)
                if current is None:
                    return _JobOutcome.ABANDONED
                if current.status != "running":
                    if attempt == 0:
                        # m2: the turn raised after a terminal had already committed
                        # (its own success, or a racer's). That terminal is never
                        # relabelled, but the discarded exception is recorded.
                        slog.warning(
                            "composer_operation.post_terminal_exception",
                            operation_id=claim.operation_id,
                            status=current.status,
                            failure_code=failure.failure_code,
                            exc_class=end.exc_class,
                        )
                    # attempt > 0: an earlier attempt committed before its error surfaced.
                    return _JobOutcome.SETTLED
                await self._write_failure(running, observation, failure, request_id=current.request_id)
                return _JobOutcome.SETTLED
            except _TERMINAL_WRITE_TRANSIENT_ERRORS as exc:
                # F-B4: today's synchronous route needs no DB write to deliver its
                # 4xx/5xx, so a transient fault here must not relabel it as the
                # reaper's 503 worker_lost. Retry while the adopted SOL is live.
                attempt += 1
                if attempt >= _TERMINAL_WRITE_MAX_ATTEMPTS or lease is None or lease.renewal_error is not None:
                    slog.error(
                        "composer_operation.terminal_write_failed",
                        operation_id=claim.operation_id,
                        failure_code=failure.failure_code,
                        exc_class=type(exc).__name__,
                        attempts=attempt,
                    )
                    raise
                slog.warning(
                    "composer_operation.terminal_write_retrying",
                    operation_id=claim.operation_id,
                    failure_code=failure.failure_code,
                    exc_class=type(exc).__name__,
                    attempt=attempt,
                )
                await asyncio.sleep(min(_TERMINAL_WRITE_BASE_BACKOFF_SECONDS * 2 ** (attempt - 1), _TERMINAL_WRITE_MAX_BACKOFF_SECONDS))
            except (ComposerOperationFenceLost, SessionOperationFenceLost) as exc:
                slog.warning(
                    "composer_operation.terminal_write_fenced",
                    operation_id=claim.operation_id,
                    failure_code=failure.failure_code,
                    exc_class=type(exc).__name__,
                )
                # F-M3: the bound fence lapsed while this instance lives. The
                # authority re-decides that under the session lock.
                return await self._settle_own_lapsed(
                    session_id=claim.session_id,
                    operation_id=claim.operation_id,
                    failure=failure,
                    request_id=request_id,
                )

    async def _write_failure(
        self,
        running: ComposerOperationRunning,
        observation: ComposerTurnObservation,
        failure: ComposerOperationError,
        *,
        request_id: str | None,
    ) -> None:
        service = self._services.session_service
        try:
            await service.fail_composer_async_operation(running, failure=failure)
        except ComposerOperationCancelledDuringTurn:
            # Task 6 contract: a committed cancel outranks every non-cancel
            # failure (spec §4), so the terminal becomes request_cancelled, with
            # the turn's cohort joined first (F-B2/C2; a no-op once durable).
            await persist_cancelled_turn_audit(self._services, running, observation)
            await service.fail_composer_async_operation(running, failure=request_cancelled_error(request_id=request_id))

    async def _settle_own_lapsed(
        self,
        *,
        session_id: UUID,
        operation_id: str,
        failure: ComposerOperationError,
        request_id: str | None,
    ) -> _JobOutcome:
        """F-M3: settle this instance's own ``running`` job whose bound fence lapsed while the instance lives.

        PostgreSQL refuses to take over a fence whose owner instance is alive
        (repository.py:5780-5790), and the owner's fenced write is refused
        too. Without this settle the row stays ``running`` and D8 answers
        every new send with 409 until a restart. The authority reads the
        fence under the session lock and never advances, renews, releases or
        takes it over.
        """
        # A committed cancel substitutes request_cancelled inside the authority,
        # which refuses request_cancelled as the caller's own failure.
        own_failure = failure if failure.failure_code != "request_cancelled" else worker_lost_error(request_id=request_id)
        try:
            settled = await run_sync_in_worker(
                self._authority.settle_own_lapsed,
                session_id=session_id,
                operation_id=operation_id,
                owner_instance_id=self._owner_instance_id,
                failure=own_failure,
                cancelled_failure=request_cancelled_error,
            )
        except ComposerOperationFenceLost:
            # The fence is live, was released or superseded (a successor settles
            # through settle_lost), or the row already settled.
            return _JobOutcome.ABANDONED
        slog.warning(
            "composer_operation.settled_own_lapsed",
            operation_id=operation_id,
            failure_code=settled.failure_code,
            audit_may_be_incomplete=True,
        )
        return _JobOutcome.SETTLED

    async def _settle_unadopted(self, job: _ComposerJob, context: SessionOperationContext, exc: Exception) -> _JobOutcome:
        """F-B4 for an adopt defect: the composite already committed ``running`` and no turn started.

        ``SessionOperationLease.adopt`` releases the minted context before it
        re-raises (``_raise_adopt_failure_after_release``, lifecycle.py:208-215).
        So the fenced fail write under that context lands only when that
        release did not complete. Otherwise it is refused, ``settle_own_lapsed``
        refuses a released fence, and the row is left to the reaper. The
        reaper's only settle for this row is 503 ``worker_lost``, the accepted
        F-B4 residual recorded in Review notes.
        """
        claim = job.claim
        record = await run_sync_in_worker(self._authority.get, session_id=claim.session_id, operation_id=claim.operation_id)
        if record is None or record.status != "running":
            return _JobOutcome.ABANDONED
        failure = project_composer_operation_error(
            exc,
            request_id=record.request_id,
            session_engine_dialect=self._services.session_engine.dialect.name,
        )
        slog.error(
            "composer_operation.adopt_failed",
            operation_id=claim.operation_id,
            exc_class=type(exc).__name__,
            failure_code=failure.failure_code,
            diagnostic_id=failure.diagnostic_id,
        )
        running = ComposerOperationRunning(claim=claim, session_operation_context=context)
        end = _TurnEnd(failure=failure, cancelled=False, exc_class=type(exc).__name__)
        outcome = await self._settle_running(running, None, ComposerTurnObservation(), end, failure, request_id=record.request_id)
        await self._release_minted_context(context)
        return outcome

    async def _release_minted_context(self, context: SessionOperationContext) -> bool:
        """Release a context no lease adopted, so the session is not blocked until its fence expires."""
        try:
            await run_sync_in_worker(self._services.session_service.session_operation_authority.release, context)
        except SessionOperationFenceLost:
            # adopt's own cleanup already released it (the normal case), or it lapsed.
            return False
        return True

    async def _close_lease(self, lease: SessionOperationLease, *, operation_id: str, settled: bool) -> BaseException | None:
        try:
            await lease.close()
        except AuditIntegrityError:
            raise
        except Exception as exc:
            # D12: a close or renewal failure after the terminal never relabels it.
            if settled:
                slog.error("composer_operation.lease_close_failed_after_settle", operation_id=operation_id, exc_class=type(exc).__name__)
            else:
                slog.error("composer_operation.lease_close_failed", operation_id=operation_id, exc_class=type(exc).__name__)
            return exc
        return None

    # ── watcher ─────────────────────────────────────────────────────────

    async def _watch(self, job: _ComposerJob, lease: SessionOperationLease) -> None:
        """Bridge SOL loss, a local Stop and a remote cancel marker to the turn task."""
        operation_id = job.claim.operation_id
        lost = asyncio.create_task(lease.wait_until_lost(), name=f"composer-async-lease-loss-{operation_id}")
        local = asyncio.create_task(job.local_cancel.wait(), name=f"composer-async-local-cancel-{operation_id}")
        consecutive_failures = 0
        try:
            while True:
                poll_seconds = min(
                    _WATCHER_POLL_SECONDS * (2 ** min(consecutive_failures, 16)),
                    _WATCHER_MAX_BACKOFF_SECONDS,
                )
                done, _pending = await asyncio.wait({lost, local}, timeout=poll_seconds, return_when=asyncio.FIRST_COMPLETED)
                if lost in done:
                    job.deliver(_COMPOSER_OPERATION_LEASE_LOST)
                    return
                if local in done:
                    job.deliver(_COMPOSER_OPERATION_CANCEL_REQUESTED)
                    return
                poll = await self._poll_marker(job.claim, consecutive_failures=consecutive_failures)
                if poll is _WatcherPoll.RETRY:
                    consecutive_failures += 1
                    continue
                consecutive_failures = 0
                if poll is _WatcherPoll.CANCEL_REQUESTED:
                    job.deliver(_COMPOSER_OPERATION_CANCEL_REQUESTED)
                    return
                if poll is _WatcherPoll.GONE:
                    job.deliver(_COMPOSER_OPERATION_LEASE_LOST)
                    return
        finally:
            lost.cancel()
            local.cancel()
            await asyncio.wait({lost, local})
            for waiter in (lost, local):
                if not waiter.cancelled():
                    # Re-raises a waiter defect (a loss event without a recorded error).
                    waiter.result()

    async def _poll_marker(self, claim: ComposerOperationClaim, *, consecutive_failures: int) -> _WatcherPoll:
        try:
            record = await run_sync_in_worker(self._authority.get, session_id=claim.session_id, operation_id=claim.operation_id)
        except _WATCHER_TRANSIENT_ERRORS as exc:
            _log_watcher_retry(exc, operation_id=claim.operation_id, consecutive_failures=consecutive_failures + 1)
            return _WatcherPoll.RETRY
        except DBAPIError as exc:
            # Transient only when disconnect detection invalidated the connection.
            if not exc.connection_invalidated:
                slog.error("composer_operation.watcher_poll_failed", operation_id=claim.operation_id, exc_class=type(exc).__name__)
                raise
            _log_watcher_retry(exc, operation_id=claim.operation_id, consecutive_failures=consecutive_failures + 1)
            return _WatcherPoll.RETRY
        except SQLAlchemyError as exc:
            slog.error("composer_operation.watcher_poll_failed", operation_id=claim.operation_id, exc_class=type(exc).__name__)
            raise
        if record is None:
            return _WatcherPoll.GONE
        if record.cancel_requested_at is not None:
            return _WatcherPoll.CANCEL_REQUESTED
        return _WatcherPoll.CLEAR

    def _on_watcher_done(self, job: _ComposerJob, completed: asyncio.Task[None]) -> None:
        if completed.cancelled():
            return
        error = completed.exception()
        if error is None:
            return
        slog.error("composer_operation.watcher_failed", operation_id=job.claim.operation_id, exc_class=type(error).__name__)
        # Without its watcher, a remote cancel or lease loss can no longer
        # reach the turn: stop it as a server fault rather than run blind.
        job.deliver(_COMPOSER_OPERATION_LEASE_LOST)

    # ── reaper ──────────────────────────────────────────────────────────

    async def _reap_queued(self, record: ComposerOperationRecord) -> _ReapOutcome:
        """Settle an unclaimed queued row whose deadline passed or whose cancel committed (Task 4 D-4c)."""
        failure = (
            request_cancelled_error(request_id=record.request_id)
            if record.cancel_requested_at is not None
            else deadline_expired_error(
                request_id=record.request_id,
                timeout_seconds=self._services.settings.composer_timeout_seconds,
            )
        )
        try:
            await run_sync_in_worker(
                self._authority.settle_unstarted,
                None,
                session_id=record.session_id,
                operation_id=record.operation_id,
                failure=failure,
            )
        except ComposerOperationFenceLost:
            # A peer claimed it between discovery and the locked write; that claim owns the row now.
            return _ReapOutcome.OWNER_ALIVE
        return _ReapOutcome.SETTLED

    async def _reap_running(self, row: ComposerOperationRecord) -> _ReapOutcome:
        service = self._services.session_service
        try:
            lease = await SessionOperationLease.acquire(
                service.session_operation_authority,
                session_id=row.session_id,
                operation_kind=SessionOperationKind.COMPOSE,
                owner_instance_id=self._owner_instance_id,
                lease_seconds=service.session_operation_lease_seconds,
            )
        except SessionOperationConflictError:
            if row.claim_owner_instance_id != self._owner_instance_id:
                return _ReapOutcome.OWNER_ALIVE
            # F-M3: this instance's own job, which it no longer runs (reap_once
            # skips live jobs), whose fence lapsed. PostgreSQL refuses the
            # takeover while this instance lives, so only the owner can settle it.
            own = await self._settle_own_lapsed(
                session_id=row.session_id,
                operation_id=row.operation_id,
                failure=worker_lost_error(request_id=row.request_id),
                request_id=row.request_id,
            )
            return _ReapOutcome.SETTLED if own is _JobOutcome.SETTLED else _ReapOutcome.OWNER_ALIVE
        except SessionOperationFenceLost as exc:
            # Discovery is advisory. Anything but these two reasons is integrity.
            if exc.reason is FenceLossReason.OWNER_INACTIVE:
                # B1: an archived session. acquire raises OWNER_INACTIVE here
                # forever, so settle_lost can never run. The archive itself proves
                # the owner is finished (it took the fence), so the authority
                # settles under the session lock without one.
                return await self._reap_inactive_session(row)
            if exc.reason is FenceLossReason.MISSING:
                # The session was hard-deleted between discovery and acquire
                # (repository.py:4789-4790); the cascade removed the row. Never
                # let this kill the reap loop, which would drain the instance.
                return _ReapOutcome.ALREADY_TERMINAL
            raise
        try:
            current = await run_sync_in_worker(self._authority.get, session_id=row.session_id, operation_id=row.operation_id)
            if current is None or current.status != "running":
                return _ReapOutcome.ALREADY_TERMINAL
            try:
                # ``failure`` is always the worker_lost envelope (Task 4 requires
                # it); the authority itself substitutes the request_cancelled
                # envelope, built from the row's own request id under the lock,
                # when a cancel marker committed first (Task 4 D-4b).
                settled = await run_sync_in_worker(
                    self._authority.settle_lost,
                    session_operation_context=lease.context,
                    session_id=row.session_id,
                    operation_id=row.operation_id,
                    failure=worker_lost_error(request_id=current.request_id),
                    cancelled_failure=request_cancelled_error,
                )
            except ComposerOperationFenceLost:
                return _ReapOutcome.ALREADY_TERMINAL
            # The dead owner's final audit may be unproducible: say so, never claim it never ran.
            slog.warning(
                "composer_operation.reaped_running",
                operation_id=row.operation_id,
                failure_code=settled.failure_code,
                audit_may_be_incomplete=True,
            )
            return _ReapOutcome.SETTLED
        finally:
            await lease.close()

    async def _reap_inactive_session(self, row: ComposerOperationRecord) -> _ReapOutcome:
        """B1: settle an archived session's ``running`` row; its D3 capacity slot is returned."""
        try:
            settled = await run_sync_in_worker(
                self._authority.settle_lost_inactive_session,
                session_id=row.session_id,
                operation_id=row.operation_id,
                failure=worker_lost_error(request_id=row.request_id),
                cancelled_failure=request_cancelled_error,
            )
        except ComposerOperationFenceLost:
            # Already terminal, or the session is no longer archived; re-read on the next sweep.
            return _ReapOutcome.ALREADY_TERMINAL
        slog.warning(
            "composer_operation.reaped_inactive_session",
            operation_id=row.operation_id,
            failure_code=settled.failure_code,
            audit_may_be_incomplete=True,
        )
        return _ReapOutcome.SETTLED
```
