    """Request body for POST /api/sessions/{id}/messages (asynchronous freeform operation).

    Strict, with the client-minted ``operation_id`` required: the route admits

    # max_length=65536 (64 KiB) caps message content at the schema boundary.
    # Phase 5b.0.5 (F-3): defense against unbounded payload allocation
    # before interpretation-event code paths can be exercised.  64 KiB
    # accommodates multi-paragraph user messages and long paste content
    # while preventing trivial-cost large-string attacks.  Mirrors the
    # _InlineBlobModel.content 256 KiB cap (web/composer/redaction.py).
    content: str = pydantic.Field(min_length=1, max_length=65536)
    state_id: UUID | None = None

    @field_validator("content")
    @classmethod
    def _validate_content(cls, value: str) -> str:
        return _require_visible_content(value, field_label="Message content")

    @field_validator("state_id", mode="before")
    @classmethod
    def _parse_state_id(cls, value: object) -> UUID | None:
        # Canonical-UUID parser of RevertStateRequest, with None admitted.
        if value is None:
            return None
        if type(value) is str:
            try:
                parsed = UUID(value)
            except ValueError as exc:
                raise ValueError("state_id must be a canonical UUID") from exc
            if str(parsed) != value:
                raise ValueError("state_id must be a canonical UUID")
            return parsed
        if type(value) is UUID:
            return value
        raise ValueError("state_id must be a canonical UUID")

"""The app-scoped dependencies one detached composer turn reads (plan Task 10).

A worker-run turn has no ``Request``. Everything the freeform send/recompose
bodies and the shared pipeline-settlement coordinator used to read from
``request.app.state`` is projected here once, into an immutable bundle.
:func:`composer_app_services` is the only reader of ``app.state`` for the turn.

It performs no type checks. ``app.state`` is ELSPETH's own Tier-1 wiring, set
by ``create_app`` (and by the hand-built route test apps). A wrong object
crashes at first use, and nothing here papers over it.
"""

from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass
from functools import partial
from typing import TYPE_CHECKING, final

from elspeth.web.auth.models import UserIdentity

if TYPE_CHECKING:
    from sqlalchemy import Engine
    from starlette.applications import Starlette

    from elspeth.contracts.secrets import WebSecretResolver
    from elspeth.web.catalog.protocol import CatalogService
    from elspeth.web.composer.progress import ComposerProgressRegistry
    from elspeth.web.composer.protocol import ComposerService
    from elspeth.web.config import WebSettings
    from elspeth.web.coordination.composer_progress_authority import DatabaseComposerProgressRegistry
    from elspeth.web.plugin_policy.models import PluginAvailabilitySnapshot
    from elspeth.web.plugin_policy.profiles import OperatorProfileRegistry
    from elspeth.web.sessions.routes._helpers import _SessionComposeLockRegistry
    from elspeth.web.sessions.service import SessionServiceImpl


@final
@dataclass(frozen=True, slots=True)
class ComposerAppServices:
    """Immutable projection of the app wiring a composer turn needs."""

    session_service: SessionServiceImpl
    composer_service: ComposerService
    settings: WebSettings
    catalog_service: CatalogService
    operator_profile_registry: OperatorProfileRegistry
    scoped_secret_resolver: WebSecretResolver | None
    session_engine: Engine
    plugin_snapshot_for_user_id: Callable[[str], PluginAvailabilitySnapshot]
    progress_registry: ComposerProgressRegistry | DatabaseComposerProgressRegistry
    compose_locks: _SessionComposeLockRegistry


def _plugin_snapshot_for_actor(
    factory: Callable[[UserIdentity], PluginAvailabilitySnapshot],
    user_id: str,
) -> PluginAvailabilitySnapshot:
    """Resolve the principal snapshot for an admitted actor id.

    ``app.state.plugin_snapshot_factory`` takes the authenticated principal.
    Its production implementation reads only ``user.user_id``
    (``RequestPluginSnapshotFactory.__call__`` → ``for_user_id``,
    ``plugin_policy/availability.py:263-264``). A detached turn holds only the
    admitted ``actor_user_id`` (D6: no username is persisted on the job), so the
    principal is rebuilt from it. ``username`` is not an input to snapshot
    policy.
    """
    return factory(UserIdentity(user_id=user_id, username=user_id))


def composer_app_services(app: Starlette) -> ComposerAppServices:
    """Build the turn's dependency bundle from the app wiring."""
    # Function-local: routes/composer/pipeline_settlement.py imports this
    # module while the routes package initialises, so a module-level import of
    # routes/_helpers would be circular.
    from elspeth.web.sessions.routes._helpers import composer_session_lock_registry

    state = app.state
    return ComposerAppServices(
        session_service=state.session_service,
        composer_service=state.composer_service,
        settings=state.settings,
        catalog_service=state.catalog_service,
        operator_profile_registry=state.operator_profile_registry,
        scoped_secret_resolver=state.scoped_secret_resolver,
        session_engine=state.session_engine,
        plugin_snapshot_for_user_id=partial(_plugin_snapshot_for_actor, state.plugin_snapshot_factory),
        progress_registry=state.composer_progress_registry,
        compose_locks=composer_session_lock_registry(app),
    )

async def _running_job(
    app: FastAPI,
    session_id: uuid.UUID,
    *,
    kind: ComposerOperationKind,
    content: str | None = None,
) -> _Job:
    """Hand-build one running job exactly as the Task 11 worker will."""
    services = composer_app_services(app)
    service = services.session_service
    operation_id = str(uuid.uuid4())
    request: SendMessageRequest | RecomposeRequest
    if kind == "compose_message":
        assert content is not None
        request = SendMessageRequest(operation_id=operation_id, content=content)
    else:
        request = RecomposeRequest(operation_id=operation_id)
    authority = ComposerAsyncOperationAuthority(
        services.session_engine,
        owner_instance_id=service.session_operation_owner_instance_id,
        claim_lease_seconds=30,
    )
    await run_sync_in_worker(
        authority.admit,
        session_id=session_id,
        operation_id=operation_id,
        kind=kind,
        request_hash=hashlib.sha256(f"t10:{operation_id}".encode()).hexdigest(),
        actor_user_id="alice",
        request_id=None,
        request_json=request.model_dump_json(),
        deadline_seconds=85.0,
        max_nonterminal=64,
    )
    (claim,) = await run_sync_in_worker(authority.claim_next, limit=1)
    context = await run_sync_in_worker(
        service.session_operation_authority.start_composer_async_operation,
        claim,
        owner_instance_id=service.session_operation_owner_instance_id,
        lease_seconds=service.session_operation_lease_seconds,
    )
    # F-C5: the worker pins the budget when the start composite returns; 85.0
    # stands in for deadline_at - started_at on the database clock.
    budget_anchor = composer_turn.ComposerBudgetAnchor(remaining_at_running_seconds=85.0, monotonic_at_running=time.monotonic())
    lease = await SessionOperationLease.adopt(
        service.session_operation_authority,
        context,
        lease_seconds=service.session_operation_lease_seconds,
    )
    request_lease = await services.progress_registry.start_request(str(session_id), "alice")
    turn = composer_turn.ComposerTurnInput(
        session_id=session_id,
        operation_id=operation_id,
        kind=kind,
        actor_user_id="alice",
        request=request,
        request_id=None,
        budget_seconds=85.0,
    )
    return _Job(
        turn=turn,
        lease=lease,
        running=ComposerOperationRunning(claim=claim, session_operation_context=context),
        request_lease=request_lease,
        authority=authority,

class _CancelCommittingComposer:
    """``compose()`` commits the job's cancel exactly as the cancel route would, then ends as told.

    No worker exists in Task 10, so nothing delivers a cancel marker: the
    committed ``cancel_requested_at`` is what the rest of the turn meets
    (Task 6's predicate on non-audit writes, and the terminal CAS).
    ``raise_after`` is any exception ``compose()`` can end with: a
    ``CancelledError`` carrying a sidecar, or a typed-ladder error such as
    ``ComposerConvergenceError`` (B2-closure).
    """

    def __init__(
        self,
        job_holder: list[_Job],
        *,
        result: ComposerResult | None = None,
        raise_after: BaseException | None = None,
    ) -> None:
        self._job_holder = job_holder
        self._result = result
        self._raise_after = raise_after

    async def compose(self, *args: Any, **kwargs: Any) -> ComposerResult:
        del args, kwargs
        job = self._job_holder[0]
        cancelled = await run_sync_in_worker(
            job.authority.request_cancel,
            session_id=job.turn.session_id,
            operation_id=job.turn.operation_id,

        )
        assert cancelled is not None
        assert cancelled.status == "running"
        assert cancelled.cancel_requested_at is not None
        if self._raise_after is not None:
            raise self._raise_after
        assert self._result is not None
        return self._result

    connection: Connection,
    running: ComposerOperationRunning,
    *,
    user_message_id: UUID,

        update(table)

                self._session_write_lock(conn, sid),

    that composite returned. :func:`run_composer_turn` measures the compose
    budget from it immediately before ``compose()``. Setup after ``running``
    (adopt, request lease, preamble reads, the user-row insert) is therefore
    spent from the admitted deadline, never added to it.
    """

    remaining_at_running_seconds: float
    monotonic_at_running: float

    def __post_init__(self) -> None:
        if type(self.remaining_at_running_seconds) is not float or not math.isfinite(self.remaining_at_running_seconds):
            raise ValueError("ComposerBudgetAnchor.remaining_at_running_seconds must be a finite float")
        if type(self.monotonic_at_running) is not float or not math.isfinite(self.monotonic_at_running):
            raise ValueError("ComposerBudgetAnchor.monotonic_at_running must be a finite float")

    def remaining_seconds(self, *, monotonic_now: float) -> float:
        """The budget left at ``monotonic_now``; zero or negative once the deadline has passed."""
        return self.remaining_at_running_seconds - (monotonic_now - self.monotonic_at_running)


@final
class ComposerTurnDeadlineExpired(RuntimeError):
    """The operation's deadline passed before ``compose()`` could start (review-pass F-C5).

    Raised by :func:`run_composer_turn` before any provider call. The worker
    (Task 11) settles it ``deadline_expired`` (Task 9 ``deadline_expired_error``)
    through ``fail_composer_async_operation``; it is never a generic
    ``operation_failed``.
    """

    def __init__(
        self,
        *,
        session_id: UUID,
        operation_id: str,
        remaining_seconds: float,
        budget_seconds_at_running: float,
    ) -> None:
        super().__init__(
            f"composer operation {operation_id} reached its deadline before compose "
            f"(remaining {remaining_seconds:.3f}s of {budget_seconds_at_running:.3f}s at running)"

        self.session_id = session_id
        self.operation_id = operation_id
        self.remaining_seconds = remaining_seconds
        self.budget_seconds_at_running = budget_seconds_at_running


@final
@dataclass(slots=True)
class ComposerTurnObservation:
    """What one running turn has produced so far, readable by the worker's job frame (review-pass F-B2/C2).

    Mutable by design. The worker creates one per job, passes it to
    :func:`run_composer_turn` (which writes it at fixed points), and reads it
    after the turn ends by any route:

    - ``compose_base_state_id``: the head the turn composes against, set just
      before ``compose()``;
    - ``compose_result``: set on the first statement after ``compose()``
      returns, before any post-compose write. A cancel that lands in the tail
      (settlement, state save, the auto-title join, or a terminal CAS that loses
      to a committed cancel) therefore still leaves the provider evidence to
      :func:`persist_cancelled_turn_audit`;
    - ``pending_exception_llm_calls``: the LLM calls a typed-ladder arm holds
      on its exception (``_llm_calls_from_exception``), set as the arm's first
      statement, before its first session write (B2-closure). A committed
      cancel can fence that arm's first non-audit write (a handler's
      partial-state save, the planner disposition row) before the arm's own
      cohort write runs. ``compose()`` raised, so there is no
      ``compose_result`` to fall back on;
    - ``audit_cohort_durable``: set on the line after each cohort persist the
      turn reaches returns: the success composite, the three ``_handle_*``
      recovery handlers (whose last write is ``_persist_turn_audit_cohort``),
      each ``_persist_llm_calls`` in the ladder, and
      :func:`persist_cancelled_turn_audit`. The ladder's persists are
      fail-soft (``plugin_crash_pending=True``): a ``SQLAlchemyError`` is
      counted and logged inside the helper, which returns normally. On that
      path the flag means "handed to the fail-soft persist", and the loss is
      not retried (today's route disposition). A write that raises past the
      helper never reaches the flag. This is what makes the frame's persist
      idempotent.
    """

    compose_base_state_id: UUID | None = None
    compose_result: ComposerResult | None = None
    pending_exception_llm_calls: tuple[ComposerLLMCall, ...] = ()
    audit_cohort_durable: bool = False

def _composer_operation_cancel_of(exc: asyncio.CancelledError) -> _ComposerOperationCancel | None:
    """Return the worker's cancel marker when ``exc`` was delivered with one."""
    if len(exc.args) != 1:
        return None
    delivered_id = id(exc.args[0])
    if delivered_id not in _COMPOSER_OPERATION_CANCELS_BY_ID:
        return None
    return _COMPOSER_OPERATION_CANCELS_BY_ID[delivered_id]


def _advisory_progress_sink(sink: ComposerProgressSink, *, session_id: UUID, operation_id: str) -> ComposerProgressSink:
    """D15: a detached turn's progress snapshots are advisory UX, never its outcome.

    The operation row is the settlement authority (spec §2). Loss of the
    request lease, a revoked identity or session (``PermissionError`` family),
    or a database fault on a publish is logged and the turn continues. The
    heartbeat still cancels the turn if the request lease is truly gone. Any
    other exception is a defect and propagates.
    """

    async def publish(event: ComposerProgressEvent) -> None:
        try:
            await sink(event)
        except (ComposerRequestLeaseLost, PermissionError, SQLAlchemyError, AsyncWorkerAdmissionTimeoutError) as exc:
            slog.warning(
                "composer_operation.progress_publish_failed",
                session_id=str(session_id),
                operation_id=operation_id,
                phase=event.phase,
                exc_class=type(exc).__name__,
            )

    return publish

_AUTO_TITLE_JOIN_SECONDS: Final = 2.0


async def _join_auto_title(task: asyncio.Task[None], *, session_id: UUID, operation_id: str) -> None:
    """Bounded, cancellation-safe join of the first-message auto-title child (Appendix A T20; review m3).

    The child is lease-owned (``lease.create_task``) and writes the session
    title under the COMPOSE fence. It is finished before this returns or
    raises: either done, or cancelled and given a bounded settle. No title
    write therefore outlives the turn and lands after the worker publishes
    the terminal (spec §1).

    - Done with a result: ``task.result()`` re-raises a title failure into the
      in-flight turn, where it lands in the turn's own class (T20), as
      ``messages.py:1173-1178`` does today.
    - Done cancelled: logged (``composer_operation.auto_title_cancelled``),
      never relabelled.
    - Not done within ``_AUTO_TITLE_JOIN_SECONDS``: cancelled, settled and
      logged (``composer_operation.auto_title_join_timed_out``), never
      relabelled.
    - The turn itself cancelled while joining (a worker marker in the tail):
      the child is cancelled and settled before the ``CancelledError``
      propagates, and that is logged
      (``composer_operation.auto_title_join_cancelled``).
    """
    try:
        done, _pending = await asyncio.wait({task}, timeout=_AUTO_TITLE_JOIN_SECONDS)
    except asyncio.CancelledError:
        task.cancel()
        await asyncio.wait({task}, timeout=_AUTO_TITLE_JOIN_SECONDS)
        slog.warning(
            "composer_operation.auto_title_join_cancelled",
            session_id=str(session_id),
            operation_id=operation_id,
            child_settled=task.done(),
        )
        raise
    if task not in done:
        task.cancel()
        await asyncio.wait({task}, timeout=_AUTO_TITLE_JOIN_SECONDS)
        slog.warning(
            "composer_operation.auto_title_join_timed_out",
            session_id=str(session_id),
            operation_id=operation_id,
            timeout_seconds=_AUTO_TITLE_JOIN_SECONDS,
            child_settled=task.done(),
        )
        return
    if task.cancelled():
        slog.warning(
            "composer_operation.auto_title_cancelled",
            session_id=str(session_id),
            operation_id=operation_id,
        )
        return
    task.result()


async def persist_cancelled_turn_audit(
    services: ComposerAppServices,
    running: ComposerOperationRunning,
    observation: ComposerTurnObservation,
) -> None:
    """Persist a cancelled turn's provider evidence before the worker publishes the cancel terminal (F-B2/C2).

    Spec §3 (l.192-194) and §4 (l.236-237): a cancel that lands after
    ``compose()`` returned must not discard the provider calls the turn already
    made and paid for. That covers a worker marker in the post-compose tail,
    and a committed cancel that fences a tail write or wins the terminal CAS.
    Nothing of the composite survived: no assistant row, no cohort. So the
    result's cohort is written here on its own, through the audit-only path
    (``add_messages_atomic(..., audit_only=True)``), which a committed cancel
    does not fence:

    - LLM-call sidecars carry ``compose_base_state_id``, as the success tail's do.
    - Tool invocations are written with ``parent_assistant_id=None``, that is as
      ``role="audit"`` rows. The exception is when the compose loop already
      persisted its tool-call turn: those tool rows are skipped. The assistant
      row they would have parented never became durable, and a ``tool`` row
      must name one.

    What it writes is a precedence, never a union (B2-closure):

    - a compose result exists: that result's LLM calls and tool rows;
    - no compose result: ``observation.pending_exception_llm_calls``, the calls
      a typed-ladder arm recorded before its first session write. Those are
      written with ``compose_base_state_id``, as the arm's own persist would
      have written them. Exception-carried tool invocations are not recorded
      (see Review notes).

    Runtime-preflight path 2 has both sources, holding the same calls; the
    result wins, so they are never written twice.

    A no-op when the cohort is already durable, or when there is neither a
    result nor a pending call. Already durable means the success composite, a
    ladder persist, or this function already ran. Neither source means the
    cancel landed inside ``compose()`` and the turn's cancelled arm already
    joined the exception-attached sidecar, or the failure carried no provider
    calls. The write is joined through repeated cancellation of the caller.

    Unwind disposition is the same as
    ``_persist_llm_calls(plugin_crash_pending=True)``. The following are
    counted and logged and do not raise:

    - a database failure;
    - a worker-pool admission timeout;
    - a lapsed COMPOSE fence (``SessionOperationFenceLost``, raised by
      ``_require_session_operation_context_on_connection``, ``service.py:4949``
      and ``:4965``). ``audit_only`` waives only the cancel predicate, never
      the fence.
 The worker's own terminal write still decides the outcome. A
    lapsed fence leaves the row to the reaper, whose ``reaped_running`` event
    carries ``audit_may_be_incomplete=True``.
    """
    if observation.audit_cohort_durable:
        return
    result = observation.compose_result
    tool_invocations: tuple[ComposerToolInvocation, ...]
    llm_calls: tuple[ComposerLLMCall, ...]
    if result is not None:
        tool_invocations = result.tool_invocations if not result.persisted_tool_call_turn else ()
        llm_calls = result.llm_calls
    elif observation.pending_exception_llm_calls:
        tool_invocations = ()
        llm_calls = observation.pending_exception_llm_calls
    else:
        return
    claim = running.claim
    drafts, _pipeline_bindings = _turn_audit_cohort_drafts(
        tool_invocations,
        llm_calls,
        tool_composition_state_id=observation.compose_base_state_id,
        llm_composition_state_id=observation.compose_base_state_id,
        parent_assistant_id=None,
    )
    if not drafts:
        observation.audit_cohort_durable = True
        return

    async def _persist() -> None:
        try:
            await services.session_service.add_messages_atomic(
                claim.session_id,
                drafts,
                composition_state_id=None,
                writer_principal="compose_loop",
                session_operation_context=running.session_operation_context,
                audit_only=True,
            )
        except (SQLAlchemyError, SessionOperationFenceLost, AsyncWorkerAdmissionTimeoutError) as exc:
            _COMPOSER_PERSIST_FAILED_DURING_UNWIND_COUNTER.add(len(drafts), {"helper": "cancelled_turn_audit"})
            slog.error(
                "composer_operation.cancelled_turn_audit_persist_failed",
                session_id=str(claim.session_id),
                operation_id=claim.operation_id,
                rows=len(drafts),
                calls=len(llm_calls),
                models_requested=[call.model_requested for call in llm_calls],
                exc_class=type(exc).__name__,
            )
            return
        observation.audit_cohort_durable = True

    await _join_shielded_task_after_cancellation(
        asyncio.create_task(_persist(), name="composer-operation-cancelled-turn-audit-persist")
    )


async def run_composer_turn(
    services: ComposerAppServices,
    turn: ComposerTurnInput,
    *,
    lease: SessionOperationLease,
    running: ComposerOperationRunning,
    request_lease: ComposerRequestLease,
    observation: ComposerTurnObservation,
    budget_anchor: ComposerBudgetAnchor,
) -> ComposerOperationRecord:
    """Run one freeform send or recompose turn to its settled success; see the module docstring."""
    if running.claim.session_id != turn.session_id or running.claim.operation_id != turn.operation_id:
        raise AuditIntegrityError("composer turn input does not match its running operation")
    if lease.context != running.session_operation_context:
        raise AuditIntegrityError("composer turn lease is not the running operation's session operation context")
    # F-C5: the worker fills both from the one start composite; a mismatch is a caller defect.
    if turn.budget_seconds != budget_anchor.remaining_at_running_seconds:
        raise ValueError("composer turn budget_seconds is not its budget anchor's remaining_at_running_seconds")
    if (
        observation.compose_result is not None
        or observation.compose_base_state_id is not None
        or observation.pending_exception_llm_calls
        or observation.audit_cohort_durable
    ):
        raise ValueError("composer turn observation must be fresh for each turn")
    labels = _SEND_LABELS if turn.kind == "compose_message" else _RECOMPOSE_LABELS
    request = turn.request
    service = services.session_service
    settings = services.settings

    # The turn needs the session record (id, title). Ownership is NOT re-checked
    # here: the worker's _require_actor_owns_session (Task 11) re-proves it
    # after adopt and before the composer request lifecycle opens, and is the
    # one surviving re-check (T00 Appendix A P15, review M7). A second copy
    # here would run after the CRL opened. On PostgreSQL the registry's own
    # ownership query would answer first, with a different code. A row gone
    # since that re-check still reads as the same non-disclosing 404
    # (routes/_helpers.py:2796-2803).
    try:
        session = await service.get_session(turn.session_id)
    except ValueError:
        raise HTTPException(status_code=404, detail="Session not found") from None

    # Common preamble (messages.py:157-227 / compose.py:124-135, 179-182).
    state_record = await service.get_current_state(session.id)
    if state_record is None:
        state = _initial_composition_state()
        pre_send_state_id: UUID | None = None
    else:
        state = _state_from_record(state_record)
        pre_send_state_id = state_record.id
        if type(request) is SendMessageRequest and request.state_id is not None:
            # Two byte-identical 404s by design (IDOR; messages.py:169-182).
            client_state_id = request.state_id
            try:
                client_state = await service.get_state(client_state_id)
            except ValueError:
                raise HTTPException(status_code=404, detail="State not found") from None
            if client_state.session_id != session.id:
                raise HTTPException(status_code=404, detail="State not found")
            pre_send_state_id = client_state_id
    # One compose base for both kinds: the ACTUAL head under the lease
    # (messages.py:199-211). Send's client-asserted id is provenance only.
    compose_base_state_id = state_record.id if state_record is not None else None
    # Tier 1, outside every try: a corrupt envelope must propagate (1e1a4832d).
    prior_completion_gates_facts = parse_completion_gates(state_record.composer_meta) if state_record is not None else None
    plugin_snapshot = services.plugin_snapshot_for_user_id(turn.actor_user_id)
    profile_registry = services.operator_profile_registry
    _guided = state.guided_session
    _guided_terminal_for_compose = (
        _guided.terminal if (_guided is not None and _guided.terminal is not None and not _guided.transition_consumed) else None
    )

    # Per-kind preamble.
    if type(request) is SendMessageRequest:
        message_text = request.content
        # The originating user row, the transcript snapshot and the job's
        # user_message_id binding: ONE transaction (spec §4).
        user_msg, records = await service.add_composer_operation_user_message(
            running,
            request.content,
            composition_state_id=pre_send_state_id,
            session_operation_context=lease.context,
        )
        turn_user_message_id = user_msg.id
    else:
        # compose.py:137-157: the precondition is the last CONVERSATIONAL row.
        records = await service.get_messages(session.id, limit=None)
        conversation_records = _composer_conversation_messages(records)
        if not conversation_records:
            raise HTTPException(status_code=400, detail="No messages to recompose from")
        if conversation_records[-1].role != "user":
            raise HTTPException(
                status_code=409,
                detail="Cannot recompose: the last message is not a user message. "
                "Recompose is only valid when the most recent message is the "
                "user turn whose composition failed.",
            )
        message_text = conversation_records[-1].content
        turn_user_message_id = conversation_records[-1].id
    chat_ingress = compartment_ingress_record(message_text, own_compartment_id=settings.compartment_id)
    chat_ingress_inputs = _chat_ingress_inputs(records, own_compartment_id=settings.compartment_id)
    # D15: the progress request id is the persisted user-message id.
    progress_sink = _advisory_progress_sink(
        await _composer_progress_sink_for_lease(
            services.progress_registry,
            lease=request_lease,
            session_id=str(session.id),
            request_id=str(turn_user_message_id),
            user_id=turn.actor_user_id,
        ),
        session_id=session.id,
        operation_id=turn.operation_id,
    )
    await _publish_progress(
        progress_sink,
        event=ComposerProgressEvent(
            phase="starting",
            headline=labels.starting_headline,
            evidence=(labels.starting_evidence,),
            likely_next="ELSPETH will prepare the composer prompt with the current pipeline.",
        ),
    )

    _COMPOSER_REQUESTS_INFLIGHT.add(1, {"endpoint": labels.endpoint})
    terminal_status: _ComposerRequestTerminalStatus = "failed"
    auto_title_task: asyncio.Task[None] | None = None
    try:
        if type(request) is SendMessageRequest:
            # messages.py:276-296 — Tier-1 snapshot guard, inside the try.
            send_conversation = _composer_conversation_messages(records)
            if not send_conversation or send_conversation[-1].id != turn_user_message_id:
                raise AuditIntegrityError(
                    "Tier 1 audit anomaly: send_message transcript snapshot "
                    f"for session {session.id} does not end at inserted user "
                    f"message {turn_user_message_id}. Refusing to compose against "
                    "interleaved session history."
                )
        # Exclude only the turn's own user row; keep the full transcript so the
        # history builder replays control audit rows in order (8719d8f37).
        chat_messages = _composer_chat_history([record for record in records if record.id != turn_user_message_id])

        # messages.py:305-341 — first-message auto-title, a child of the lease.
        if type(request) is SendMessageRequest and len(records) == 1 and is_default_session_title(session.title):
            auto_title_task = lease.create_task(
                maybe_auto_title_session(
                    service=service,
                    session_id=session.id,
                    user_message=request.content,
                    model=settings.composer_model,
                    temperature=settings.composer_temperature,
                    seed=settings.composer_seed,
                    session_operation_context=lease.context,
                    api_base=settings.composer_endpoint_base_url,
                    api_key=(settings.composer_endpoint_api_key.get_secret_value() if settings.composer_endpoint_api_key is not None else None),
                )
            )

        composer: ComposerService = services.composer_service
        from litellm.exceptions import APIError as LiteLLMAPIError
        from litellm.exceptions import AuthenticationError as LiteLLMAuthError

        # Review-pass F-C5: the budget is what is left of the admitted deadline
        # NOW, measured with no await between here and the provider call. It is
        # never the at-running value, which would silently extend the operation
        # past its deadline by every second of setup since `running`.
        compose_budget_seconds = budget_anchor.remaining_seconds(monotonic_now=time.monotonic())
        observation.compose_base_state_id = compose_base_state_id
        if compose_budget_seconds <= 0:
            terminal_status = "timed_out"
            await _publish_progress(progress_sink, event=convergence_progress_event(budget_exhausted="timeout"))
            raise ComposerTurnDeadlineExpired(
                session_id=turn.session_id,
                operation_id=turn.operation_id,
                remaining_seconds=compose_budget_seconds,
                budget_seconds_at_running=budget_anchor.remaining_at_running_seconds,
            )
        try:
            # Awaited inline in the turn task: attach_llm_calls rides on the
            # CancelledError instance and a task boundary would drop it. No
            # disconnect watcher — there is no socket (spec §3).
            result = await composer.compose(
                message_text,
                chat_messages,
                state,
                session_id=str(turn.session_id),
                current_state_id=str(compose_base_state_id) if compose_base_state_id is not None else None,
