```python
"""Operator and release documentation for asynchronous Composer operations.

The environment reference must document every ``composer_async_*`` setting the
live ``WebSettings`` declares, with its live default, so a new knob cannot ship
undocumented. The current release notes must name the operation routes, the
admission refusals, the unchanged guided routes and the synchronous budget that
replaced the transport-ceiling coupling.
"""

from __future__ import annotations

import tomllib
from pathlib import Path

from elspeth.web.config import WebSettings

REPO_ROOT = Path(__file__).resolve().parents[3]
ENVIRONMENT_REFERENCE = REPO_ROOT / "docs" / "reference" / "environment-variables.md"
CHANGELOG = REPO_ROOT / "CHANGELOG.md"

# The contract's five knobs (contract §Settings). Pinned as a literal so the
# live-field scan below cannot pass vacuously by matching nothing.
_ASYNC_SETTING_NAMES = (
    "composer_async_claim_lease_seconds",
    "composer_async_max_queued_operations",
    "composer_async_poll_after_ms",
    "composer_async_scan_interval_seconds",
    "composer_async_worker_concurrency",
)


def _reference_row(text: str, variable: str) -> str:
    prefix = f"| `{variable}` |"
    rows = [line for line in text.splitlines() if line.startswith(prefix)]
    assert len(rows) == 1, (variable, len(rows))
    return rows[0]


def _current_release_notes() -> str:
    version = tomllib.loads((REPO_ROOT / "pyproject.toml").read_text(encoding="utf-8"))["project"]["version"]
    changelog = CHANGELOG.read_text(encoding="utf-8")
    heading = f"\n## {version} - "
    assert changelog.count(heading) == 1
    section = changelog.split(heading, maxsplit=1)[1].split("\n## ", maxsplit=1)[0]
    return " ".join(section.split())


def test_every_async_composer_setting_is_documented_with_its_live_default() -> None:
    live_names = tuple(sorted(name for name in WebSettings.model_fields if name.startswith("composer_async_")))
    assert live_names == _ASYNC_SETTING_NAMES

    text = ENVIRONMENT_REFERENCE.read_text(encoding="utf-8")
    for name in live_names:
        row = _reference_row(text, f"ELSPETH_WEB__{name.upper()}")
        assert f"Default `{WebSettings.model_fields[name].default}`" in row, name


def test_composer_timeout_row_states_the_async_decoupling_and_the_sync_budget() -> None:
    row = _reference_row(ENVIRONMENT_REFERENCE.read_text(encoding="utf-8"), "ELSPETH_WEB__COMPOSER_TIMEOUT_SECONDS")

    assert "independent of the transport idle ceiling" in row
    assert "composer_sync_timeout_seconds" in row
    assert "/api/system/status" in row


def test_current_release_notes_describe_the_operation_contract() -> None:
    notes = _current_release_notes()

    for fragment in (
        "`POST /api/sessions/{session_id}/messages`",
        "`POST /api/sessions/{session_id}/recompose`",
        "`202 Accepted`",
        "`GET /api/sessions/{session_id}/operations/{operation_id}`",
        "`POST /api/sessions/{session_id}/operations/{operation_id}/cancel`",
        "`composer_operation_conflict`",
        "`composer_operation_active`",
        "`composer_queue_full`",
        # Review item M8: a deploy or drain cancels in-flight freeform turns immediately (spec l.200-201),
        # so the release notes must say what the user sees and does.
        "Every deploy, restart or drain of a web instance stops the freeform turns that instance is running",
        "`composer_operation_worker_lost` at once (there is no drain grace window)",
        "reloads to see what was saved, then resubmits",
        "`/guided/plan`, `/guided/respond` and `/guided/chat`",
        "composer_sync_timeout_seconds",
        "independent of the transport idle ceiling",
    ):
        assert fragment in notes, fragment
```
