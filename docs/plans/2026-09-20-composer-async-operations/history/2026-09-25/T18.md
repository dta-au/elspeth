```python
def owner_of(path: str, message: str) -> str:

def enclosing(source: pathlib.Path, line: int) -> str:
    best: tuple[int, str] = (-1, "<module>")

    def visit(node: ast.AST, prefix: str, depth: int) -> None:
        nonlocal best
        for child in ast.iter_child_nodes(node):
            if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                name = f"{prefix}{child.name}"
                inner = f"{name}.<locals>." if not isinstance(child, ast.ClassDef) else f"{name}."
                if child.lineno <= line <= (child.end_lineno or child.lineno):
                    if not isinstance(child, ast.ClassDef) and depth >= best[0]:
                        best = (depth, name)
                    visit(child, inner, depth + 1)
            else:
                visit(child, prefix, depth)

    visit(ast.parse(source.read_text(encoding="utf-8")), "", 0)
    return best[1]

        functions = sorted({enclosing(src_root / raw_path, line) for raw_path, line in raw_lines})
        if raw_lines and all(re.fullmatch(function_re, name) for name in functions):
            return label, f"{why} [in {', '.join(functions)}]"
        return "UNCLASSIFIED", f"{owner_of(path, message)}: matched a move row but sits in {functions or 'no raw line'}"

    if churn is not None and churn["bound"] in touched:

    if rule.startswith("R_TB_TESTS"):

    return "UNCLASSIFIED", owner_of(path, message)
```

```python
def _functions(tree: ast.Module) -> dict[str, list[ast.FunctionDef | ast.AsyncFunctionDef]]:
    found: dict[str, list[ast.FunctionDef | ast.AsyncFunctionDef]] = {}
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            found.setdefault(node.name, []).append(node)
    return found


def _called_names(fn: ast.AST) -> set[str]:
    names: set[str] = set()
    for node in ast.walk(fn):
        if isinstance(node, ast.Call):
            if isinstance(node.func, ast.Name):
                names.add(node.func.id)
            elif isinstance(node.func, ast.Attribute):
                names.add(node.func.attr)
    return names


def _is_cancel_column(node: ast.AST) -> bool:
    return isinstance(node, ast.Attribute) and node.attr == "cancel_requested_at"


def _sites(fn: ast.AST) -> list[tuple[str, ast.expr]]:
    sites: list[tuple[str, ast.expr]] = []
    for node in ast.walk(fn):
        if (
            isinstance(node, ast.Call)
            and isinstance(node.func, ast.Attribute)
            and node.func.attr == "is_"
            and _is_cancel_column(node.func.value)
            and len(node.args) == 1
            and isinstance(node.args[0], ast.Constant)
            and node.args[0].value is None
        ):
            sites.append(("expr", node))
        elif (
            isinstance(node, ast.Compare)
            and _is_cancel_column(node.left)
            and len(node.ops) == 1
            and isinstance(node.ops[0], ast.Eq)
            and isinstance(node.comparators[0], ast.Constant)
            and node.comparators[0].value is None
        ):
            sites.append(("expr", node))
        elif isinstance(node, ast.Constant) and isinstance(node.value, str) and SQL_NEEDLE in node.value:
            sites.append(("sql", node))
    return sites


def _offset(lines: list[bytes], lineno: int, col: int) -> int:
    return sum(len(line) for line in lines[: lineno - 1]) + col


def mutate(module: pathlib.Path, function: str, state: pathlib.Path, site_line: int | None) -> int:
    raw = module.read_bytes()
    tree = ast.parse(raw)
    functions = _functions(tree)
    if function not in functions:
        print(f"MUTATION-INSTRUMENT: no function {function!r} in {module}")
        return 2
    scope = set(functions[function])
    for fn in functions[function]:
        for name in _called_names(fn):
            scope.update(functions.get(name, []))
    sites: dict[tuple[int, int], tuple[str, ast.expr, str]] = {}
    for fn in scope:
        for form, node in _sites(fn):
            sites[(node.lineno, node.col_offset)] = (form, node, fn.name)
    for (lineno, col), (form, _node, owner) in sorted(sites.items()):
        print(f"SITE {module}:{lineno}:{col} form={form} in={owner}")
    if site_line is not None:
        sites = {key: value for key, value in sites.items() if key[0] == site_line}
    if len(sites) != 1:
        print(f"MUTATION-INSTRUMENT: expected exactly 1 site, found {len(sites)}; stop and read the listing")
        return 2
    form, node, _owner = next(iter(sites.values()))
    lines = raw.splitlines(keepends=True)
    start = _offset(lines, node.lineno, node.col_offset)
    end = _offset(lines, node.end_lineno or node.lineno, node.end_col_offset or node.col_offset)
    segment = raw[start:end]
    if form == "expr":
        replacement = b"((" + segment + b") | True)"
    else:
        replacement = segment.replace(SQL_NEEDLE.encode(), b"(" + SQL_NEEDLE.encode() + b" OR 1 = 1)")
    state.mkdir(parents=True, exist_ok=True)
    (state / "original.bytes").write_bytes(raw)
    (state / "original.sha256").write_text(hashlib.sha256(raw).hexdigest() + "\n", encoding="utf-8")
    module.write_bytes(raw[:start] + replacement + raw[end:])
    ast.parse(module.read_bytes())
    print(f"MUTATED {module}:{node.lineno}: {segment.decode()} -> {replacement.decode()}")
    return 0


def restore(module: pathlib.Path, state: pathlib.Path) -> int:
    original = (state / "original.bytes").read_bytes()
    expected = (state / "original.sha256").read_text(encoding="utf-8").strip()
    module.write_bytes(original)
    actual = hashlib.sha256(module.read_bytes()).hexdigest()
    print(f"RESTORED {module} sha256_match={actual == expected}")
    return 0 if actual == expected else 1


if __name__ == "__main__":
    mode, module_arg = sys.argv[1], pathlib.Path(sys.argv[2])
    if mode == "mutate":
        site = int(sys.argv[5].removeprefix("--site=")) if len(sys.argv) > 5 else None
        sys.exit(mutate(module_arg, sys.argv[3], pathlib.Path(sys.argv[4]), site))
    if mode == "restore":
        sys.exit(restore(module_arg, pathlib.Path(sys.argv[4])))
    print(f"unknown mode {mode!r}")
    sys.exit(2)
```

```python
"""T18 acceptance provider stub (lane-local, never committed).

An OpenAI-compatible POST .../chat/completions. A request is HELD for <hold_seconds> when its
"model" equals <hold_model>, its LAST message has role "user", and that message's text contains a
token [[hold:<id>]]. Every other request (the advisor model, tool-result turns, repair turns)
answers at once. The reply is one plain assistant message with no tool calls, so the compose
loop terminates after it. GET .../hold-control sleeps <hold_seconds> and answers 200: it is the
proxy read-timeout control. Every request and response is one JSON line in <log.jsonl>.

Usage: python stub_provider.py <port> <hold_seconds> <hold_model> <log.jsonl>
"""

from __future__ import annotations

import json
import re
import sys
import threading
import time
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

PORT = int(sys.argv[1])
HOLD_SECONDS = float(sys.argv[2])
HOLD_MODEL = sys.argv[3]
LOG_PATH = sys.argv[4]
HOLD_TOKEN = re.compile(r"\[\[hold:([A-Za-z0-9_-]{1,32})\]\]")
_lock = threading.Lock()
_counter = 0


def _log(record: dict[str, object]) -> None:
    with _lock, open(LOG_PATH, "a", encoding="utf-8") as handle:
        handle.write(json.dumps({"t": round(time.time(), 3), **record}) + "\n")


def _text(content: object) -> str:
    if isinstance(content, str):
        return content
    if isinstance(content, list):
        return " ".join(str(part.get("text", "")) for part in content if isinstance(part, dict))
    return ""


class Handler(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def log_message(self, format: str, *args: object) -> None:
        return

    def _send(self, status: int, payload: dict[str, object]) -> bool:
        body = json.dumps(payload).encode("utf-8")
        try:
            self.send_response(status)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)
            self.wfile.flush()
        except (BrokenPipeError, ConnectionResetError):
            return False
        return True

    def do_GET(self) -> None:
        if self.path.rstrip("/").endswith("/hold-control"):
            _log({"event": "hold_control_start"})
            time.sleep(HOLD_SECONDS)
            _log({"event": "hold_control_end", "delivered": self._send(200, {"held_seconds": HOLD_SECONDS})})
            return
        self._send(404, {"error": {"message": "not found"}})

    def do_POST(self) -> None:
        global _counter
        length = int(self.headers.get("Content-Length", "0"))
        request = json.loads(self.rfile.read(length) or b"{}")
        if not self.path.rstrip("/").endswith("/chat/completions"):
            self._send(404, {"error": {"message": "the stub serves /chat/completions only"}})
            return
        with _lock:
            _counter += 1
            call = _counter
        messages = request.get("messages") or []
        last = messages[-1] if messages else {}
        match = HOLD_TOKEN.search(_text(last.get("content")))
        hold_id = match.group(1) if match else None
        model = request.get("model")
        held = hold_id is not None and last.get("role") == "user" and model == HOLD_MODEL
        _log({
            "event": "request", "call": call, "path": self.path, "model": model, "messages": len(messages),
            "last_role": last.get("role"), "hold_id": hold_id, "held": held, "has_tools": bool(request.get("tools")),
            "has_response_format": "response_format" in request, "stream": bool(request.get("stream")),
        })
        started = time.monotonic()
        if held:
            time.sleep(HOLD_SECONDS)
        reply = f"STUB-REPLY call={call} hold={hold_id or '-'}"
        delivered = self._send(200, {
            "id": f"chatcmpl-stub-{call}",
            "object": "chat.completion",
```

```javascript
"use strict";
// T18 local acceptance driver (lane-local, never committed).
// Usage: node accept.cjs <A|B|C> <sessionId> <out.jsonl> <screenshot-prefix>
// Env: FRONTEND_DIR (the worktree frontend dir), ACCEPT_ORIGIN (the nginx origin), ACCEPT_TOKEN (bearer token).
// A: send; the provider holds 60 s; the turn completes.
// B: send; Stop while running -> request_cancelled; then Retry in the SAME page -> POST /recompose; completes.
// C: send; reload while running; the same operation id resumes without a resubmit; completes.
const fs = require("node:fs");
const path = require("node:path");
const { chromium } = require(path.join(process.env.FRONTEND_DIR, "node_modules", "playwright"));

const ORIGIN = process.env.ACCEPT_ORIGIN;
const TOKEN = process.env.ACCEPT_TOKEN;
const [scenario, sessionId, outPath, shotPrefix] = process.argv.slice(2);
const SUBMIT_PATH = /^\/api\/sessions\/[0-9a-f-]{36}\/(messages|recompose)$/;
const OP_PATH = /^\/api\/sessions\/[0-9a-f-]{36}\/operations\/([0-9a-f-]{36})(\/cancel)?$/;
const HOLD_FLOOR_MS = 55_000;
const t0 = Date.now();
const events = [];

function log(record) {
  const rec = { t_ms: Date.now() - t0, scenario, ...record };
  events.push(rec);
  fs.appendFileSync(outPath, JSON.stringify(rec) + "\n");
}

function fail(message) {
  log({ kind: "assertion_failed", message });
  throw new Error(message);
}

async function waitForEvent(predicate, timeoutMs, label) {
  const deadline = Date.now() + timeoutMs;
  while (Date.now() < deadline) {
    const hit = events.find(predicate);
    if (hit) return hit;
    await new Promise((resolve) => setTimeout(resolve, 200));
  }
  return fail(`timed out after ${timeoutMs} ms waiting for ${label}`);
}

async function runTurn(page, input, { phase, trigger, route, mode }) {
  const since = events.length;
  const fresh = (predicate) => (e) => events.indexOf(e) >= since && predicate(e);
  await trigger();
  const submit = await waitForEvent(fresh((e) => e.kind === "submit"), 15_000, `${phase}: the submit response`);
  if (submit.route !== route) fail(`${phase}: submit went to /${submit.route}, expected /${route}`);
  if (submit.http !== 202) fail(`${phase}: submit answered HTTP ${submit.http}, expected 202`);
  if (submit.ms > 2_000) fail(`${phase}: submit took ${submit.ms} ms, expected under 2000 ms`);
  if (submit.operation_id !== submit.request_operation_id) fail(`${phase}: 202 operation_id differs from the request body's operation_id`);
  const opId = submit.operation_id;
  const running = await waitForEvent(fresh((e) => e.kind === "poll" && e.operation_id === opId && e.status === "running"), 30_000, `${phase}: a running poll`);
  if (running.cache_control !== "no-store") fail(`${phase}: poll Cache-Control was ${running.cache_control}, expected no-store`);
  if (mode === "stop") {
    await page.getByRole("button", { name: "Stop composing" }).click();
    const cancel = await waitForEvent(fresh((e) => e.kind === "cancel"), 15_000, `${phase}: the cancel response`);
    if (cancel.http !== 202 && cancel.http !== 200) fail(`${phase}: cancel answered HTTP ${cancel.http}`);
  }
  if (mode === "reload") {
    const reloadAt = Date.now() - t0;
    log({ kind: "reload", phase });
    await page.reload();
    await input.waitFor({ state: "visible", timeout: 30_000 });
    await waitForEvent(fresh((e) => e.kind === "poll" && e.operation_id === opId && e.t_ms > reloadAt), 20_000, `${phase}: a poll of the same operation after reload`);
  }
  const terminal = await waitForEvent(
    fresh((e) => e.kind === "poll" && e.operation_id === opId && (e.status === "completed" || e.status === "failed")),
    150_000,
    `${phase}: a terminal poll`,
  );
  const turnEvents = events.slice(since);
  const submits = turnEvents.filter((e) => e.kind === "submit").length;
  if (submits !== 1) fail(`${phase}: expected exactly one submit, saw ${submits}`);
  const gatewayErrors = turnEvents.filter((e) => e.http >= 502 && e.http <= 504).length;
  if (gatewayErrors > 0) fail(`${phase}: saw ${gatewayErrors} gateway error responses`);
  const elapsedMs = terminal.t_ms - submit.t_ms;
  if (mode === "stop") {
    if (terminal.status !== "failed" || terminal.error?.failure_code !== "request_cancelled" || terminal.error?.http_status !== 499) {
      fail(`${phase}: terminal was ${JSON.stringify({ status: terminal.status, error: terminal.error })}`);
    }
  } else {
    if (terminal.status !== "completed") fail(`${phase}: terminal status ${terminal.status}, expected completed`);
    if (!/^STUB-REPLY /.test(terminal.result_content ?? "")) fail(`${phase}: result content ${terminal.result_content}`);
    if (elapsedMs < HOLD_FLOOR_MS) fail(`${phase}: terminal after ${elapsedMs} ms; the provider hold was not observed`);
    await page.getByText(terminal.result_content).first().waitFor({ state: "visible", timeout: 15_000 });
  }
  await page.getByRole("button", { name: "Stop composing" }).waitFor({ state: "hidden", timeout: 15_000 });
  await page.screenshot({ path: `${shotPrefix}-${phase}.png` });
  const summary = {
    phase,
    operation_id: opId,
    submit_route: submit.route,
    submit_http: submit.http,
    submit_ms: submit.ms,
    polls: turnEvents.filter((e) => e.kind === "poll" && e.operation_id === opId).length,
    terminal_status: terminal.status,
    terminal_error: terminal.error,
    elapsed_to_terminal_ms: elapsedMs,
  };
  log({ kind: "summary", ...summary });
  console.log(`SUMMARY ${JSON.stringify(summary)}`);
}

async function main() {
  const browser = await chromium.launch({ headless: true });
  const context = await browser.newContext();
  await context.addInitScript((token) => window.localStorage.setItem("auth_token", token), TOKEN);
  const page = await context.newPage();
  const sentAt = new Map();
  page.on("request", (req) => sentAt.set(req, Date.now()));
  page.on("requestfailed", (req) => {
    const p = new URL(req.url()).pathname;
    if (p.startsWith("/api/sessions/")) log({ kind: "requestfailed", method: req.method(), path: p, failure: req.failure()?.errorText ?? null });
  });
  page.on("response", async (res) => {
    const req = res.request();
    const p = new URL(req.url()).pathname;
    const submit = req.method() === "POST" ? SUBMIT_PATH.exec(p) : null;
    const op = OP_PATH.exec(p);
    if (!submit && !op) return;
    const rec = { kind: submit ? "submit" : op[2] ? "cancel" : "poll", method: req.method(), path: p, http: res.status(), ms: Date.now() - (sentAt.get(req) ?? Date.now()) };
    if (submit) {
      rec.route = submit[1];
      rec.request_operation_id = req.postDataJSON()?.operation_id ?? null;
    } else {
      rec.operation_id = op[1];
    }
    try {
      const body = await res.json();
      if (submit) rec.operation_id = body.operation_id ?? body.detail?.operation_id ?? null;
      rec.status = body.status ?? null;
      rec.cancel_requested = body.cancel_requested ?? null;
      rec.error = body.error ? { http_status: body.error.http_status, failure_code: body.error.failure_code, error_type: body.error.error_type } : null;
      rec.result_content = body.result ? body.result.message.content : null;
      rec.cache_control = res.headers()["cache-control"] ?? null;
    } catch (err) {
      rec.body = `unparsed: ${String(err)}`;
    }
    log(rec);
  });

  await page.goto(`${ORIGIN}/#/${sessionId}`);
  const input = page.getByRole("textbox", { name: "Message input" });
  await input.waitFor({ state: "visible", timeout: 30_000 });
  const send = (text) => async () => {
    await input.fill(text);
    await page.getByRole("button", { name: "Send message" }).click();
  };
  if (scenario === "A") {
    await runTurn(page, input, { phase: "A", trigger: send("T18 scenario A: please just say hello. [[hold:A]]"), route: "messages", mode: "complete" });
  } else if (scenario === "B") {
    await runTurn(page, input, { phase: "B-stop", trigger: send("T18 scenario B: please just say hello. [[hold:B]]"), route: "messages", mode: "stop" });
    const retry = page.getByRole("button", { name: "Retry" }).first();
    await retry.waitFor({ state: "visible", timeout: 15_000 });
    await runTurn(page, input, { phase: "B-retry", trigger: () => retry.click(), route: "recompose", mode: "complete" });
  } else if (scenario === "C") {
    await runTurn(page, input, { phase: "C", trigger: send("T18 scenario C: please just say hello. [[hold:C]]"), route: "messages", mode: "reload" });
  } else {
    fail(`unknown scenario ${scenario}`);
  }
  await browser.close();
}

main().catch((err) => {
  console.error(`ACCEPTANCE-FAILED ${err.message}`);
  process.exit(1);
});
```
