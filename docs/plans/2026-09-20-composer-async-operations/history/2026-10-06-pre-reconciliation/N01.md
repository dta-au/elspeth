def _accepted_budget_triples(draw: st.DrawFn) -> tuple[float, float, float]:
    """(ceiling, headroom, timeout) triples the current coupling validator accepts.

    ``headroom`` lies strictly inside (0, ceiling), so ``ceiling - headroom`` is
    a positive float: finite IEEE subtraction of distinct values is never zero.
    ``timeout`` ranges over (0, ceiling - headroom] INCLUDING the boundary, which
    the validator accepts because it rejects only ``>``.

    ceiling = draw(st.floats(min_value=1e-3, max_value=1e6, allow_nan=False, allow_infinity=False))

    def composer_sync_timeout_seconds(self) -> float:
        """The budget a SYNCHRONOUS Composer consumer may spend.

        A consumer that answers inside one held HTTP request must finish
        before the smallest idle timeout in front of this process, so it gets
        the smaller of the compose budget and the transport-safe budget
