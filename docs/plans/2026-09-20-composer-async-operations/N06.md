CONTRACT DEVIATION: **N06 appends one select-only, connection-taking reader to N05's module**
`coordination/composer_operation_authority.py`: `lock_composer_operation_for_start_on_connection(conn, claim)`.

- **Reason.** Contract §Composite start step 1 has the composite "re-read the job", and §Table › Policy says "All SQL
  lives in the authority module; the composites call its connection-taking helpers". The contract names three
  connection-taking helpers (start, bind, settle), and none of them reads. `ComposerAsyncOperationAuthority.get` cannot
  serve, for three reasons:
  - it opens its own connection, and on SQLite `StaticPool` that is the connection the composite already holds;
  - contract §Owned types says poll reads load `request_json` as `None`, and recompose needs that body (E9);
  - it takes no row lock.
- **Why the token is compared in SQL.** `ComposerOperationRecord` carries `claim_token_present`, not the token. So the
  reader's `WHERE` names the claim token and attempt, and a missing row, a foreign token and a stale attempt all return
  `None`. The composite reports each of these as `ComposerOperationFenceLost`.
- **Smallest change.** A reader with no DML. The writer manifest gains no row; Step 11 measures this.

                break
            for claim in claims:
                if claim.session_id == session_id and claim.operation_id == operation_id:
                    mine.append(claim)
                else:
                    foreign.append(claim)
    finally:
        for claim in foreign:
            jobs.release_claim(claim)
    assert len(mine) == 1
    return mine[0]


def _age_job(engine: Engine, session_id: UUID, operation_id: str, *, by: timedelta) -> None:
    """Move every time column of one queued claimed job back together.

    Shifting created/updated/deadline/claim-expiry by the same amount keeps
    every time-ordering CHECK on the row satisfied while putting the claim
    expiry behind the database clock. Admit with a deadline longer than ``by``
    so only the claim lapses.
    """
    with engine.begin() as conn:
        row = conn.execute(
            select(
                composer_async_operations_table.c.created_at,
                composer_async_operations_table.c.updated_at,
                composer_async_operations_table.c.deadline_at,
                composer_async_operations_table.c.claim_expires_at,
            ).where(
                composer_async_operations_table.c.session_id == str(session_id),
                composer_async_operations_table.c.operation_id == operation_id,
            )
        ).one()
        conn.execute(
            update(composer_async_operations_table)
            .where(
                composer_async_operations_table.c.session_id == str(session_id),
                composer_async_operations_table.c.operation_id == operation_id,
            )
            .values(
                created_at=row.created_at - by,
                updated_at=row.updated_at - by,
                deadline_at=row.deadline_at - by,
                claim_expires_at=row.claim_expires_at - by,
            )
        )


def _job(engine: Engine, session_id: UUID, operation_id: str) -> RowMapping:

async def _arrange_refusal(rig: _Rig, case: str, monkeypatch: pytest.MonkeyPatch) -> tuple[str, str, ComposerOperationError]:
    """Admit the job for one E5 refusal; return (operation_id, auth_provider_type, expected terminal error)."""
    if case == "ownership_actor":
        return _admit(rig, actor_user_id=_INTRUDER), "local", _SESSION_NOT_FOUND
    if case == "ownership_provider":
        return _admit(rig), "oidc", _SESSION_NOT_FOUND
    if case in {"foreign_base", "unknown_base"}:
        if case == "foreign_base":
            other = await rig.service.create_session(_OWNER, "foreign base owner", "local")
            forged_base = await _seed_head(rig, other.id)
        else:
            forged_base = uuid4()
        real_reader = coordination_repository.lock_composer_operation_for_start_on_connection

        def forged_reader(conn: Connection, claim: ComposerOperationClaim) -> ComposerOperationRecord | None:
            # E19's composite FK job->composition_states refuses a foreign or unknown base at
            # admission, so the gate's 404 arm is defence in depth: forge the base on the
            # record the composite reads.
            record = real_reader(conn, claim)
            assert record is not None
            return replace(record, base_state_id=forged_base)

        monkeypatch.setattr(coordination_repository, "lock_composer_operation_for_start_on_connection", forged_reader)
        return _admit(rig), "local", _STATE_NOT_FOUND
    if case == "moved_base":
        seen = await _seed_head(rig)
        operation_id = _admit(rig, base_state_id=seen)
        await _seed_head(rig)  # another tab's Accept, revert or import moves the head while the job waits
        return operation_id, "local", _STALE_BASE
    if case == "absent_base_with_head":
        await _seed_head(rig)
        return _admit(rig, base_state_id=None), "local", _STALE_BASE  # ruling 7: absent means "no state"
    if case == "recompose_no_conversation":
        await _seed_message(rig, "audit", "Provider call result recorded.", tool_calls=[{"_kind": "llm_call_audit"}])
        return _admit(rig, kind="compose_recompose"), "local", _NO_MESSAGES
    if case == "recompose_last_not_user":
        user = await _seed_message(rig, "user", "Build it.")
        await _seed_message(rig, "assistant", "Here is the pipeline.")
        return _admit(rig, kind="compose_recompose", expected_user_message_id=user), "local", _LAST_NOT_USER
    if case == "recompose_user_message_mismatch":
        await _seed_message(rig, "user", "Build it.")
        return _admit(rig, kind="compose_recompose", expected_user_message_id=uuid4()), "local", _USER_MESSAGE_MISMATCH
    raise AssertionError(f"unknown refusal case {case!r}")

def _functions(tree: ast.Module) -> Iterator[tuple[str, ast.FunctionDef | ast.AsyncFunctionDef]]:
    stack: list[tuple[str, ast.AST]] = [("", tree)]
    while stack:
        prefix, node = stack.pop()
        for child in ast.iter_child_nodes(node):
            if isinstance(child, ast.ClassDef):
                stack.append((f"{prefix}{child.name}.", child))
            elif isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                yield f"{prefix}{child.name}", child
                stack.append((f"{prefix}{child.name}.", child))


def _own_nodes(function: ast.AST) -> Iterator[ast.AST]:
    """Walk one function body without descending into nested definitions (they are walked as their own qualname)."""
    stack = list(ast.iter_child_nodes(function))
    while stack:
        node = stack.pop()
        yield node
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            stack.extend(ast.iter_child_nodes(node))


def _callee_name(func_node: ast.expr) -> str | None:
    if isinstance(func_node, ast.Name):
        return func_node.id
    if isinstance(func_node, ast.Attribute):
        return func_node.attr
    return None


def _dml_table(call: ast.Call) -> str | None:
    """The table at the root of an ``update(t)…values(…)`` or ``t.update()…values(…)`` chain."""
    current: ast.expr = call.func
    while True:
        if isinstance(current, ast.Attribute):
            if current.attr in {"update", "insert"} and isinstance(current.value, ast.Name):
                return current.value.id
            current = current.value
        elif isinstance(current, ast.Call):
            if isinstance(current.func, ast.Name) and current.func.id in {"update", "insert"}:
                return current.args[0].id if current.args and isinstance(current.args[0], ast.Name) else None
            current = current.func
        else:
            return None


def _module_running_aliases(tree: ast.Module) -> frozenset[str]:
    aliases: set[str] = set()
    for statement in tree.body:
        if isinstance(statement, ast.Assign) and isinstance(statement.value, ast.Constant) and statement.value.value == "running":
            aliases.update(target.id for target in statement.targets if isinstance(target, ast.Name))
        elif (
            isinstance(statement, ast.AnnAssign)
            and isinstance(statement.target, ast.Name)
            and isinstance(statement.value, ast.Constant)
            and statement.value.value == "running"
        ):
            aliases.add(statement.target.id)
    return frozenset(aliases)


def _is_running(value: ast.expr | None, aliases: frozenset[str]) -> bool:
    return (isinstance(value, ast.Constant) and value.value == "running") or (isinstance(value, ast.Name) and value.id in aliases)


def _running_status_writers(trees: Iterable[tuple[str, ast.Module]]) -> set[tuple[str, str]]:
    """Every function that writes ``status='running'`` into ``composer_async_operations``."""
    writers: set[tuple[str, str]] = set()
    for path, tree in trees:
        aliases = _module_running_aliases(tree)
        for qualname, function in _functions(tree):
            for node in _own_nodes(function):
                if not (isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute) and node.func.attr == "values"):
                    continue
                if _dml_table(node) != "composer_async_operations_table":
                    continue
                keyword_hit = any(keyword.arg == "status" and _is_running(keyword.value, aliases) for keyword in node.keywords)
                dict_hit = any(
                    isinstance(argument, ast.Dict)
                    and any(
                        isinstance(key, ast.Constant) and key.value == "status" and _is_running(value, aliases)
                        for key, value in zip(argument.keys, argument.values, strict=True)
                    )
                    for argument in node.args
                )
                if keyword_hit or dict_hit:
                    writers.add((path, qualname))
    return writers


def _running_cas_callers(trees: Iterable[tuple[str, ast.Module]]) -> set[tuple[str, str]]:
    """Every function that calls N05's queued->running helper."""
    callers: set[tuple[str, str]] = set()
    for path, tree in trees:
        for qualname, function in _functions(tree):
            for node in _own_nodes(function):
                if isinstance(node, ast.Call) and _callee_name(node.func) == "start_composer_operation_on_connection":
                    callers.add((path, qualname))
    return callers


def _gate_trees() -> list[tuple[str, ast.Module]]:
    return [(parsed.path.relative_to(_REPO_ROOT).as_posix(), parsed.tree) for parsed in iter_gate_sources(_REPO_ROOT / "src" / "elspeth")]

def composer_conversation_tail_on_connection(conn: Connection, *, session_id: UUID) -> ComposerConversationTail | None:
    """Return the session's last conversational chat row, or ``None`` when it has none."""
    row = conn.execute(
        select(chat_messages_table.c.id, chat_messages_table.c.role)
        .where(
            chat_messages_table.c.session_id == str(session_id),
            chat_messages_table.c.role.not_in(sorted(COMPOSER_NON_CONVERSATIONAL_ROLES)),
        )
        .order_by(desc(chat_messages_table.c.sequence_no))
        .limit(1)
    ).one_or_none()
    if row is None:
        return None
    if row.role not in _CONVERSATIONAL_ROLES:
        raise AuditIntegrityError(f"Tier 1: chat_messages.role is {row.role!r}, expected a conversational role")
    return ComposerConversationTail(message_id=UUID(row.id), role=row.role)


def recompose_expected_user_message_id(job: ComposerOperationRecord) -> UUID:
    """The ``expected_user_message_id`` a recompose job was admitted with, re-validated from its stored body."""

        raise TypeError("job must be an exact ComposerOperationRecord")

        raise ValueError("only a compose_recompose job carries an expected user message id")
    if job.request_json is None:
        raise AuditIntegrityError("Tier 1: a recompose job read for start has no stored request body")

        session is None
        or session.archived_at is not None
        or session.user_id != job.actor_user_id
        or session.auth_provider_type != auth_provider_type

        raise ComposerOperationPreconditionRefused(error=_string_detail_error(404, "Session not found"))

        select(composition_states_table.c.id)
        .where(composition_states_table.c.session_id == session_id_text)
        .order_by(desc(composition_states_table.c.version))
        .limit(1)
    ).one_or_none()
    head_state_id = None if head is None else UUID(head.id)
    if head_state_id != job.base_state_id:
        raise ComposerOperationPreconditionRefused(error=stale_base_error(request_id=job.request_id))

def _race(calls: tuple[Callable[[], object], Callable[[], object]]) -> list[object]:
    barrier = Barrier(2)

    def run(call: Callable[[], object]) -> object:
        barrier.wait(timeout=10)
        try:
            return call()
        except (ComposerOperationFenceLost, ComposerOperationCancelledBeforeStart, SessionOperationConflictError) as exc:
            return exc

    with ThreadPoolExecutor(max_workers=2) as pool:
        return list(pool.map(run, calls))

2. **Deadline via real time.** The SQLite deadline test sleeps 3.1 s against a 2 s deadline, because SQLite's clock
   has one-second resolution. The claim is taken immediately after admission, with at least 1 s of margin. On a
   heavily loaded host the admit→claim gap could exceed that margin and the claim scan would skip the row. The test
   then fails at `_claim_exact` (`assert len(mine) == 1`), never with a false green. This was chosen over UPDATE-ageing
   the deadline, because N04's time-ordering CHECK set is unknown here.
